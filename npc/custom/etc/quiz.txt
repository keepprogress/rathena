//===== rAthena Script =======================================
//= 知識猜謎
//===== Description: =========================================
//= 奇數小時 :20 全服搶答 10 題，每題 30 秒，打 @ans 數字 作答，每題只能答一次。
//= 最先答對拿 10 商城點數，其他答對的人拿 2 點。GM 手動開：@quiz
//= 題目一半出自下面的題庫，一半當場從魔物資料出（屬性、種族、體型、卡片、等級），
//= 所以答案一定跟這個伺服器的資料一致。
//= .mob 是出題用的魔物：野外有生 20 隻以上、99 級以下、非 MVP、中文名不重複，
//= 從 npc/re/mobs 與 db/import/mob_db.yml 篩出來的。
//============================================================

-	script	KnowledgeQuiz	-1,{
OnMinute20:
	if (gettime(DT_HOUR) % 2 == 0)
		end;
OnStart:
	detachrid;	// @quiz 會帶著 GM，GM 下線不能讓活動跟著中斷
	if (.running)
		end;
	.running = 1;
	announce "[知識猜謎] 10 秒後開始！共 " + .questions + " 題，每題 " + .time + " 秒，打 @ans 數字 作答，每題只能答一次。最先答對 " + .pt_first + " 點，其他答對 " + .pt_other + " 點商城點數。", bc_all;
	sleep 10000;
	for (.@q = 1; .@q <= .questions; .@q++) {
		callsub L_Make;
		.first$ = "";
		.right = 0;
		.qid = gettimetick(2) * 100 + .@q;
		.@o$ = "";
		for (.@i = 0; .@i < getarraysize(.opt$); .@i++)
			.@o$ += (.@i ? "　" : "") + (.@i + 1) + ". " + .opt$[.@i];
		announce "[知識猜謎] 第 " + .@q + " 題：" + .ask$, bc_all, 0x00FFFF;
		announce "[知識猜謎] " + .@o$ + "（@ans 數字）", bc_all, 0x00FFFF;
		sleep .time * 1000;
		.qid = 0;
		announce "[知識猜謎] 答案是 " + (.ans + 1) + ". " + .opt$[.ans] + (.first$ == "" ? "，沒有人答對。" : "。最先答對：" + .first$ + "，共 " + .right + " 人答對。"), bc_all, 0x00FFFF;
		sleep 5000;
	}
	announce "[知識猜謎] 本場結束，下次再來！", bc_all, 0x00FFFF;
	.running = 0;
	end;

OnAns:
	if (!.qid) {
		dispbottom "[知識猜謎] 現在沒有題目。";
		end;
	}
	.@a = atoi(.@atcmd_parameters$[0]) - 1;
	if (.@a < 0 || .@a >= getarraysize(.opt$)) {
		dispbottom "[知識猜謎] 請打 @ans 加選項數字，例如 @ans 2";
		end;
	}
	if (#quiz_qid == .qid) {
		dispbottom "[知識猜謎] 這題你已經答過了。";
		end;
	}
	#quiz_qid = .qid;
	if (.@a != .ans) {
		dispbottom "[知識猜謎] 答錯了！";
		end;
	}
	.right++;
	.@p = .pt_other;
	if (.first$ == "") {
		.first$ = strcharinfo(0);
		.@p = .pt_first;
	}
	#CASHPOINTS += .@p;
	dispbottom "[知識猜謎] 答對了！獲得 " + .@p + " 商城點數，目前共 " + #CASHPOINTS + " 點。";
	end;

// 出一題：.ask$ 題目、.opt$[] 選項、.ans 正解的索引
// 先把正解放 .@c$[0]、錯的接在後面，最後再洗牌
L_Make:
	.@t = rand(10);	// 0-3 題庫、4-5 屬性、6 種族、7 體型、8 卡片、9 等級
	if (.@t == 8) {	// 卡片：找一隻會掉卡片的魔物
		for (.@try = 0; .@try < 20 && !.@card; .@try++) {
			.@m = .mob[rand(getarraysize(.mob))];
			getmobdrops(.@m);
			for (.@i = 0; .@i < $@MobDrop_count; .@i++)
				if (getiteminfo($@MobDrop_item[.@i], ITEMINFO_TYPE) == IT_CARD)
					.@card = $@MobDrop_item[.@i];
		}
		if (!.@card)
			.@t = 4;
	}
	if (.@t < 4) {
		explode(.@c$, .Q$[rand(getarraysize(.Q$))], "|");
		.ask$ = .@c$[0];
		deletearray .@c$[0], 1;
	} else if (.@t == 8) {
		.ask$ = "「" + getitemname(.@card) + "」是哪一隻魔物掉的？";
		.@c$[0] = getmonsterinfo(.@m, MOB_NAME);
		while (getarraysize(.@c$) < 4) {
			.@s$ = getmonsterinfo(.mob[rand(getarraysize(.mob))], MOB_NAME);
			if (inarray(.@c$, .@s$) < 0)
				.@c$[getarraysize(.@c$)] = .@s$;
		}
	} else if (.@t == 9) {	// 等級：四隻魔物比等級，等級不能重複
		.ask$ = "下面哪一隻魔物的等級最高？";
		while (getarraysize(.@lv) < 4) {
			.@m = .mob[rand(getarraysize(.mob))];
			if (inarray(.@lv, getmonsterinfo(.@m, MOB_LV)) < 0) {
				.@mm[getarraysize(.@lv)] = .@m;
				.@lv[getarraysize(.@lv)] = getmonsterinfo(.@m, MOB_LV);
			}
		}
		.@top = 0;
		for (.@i = 1; .@i < 4; .@i++)
			if (.@lv[.@i] > .@lv[.@top])
				.@top = .@i;
		.@c$[0] = getmonsterinfo(.@mm[.@top], MOB_NAME);
		for (.@i = 0; .@i < 4; .@i++)
			if (.@i != .@top)
				.@c$[getarraysize(.@c$)] = getmonsterinfo(.@mm[.@i], MOB_NAME);
	} else {
		.@m = .mob[rand(getarraysize(.mob))];
		.@name$ = getmonsterinfo(.@m, MOB_NAME);
		if (.@t < 6) {
			.ask$ = "「" + .@name$ + "」是什麼屬性？";
			.@c$[0] = .ele$[getmonsterinfo(.@m, MOB_ELEMENT)];
			copyarray .@pool$[0], .ele$[0], getarraysize(.ele$);
		} else if (.@t == 7) {
			.ask$ = "「" + .@name$ + "」是什麼體型？";
			.@c$[0] = .size$[getmonsterinfo(.@m, MOB_SIZE)];
			copyarray .@pool$[0], .size$[0], getarraysize(.size$);
		} else {
			.ask$ = "「" + .@name$ + "」是什麼種族？";
			.@c$[0] = .race$[getmonsterinfo(.@m, MOB_RACE)];
			copyarray .@pool$[0], .race$[0], getarraysize(.race$);
		}
		while (getarraysize(.@c$) < min(4, getarraysize(.@pool$))) {
			.@s$ = .@pool$[rand(getarraysize(.@pool$))];
			if (inarray(.@c$, .@s$) < 0)
				.@c$[getarraysize(.@c$)] = .@s$;
		}
	}
	// 洗牌，記下正解（原本在 0）洗到哪
	deletearray .opt$;
	.@n = getarraysize(.@c$);
	.@k = rand(.@n);
	.opt$[.@k] = .@c$[0];
	.ans = .@k;
	.@j = 1;
	for (.@i = 0; .@i < .@n; .@i++)
		if (.@i != .@k)
			.opt$[.@i] = .@c$[.@j++];
	return;
