//===== rAthena Script =======================================
//= 活動引導員
//===== Description: =========================================
//= 列出接下來的活動時間，選一個看怎麼參加，並在小地圖標出入口。
//= 活動時間寫在 OnInit 的 .s_*，改了活動時間記得一起改（還有 event_notice.txt）。
//============================================================

prontera,160,186,4	script	活動引導員	4_F_KAFRA2,{
	.@now = gettime(DT_HOUR) * 60 + gettime(DT_MINUTE);
	// 每一場算還有幾分鐘，再照時間排序
	.@n = getarraysize(.s_ev);
	for (.@i = 0; .@i < .@n; .@i++) {
		.@d[.@i] = ((.s_off[.@i] - .@now) % .s_per[.@i] + .s_per[.@i]) % .s_per[.@i];
		.@o[.@i] = .@i;
	}
	for (.@i = 0; .@i < .@n; .@i++)
		for (.@j = .@i + 1; .@j < .@n; .@j++)
			if (.@d[.@o[.@j]] < .@d[.@o[.@i]]) {
				.@t = .@o[.@i];
				.@o[.@i] = .@o[.@j];
				.@o[.@j] = .@t;
			}
	mes "[活動引導員]";
	mes "現在 " + sprintf("%02d:%02d", .@now / 60, .@now % 60) + "，接下來的活動：";
	for (.@i = 0; .@i < .@n; .@i++) {
		.@k = .@o[.@i];
		.@at = (.@now + .@d[.@k]) % 1440;
		mes "^0055FF" + sprintf("%02d:%02d", .@at / 60, .@at % 60) + "^000000 " + .name$[.s_ev[.@k]] + (.@d[.@k] ? "^777777（再 " + .@d[.@k] + " 分）^000000" : "^FF0000（現在）^000000");
	}
	mes "隨時可玩：MVP 天梯、狩獵任務、每日簽到。";
	next;
	while (1) {
		.@e = select(implode(.name$, ":") + ":離開") - 1;
		if (.@e >= getarraysize(.name$))
			close;
		mes "[活動引導員]";
		mes "^FF0000" + .name$[.@e] + "^000000";
		explode(.@line$, .desc$[.@e], "|");
		for (.@i = 0; .@i < getarraysize(.@line$); .@i++)
			mes .@line$[.@i];
		if (.x[.@e]) {
			viewpoint 1, .x[.@e], .y[.@e], .@e, 0xFF0000;
			mes "入口 (" + .x[.@e] + "," + .y[.@e] + ") 已經標在小地圖上了。";
		}
		next;
	}

OnInit:
	// 活動：名字、入口座標（0 = 沒有固定入口）、說明（| 換行）
	setarray .name$, "惡魔廣場", "尋找蘑菇", "波利大作戰", "知識猜謎", "咯咯雞", "MVP 天梯", "狩獵任務", "每日簽到";
	setarray .x, 156, 0, 165, 0, 156, 164, 152, 0;
	setarray .y, 185, 0, 195, 0, 219, 171, 187, 0;
	setarray .desc$,
		"偶數整點開放報名 5 分鐘，找守衛進場。|限 70 級以上，報名費 20,000z。|三波怪物打完開 10 個寶箱，打破寶箱 20 商城點數。",
		"每小時 :10，蘑菇出現在依斯魯得、吉芬、夢羅克或普隆德拉（看公告）。|每打倒一隻 5 商城點數。",
		"偶數小時 :40 開放報名 2 分鐘，找入口波利進場。|大家一起點波利，每關點滿 10 下拿 關數×3 點，|5 關全過 45 點，最後一擊再 50 點。",
		"奇數小時 :20，全服搶答 10 題，不用報名。|打 @ans 數字 作答，每題只能答一次，|最先答對 10 點，其他答對 2 點。",
		"每天 15:50、21:50，到噴水池後面一起點雞。|會被炸死或冰凍，普隆德拉死亡不掉經驗。|先擠出獎品的人免費抽最新一期轉蛋 10 次。",
		"隨時，組隊由隊長報名，入場費 100,000z。|依序打倒 MVP，每關 5 點、全破 50 點，限時內全破再 30 點。",
		"每 12 小時接一次，打完指定魔物回來交。|給經驗、zeny 與商城點數。",
		"每天登入會跳出簽到視窗，點今天的格子領獎，|道具寄到信箱，記得打開信箱領取。|每月 20 天，斷簽不歸零；每天 10 商城點數，每 5 天再加 30 點。";
	// 排程：哪個活動、每幾分鐘一場、一天中的第幾分鐘開始（咯咯雞一天兩場分開寫）
	setarray .s_ev, 0, 1, 2, 3, 4, 4;
	setarray .s_per, 120, 60, 120, 120, 1440, 1440;
	setarray .s_off, 0, 10, 40, 80, 950, 1310;
	end;
}
