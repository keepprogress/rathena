-	script	::merchant_13_2	-1,{
	set .@n$, "["+strnpcinfo(1)+"]";
	mes .@n$;
	if (isequipped(2782) && ep13_2_rhea == 100) {
		if (!checkweight(1201,1)) {
			mes "看來你帶的東西太多了。";
			mes "何不把一些道具存進倉庫再回來呢？";
			close;
		}
		mes "你好。";
		mes "你在找什麼？";
		next;
		if (strnpcinfo(1) == "Merchant of Manuk") {
			setarray .@Items[2],12342,12343,12348;
			set .@Book,11019; //Manuk_Selling_Item
			set .@Coin,6080; //Manuk_Coin
			set .@i, select("查看道具說明:魔怒克的良機:魔怒克的意志:魔怒克的信念:取消");
		}
		else {
			setarray .@Items[2],12344,12345,12349;
			set .@Book,11018; //Splendide_Selling_Item
			set .@Coin,6081; //Splendide_Coin
			set .@i, select("查看道具說明:購買捕蟲堇妖果醬:購買魯修拉的蜜醬:購買獨角飛馬之淚:什麼都不做");
		}
		mes .@n$;
		if (.@i == 1) {
			mes "這是道具的說明。";
			close2;
			readbook .@Book,1;
			end;
		}
		else if (.@i < 5) {
			mes "我可以賣你 3 個 "+getitemname(.@Items[.@i])+" 售價為 ^3131FF1 枚金幣^000000。";
			next;
			switch(select("購買:不購買")) {
			case 1:
				if (countitem(.@Coin)) {
					mes .@n$;
					mes "感謝光臨。";
					delitem .@Coin,1; 
					getitem .@Items[.@i],3;
					close;
				}
				else {
					mes .@n$;
					mes "你的金幣不夠。";
					close;
				}
			case 2:
				mes .@n$;
				break;
			}
		}
		mes "如果你改變心意，歡迎再來。";
		close;
	}
	else {
		if (strnpcinfo(1) == "Merchant of Manuk") {
			mes "Rtt od d";
			mes "Qwo hd is d irr";
		}
		else {
			mes "BurWehAla";
			mes "tasnarAndu Ie Ru";
		}
		close;
	}
}
man_in01,286,16,1	duplicate(merchant_13_2)	Merchant of Manuk	454
