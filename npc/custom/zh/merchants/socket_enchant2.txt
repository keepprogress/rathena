					mes "[莉亞布萊姆]";

				else
				{

					mes "[莉亞布萊姆]";

						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					next;
					mes "[莉亞布萊姆]";
					mes "下次見，夥伴！";

			{
				mes "[莉亞布萊姆]";
				mes "你是笨蛋嗎？你沒有帶齊所有需要的道具！";
				mes "快去帶來！";
				close;
			}
		case 2:
