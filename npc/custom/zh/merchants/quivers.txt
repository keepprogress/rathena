payon_in01,5,134,5	script	發明家賈克斯::Inventor Jaax	89,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
		mes "[發明家賈克斯]";
		mes "嘿，你身上帶著";
		mes "太多東西了。何不把它們存進卡普拉倉庫？存好之後我們再聊，好嗎？";
		close;
	}
	mes "[發明家賈克斯]";
	mes "我叫賈克斯。";
	mes "不是我自誇，我";
	mes "可以說我大概是";
	mes "^663300這個時代最偉大的發明家^000000。";
	next;
	mes "[發明家賈克斯]";
	mes "這次，我";
	mes "做出了一樣";
	mes "非凡的東西……我稱它們為 ^663300魔法箭矢筒^000000！！這將會被載入史冊，成為一場";
	mes "箭矢革命！";
	emotion ET_BEST;
	next;
	mes "[發明家賈克斯]";
	mes "我研究魔法與箭矢筒已經好幾年了，日以繼夜地工作，終於想出如何用魔法將箭矢壓縮！有了魔法箭矢筒，你能帶更多箭矢，重量卻更輕！";
	next;
	mes "[發明家賈克斯]";
	mes "要不要試用一下我的箭矢筒？我相信像你這樣的人一定能體會我的天才之處！";
	next;
	switch(select("箭矢筒:鐵箭矢筒:鋼鐵箭矢筒:神之金屬箭矢筒:火箭矢筒:銀箭矢筒:風靈箭矢筒:地靈箭矢筒:水靈箭矢筒:影子箭矢筒:無形箭矢筒:鐵鏽箭矢筒")) {
	case 1: callsub S_BuyQuiver,1750,500,500,12004;
	case 2: callsub S_BuyQuiver,1770,500,500,12005;
	case 3: callsub S_BuyQuiver,1753,500,500,12006;
	case 4: callsub S_BuyQuiver,1765,500,500,12007;
	case 5: callsub S_BuyQuiver,1752,500,500,12008;
	case 6: callsub S_BuyQuiver,1751,500,500,12009;
	case 7: callsub S_BuyQuiver,1755,500,500,12010;
	case 8: callsub S_BuyQuiver,1756,500,500,12011;
	case 9: callsub S_BuyQuiver,1754,500,500,12012;
	case 10: callsub S_BuyQuiver,1767,500,500,12013;
	case 11: callsub S_BuyQuiver,1757,500,500,12014;
	case 12: callsub S_BuyQuiver,1762,500,500,12015;
	}
	//This doesn't seem to be accessable...
	//mes "[Inventor Jaax]";
	//mes "Is there";
	//mes "anything you want?";
	//close;

S_BuyQuiver:
	.@item_id = getarg(0);
	.@amount = getarg(1);
	.@zeny_req = getarg(2);

	if (countitem(.@item_id) >= .@amount) {
		mes "[發明家賈克斯]";
		mes "太棒了！";
		mes "你身上有帶箭矢嗎？我可以為你提供一個能裝下 "+ .@amount +" 支你的 "+ getitemname(.@item_id) +"，只要 ^FF3131"+ .@zeny_req +" Zeny^000000。";
		next;
		switch(select("盡量把箭矢存進箭矢筒:購買 1 個箭矢筒:取消")) {
		case 1:
			.@arrows = countitem(.@item_id);
			set .@quiver,.@arrows / .@amount;
			set .@arrows_used,.@quiver * .@amount;
			.@arrow_zeny01 = .@quiver * .@zeny_req;
			mes "持有的";
			mes "箭矢: ^3131FF"+.@arrows+" ^000000";
			mes "最多可購買";
			mes "的";
			mes "箭矢筒: ^3131FF"+.@quiver+" ^000000";
			mes "所需 Zeny: ^3131FF"+.@arrow_zeny01+" Zeny^000000";
			next;
			mes "[發明家賈克斯]";
			mes "您想要";
			mes "要盡量購買箭矢筒";
			mes "，用你能換的箭矢";
			mes "，就是你目前身上帶著的箭矢嗎？";
			next;
			if (select("是:取消") == 1) {
				if (.@arrow_zeny01 < Zeny) {
					mes "[發明家賈克斯]";
					mes "給你！";
					mes "記住，^FF0000當你的負重達到最大負重的 90% 時，就無法使用箭矢筒^000000。";
					set Zeny, Zeny-.@arrow_zeny01;
					delitem .@item_id,.@arrows_used;// Arrow
					getitem getarg(3),.@quiver; //Quiver
					next;
					mes "[發明家賈克斯]";
					mes "所以……";
					mes "只要留意";
					mes "自己身上";
					mes "帶了多少";
					mes "，隨時注意就好，";
					mes "應該就沒問題。";
					close;
				}
				else {
					mes "[發明家賈克斯]";
					mes "很抱歉，你的 Zeny 不夠。我不能白白送出這些東西，畢竟我花了好多年";
					mes "才研發出這項革命性的技術！";
					close;
				}
			}
			mes "[發明家賈克斯]";
			mes "你改變主意了嗎？";
			mes "擁有箭矢筒的榮耀";
			mes "近在咫尺，你還猶豫什麼？";
			close;
		case 2:
			if (Zeny > .@zeny_req) {
				mes "[發明家賈克斯]";
				mes "給你！";
				mes "記住，^FF0000當你的負重達到最大負重的 90% 時，就無法使用箭矢筒^000000。";
				Zeny = Zeny -.@zeny_req;
				delitem .@item_id,.@amount;// Arrow
				getitem getarg(3),1; //Quiver
				next;
				mes "[發明家賈克斯]";
				mes "所以……";
				mes "只要留意";
				mes "自己身上";
				mes "帶了多少";
				mes "，隨時注意就好，";
				mes "應該就沒問題。";
				close;
			}
			else {
				mes "[發明家賈克斯]";
				mes "你連";
				mes "有 "+ .@zeny_req +" Zeny 嗎？";
				mes "真是抱歉，我實在沒";
				mes "想到你竟然是那麼……";
				mes "^333333窮困潦倒^000000。";
				close;
			}
		case 3:
			mes "[發明家賈克斯]";
			mes "什麼……？";
			mes "難道你看不出來，這項發明能永遠改變箭矢的攜帶方式嗎？！未來已經到來了！";
			close;
		}
	}
	else {
		mes "[發明家賈克斯]";
		mes "這個箭矢筒最多能裝 "+ .@amount +" 支箭矢。這是用我的秘法製作的，";
		mes "所以箭矢加上箭矢筒的總重量，比單獨攜帶箭矢還要輕。";
		next;
		mes "[發明家賈克斯]";
		mes "這可是科學的奇蹟！只要你帶來至少 "+ .@amount +" 支箭矢和 "+ .@zeny_req +" Zeny 作為每個箭矢筒的費用。";
		close;
	}
}
