alde_alche,24,188,3	script	公會莊家::Guild Dealer	740,{
	if (checkweight(1201,1) == 0) {
		mes "- 等一下！ -";
		mes "- 你目前攜帶了 -";
		mes "- 道具太多了。 -";
		mes "- 請再來 -";
		mes "- 在你把部分道具存入卡普拉倉庫後。 -";
		close;
	}
	mes "[傑維爾]";
	mes "歡迎來到";
	mes "煉金師公會。";
	mes "今天有什麼可以為您效勞的嗎？";
	next;
	switch(select("購買材料。:購買製作說明書。:離開。")) {
	case 1:
		mes "[傑維爾]";
		mes "您想要什麼？";
		next;
		if (select("藥缽 - 8 Zeny:取消。") == 1) {
			mes "[傑維爾]";
			mes "你要多少？";
			mes "若要離開請輸入 '0'。";
			next;
			while(1) {
				input .@input,0,2001;
				if (.@input == 0) {
					mes "[傑維爾]";
					mes "交易已取消。";
					mes "下次再來。";
					close;
				}
				else if (.@input < 1 || .@input > 2000) {
					mes "[傑維爾]";
					mes "數量必須";
					mes "少於 2000。";
					next;
				}
				else {
					break;
				}
			}
			set .@sell,.@input * 8;
			set .@item_weight,.@input * 10;
			if (Zeny < .@sell) {
				mes "[傑維爾]";
				mes "你沒有";
				mes "足夠的 zeny。";
				mes "請先確認";
				mes "你有多少 zeny。";
				close;
			}
			if ((MaxWeight - Weight) < .@item_weight) {
				mes "[傑維爾]";
				mes "看來";
				mes "你無法全部攜帶。";
				mes "請確認";
				mes "道具欄的空間。";
				close;
			}
			set Zeny, Zeny-.@sell;
			getitem 7134,.@input; //Medicine_Bowl
			mes "[傑維爾]";
			mes "謝謝。";
			mes "下次再來。";
			close;
		}
		mes "[傑維爾]";
		mes "那麼，";
		mes "請在";
		mes "需要購買";
		mes "與";
		mes "煉金術相關的材料時再來，好嗎？";
		close;
	case 2:
		mes "[傑維爾]";
		mes "你需要什麼？";
		mes "製作說明書一般為 100,000 zeny。不過有幾本特殊的說明書價格更高。";
		next;
		switch(select("一般藥水製作說明書:酒精製作說明書:火煙瓶製作說明書:鹽酸瓶製作說明書:植物瓶製作說明書:刺錘瓶製作說明書:護貝藥製作說明書:纖細藥水製作說明書:取消交易。")) {
		case 1:
			callsub S_SellManual,7144,100000; //Normal_Potion_Book
			break;
		case 2:
			callsub S_SellManual,7127,100000; //Alcol_Create_Book
			break;
		case 3:
			callsub S_SellManual,7128,100000; //FireBottle_Create_Book
			break;
		case 4:
			callsub S_SellManual,7129,100000; //Acid_Create_Book
			break;
		case 5:
			callsub S_SellManual,7130,100000; //Plant_Create_Book
			break;
		case 6:
			callsub S_SellManual,7131,100000; //Mine_Create_Book
			break;
		case 7:
			callsub S_SellManual,7132,100000; //Coating_Create_Book
			break;
		case 8:
			callsub S_SellManual,7133,240000; //Slim_Potion_Create_Book
			break;
		case 9:
			mes "[傑維爾]";
			mes "那麼……";
			mes "如果你";
			mes "之後需要購買";
			mes "製作說明書，歡迎再來。";
			close;
		}
	case 3:
		mes "[傑維爾]";
		mes "那好吧，";
		mes "祝你有美好的一天。";
		close;
	}

S_SellManual:
	.@item_id = getarg(0);
	.@zeny_req = getarg(1);

	if (Zeny < .@zeny_req) {
		mes "[傑維爾]";
		mes "你沒有";
		mes "足夠的 zeny。";
		mes "請先確認";
		mes "你有多少 zeny。";
		close;
	}
	mes "[傑維爾]";
	mes getitemname(.@item_id) +"?";
	mes "那將會是 "+ .@zeny_req +" Zeny。";
	next;
	if (select("購買。:離開。") == 1) {
		Zeny = Zeny - .@zeny_req;
		getitem .@item_id,1;
		mes "[傑維爾]";
		mes "感謝您";
		mes "您的惠顧。";
		close;
	}
	mes "[傑維爾]";
	mes "歡迎再來";
	mes "下次見。";
	close;
}
