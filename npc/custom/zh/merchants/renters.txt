// PecoPeco Breeder (for Knights) :: breeder
//============================================================
prontera,55,350,5	script	大嘴鳥飼育員#knt::Peco Peco Breeder#knt	105,{
	if(Upper==0) set .@price,2500; //Normal Peco - default price
	if(Upper==1) set .@price,2500; //Armored Peco
	if(Upper==2) set .@price,2500; //Baby Peco

	mes "[大嘴鳥飼育員]";
	if (BaseJob == Job_Knight && Class < Job_Rune_Knight) {
		mes "歡迎。";
		mes "尊貴的騎士，";
		mes "要不要租借";
		mes "一隻大嘴鳥？租借";
		mes "費用為 "+.@price+" Zeny。";
		next;
		switch(select("租借大嘴鳥:取消")) {
		case 1:
			if (Zeny < .@price) {
				mes "[大嘴鳥飼育員]";
				mes "你該不會";
				mes "足夠的 zeny。";
				mes "是……";
				mes "破產了吧？";
				close;
			}
			else if(getskilllv("KN_RIDING") == 0) {
				mes "[大嘴鳥飼育員]";
				mes "很抱歉，你";
				mes "沒有資格使用";
				mes "這項服務。請先去學";
				mes "大嘴鳥騎乘技能再來。";
				close;
			}
			else if(checkriding()) {
				mes "[大嘴鳥飼育員]";
				mes "你已經";
				mes "騎著";
				mes "大嘴鳥了。";
				close;
			}
			else if(ismounting()) {
				mes "[大嘴鳥飼育員]";
				mes "請先卸下你的點券坐騎。";
				close;
			}
			set Zeny, Zeny-.@price;
			setriding;
			close;
		case 2:
			mes "[大嘴鳥飼育員]";
			mes "我知道了。";
			mes "那麼，";
			mes "祝你有美好的一天。";
			close;
		}
	} else {
		mes "很抱歉，這些";
		mes "大嘴鳥只";
		mes "提供給騎士";
		mes "和騎士領主。";
		close;
	}
}

// Grand PecoPeco Breeder (for Crusaders)
//============================================================
prontera,232,318,3	script	大嘴鳥飼育員#cru::Peco Peco Breeder#cru	105,{
	if(Upper==0) set .@price,3500; //Normal Peco - default price
	if(Upper==1) set .@price,3500; //Armored Peco
	if(Upper==2) set .@price,3500; //Baby Peco

	mes "[大嘴鳥飼育員]";
	if (BaseJob == Job_Crusader && Class < Job_Rune_Knight) {
		if(Upper != 1 ) mes "Welcome, Crusader.";
		else mes "Welcome, Paladin.";
		mes "我們準備了一隻";
		mes "特別的大嘴鳥，";
		mes "專門給你。要租借一隻";
		mes "的話，費用是 "+.@price+" Zeny。";
		next;
		switch(select("租借大嘴鳥:離開")) {
		case 1:
			if (Zeny < .@price) {
				mes "[大嘴鳥飼育員]";
				mes "你該不會";
				mes "足夠的 zeny。";
				mes "如果你想要";
				mes "大嘴鳥的話，請";
				mes "帶來 "+.@price+" Zeny……";
				close;
			}
			else if(getskilllv("KN_RIDING") == 0) {
				mes "[大嘴鳥飼育員]";
				mes "你必須先學會";
				mes "騎乘大嘴鳥，";
				mes "我才能租給你。";
				close;
			}
			else if(checkriding()) {
				mes "[大嘴鳥飼育員]";
				mes "你已經";
				mes "騎著大嘴鳥了。";
				close;
			}
			else if(ismounting()) {
				mes "[大嘴鳥飼育員]";
				mes "請先卸下你的點券坐騎。";
				close;
			}
			set Zeny, Zeny-.@price;
			setriding;
			close;
		case 2:
			mes "[大嘴鳥飼育員]";
			mes "改天見。";
			close;
		}
	} else {
		mes "有什麼我能為你做的嗎？";
		mes "請注意，";
		mes "這項大嘴鳥租借";
		mes "服務只限於";
		mes "十字軍與聖騎士。";
		close;
	}
}

hu_in01,381,304,5	script	獵鷹飼育員#hnt::Falcon Breeder#hnt	105,{
	if(Upper==0) set .@price,2500; //Normal Falcon - default price
	if(Upper==1) set .@price,2500; //Scarf Falcon
	if(Upper==2) set .@price,2500; //Baby Falcon

	mes "[獵鷹飼育員]";
	if (BaseJob == Job_Hunter) {
		if (checkwug()) {
			mes "嗯……";
			mes "你不能租借獵鷹，";
			mes "除非你先解散你的狼！";
			close;
		}
		mes "你需要獵鷹嗎？";
		mes "你可以租借屬於你自己的";
		mes "可靠的猛禽，";
		mes "費用只要 "+.@price+" Zeny～";
		next;
		switch(select("租借獵鷹:取消")) {
		case 1:
			if (checkwug()) {
				mes "[獵鷹飼育員]";
				mes "嗯……";
				mes "你不能租借獵鷹，";
				mes "除非你先解散你的狼！";
				close;
			}
			if (Zeny < .@price) {
				mes "[獵鷹飼育員]";
				mes "這是什麼意思？";
				mes "你沒有";
				mes "連 Zeny 都不夠？！";
				mes "你最好開始";
				mes "去狩獵賺錢";
				mes "而不是光打怪～";
				close;
			}
			else if(getskilllv("HT_FALCON") == 0) {
				mes "[獵鷹飼育員]";
				mes "天哪～";
				mes "先去學";
				mes "如何駕馭獵鷹";
				mes "吧！不然我不能租";
				mes "給你，如果你";
				mes "駕馭不了的話，你知道的。";
				close;
			}
			else if(checkfalcon()) {
				mes "[獵鷹飼育員]";
				mes "嗯……";
				mes "您已經裝備了";
				mes "獵鷹。它就在";
				mes "那裡，你沒看到嗎？";
				close;
			}
			set Zeny, Zeny-.@price;
			setfalcon;
			close;
		case 2:
			mes "[獵鷹飼育員]";
			mes "等、等一下，你要";
			mes "去哪？這些";
			mes "獵鷹可是頂級的，";
			mes "我保證！來吧，走～";
			close;
		}
	} else {
		mes "年輕人！";
		mes "獵鷹只能";
		mes "此物僅限獵人";
		mes "與狙擊手使用，懂了嗎？";
		mes "……呵呵，羨慕嗎？";
		close;
	}
}
