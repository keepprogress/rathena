payon,157,146,6	script	蘇恩比#cash::Suhnbi#cash	85,{
	disable_items;
	mes "[蘇恩比]";
	mes "我是武器鍛造師";
	mes "我能精煉各種武器，";
	mes "防具和裝備，所以告訴我";
	mes "你想精煉什麼。";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(set .@i,1; .@i<=10; set .@i,.@i+1) {
		if (getequipisequiped(.@indices[.@i])) {
			set .@menu$, .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			set .@equipped,1;
		}
		set .@menu$, .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "[蘇恩比]";
		mes "我想我無法精煉你擁有的任何道具……";
		close;
	}
	set .@part, .@indices[ select(.@menu$) ];

	if (!getequipisequiped(.@part)) //custom check
		close;
	if (!getequipisenableref(.@part)) {
		mes "[蘇恩比]";
		mes "去找別的鐵匠吧。這東西我無法精煉。";
		close;
	}
	if (getequiprefinerycnt(.@part) >= 10) {
		mes "[蘇恩比]";
		mes "嗯……這已經有人精煉到極致了，我想我無法再處理了。";
		close;
	}

	.@refineitemid = getequipid(.@part); // save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	.@price = getequiprefinecost(.@part, REFINE_COST_ENRICHED, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_ENRICHED, REFINE_MATERIAL_ID);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );

	// Make sure you have the necessary items and Zeny to refine your items
	// Determines chance of failure and verifies that you want to continue.
	callsub S_RefineValidate,.@itemtype,.@material,.@price,.@part,.@refineitemid,.@refinerycnt;

	mes "[蘇恩比]";
	mes "鏘！鏘！鏘！";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		next;
		emotion ET_BEST;
		mes "[蘇恩比]";
		mes "好了！完成了。";
		mes "好久沒有做出這麼精良的 "+((.@itemtype == IT_WEAPON)?"武器":"防具")+"。你一定很高興，因為它變強了！";
		close;
	}
	failedrefitem .@part;
	next;
	emotion (!rand(5))?ET_MONEY:ET_HUK;
	mes "[蘇恩比]";
	mes "嗚嗚嗚嗚嗚嗯！！！";
	next;
	mes "[蘇恩比]";
	mes "...";
	mes ".....";
	mes "……呼呼呼呼～";
	mes "……這是你的選擇，也是我的能力，沒有後悔。";
	close;

S_RefineValidate:
	.@itemtype = getarg(0);
	.@item_req = getarg(1);
	.@price = getarg(2);
	.@part = getarg(3);
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE)) {
			switch(.@equip_lv) {
				case 1:
					set .@price, .@price * 10;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE)) {
			switch( .@equip_lv ){
				case 1:
					set .@price, .@price * 40;
					break;
				case 2:
					set .@price, .@price * 50;
					break;
				case 3:
					set .@price, .@price * 2;
					break;
				case 4:
					set .@price, .@price * 2;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else{
		// TODO:
		close;
	}

	mes "[蘇恩比]";
	if (.@itemtype == IT_WEAPON)
		mes "你想精煉等級 "+ .@equip_lv +" 的武器嗎？";
	mes "要精煉它，你需要擁有一個 ^ff9999"+ getitemname(.@item_req) +"^000000 和 "+ .@price +" Zeny。";
	mes "您要繼續嗎？";
	next;
	if(select("是:否") == 1) {
		if (getequippercentrefinery(.@part) < 100) {
			if (.@itemtype == IT_WEAPON) {
				mes "[蘇恩比]";
				mes "哇！！";
				mes "這把武器大概";
				mes "看起來已經精煉過……";
				mes "很多次了……";
				mes "如果再精煉，";
				mes "可能會壞掉。";
				next;
				mes "而如果壞掉，";
				mes "你就再也無法使用了！";
				mes "裡面所有卡片與屬性都會^ff0000消失^000000！";
				mes "^ff0000而且，裝備會壞掉！^000000";
				mes "你確定還要繼續嗎？";
				next;
				if(select("是:否") == 2) {
					mes "[蘇恩比]";
					mes "很好。";
					mes "因為如果武器因為不合理的精煉而壞掉，我也會很不高興。";
					close;
				}
			} else {
				mes "[蘇恩比]";
				mes "呵呵。呵呵。哦，你還真有膽量，敢精煉這個。";
				mes "你知道這很冒險吧？";
				next;
				mes "如果你的防禦裝備壞了，你就再也無法使用了。";
				mes "連你的卡片和改造也會^ff0000完全消失^000000。";
				//mes "Everything will disappear. As in... GONE!";
				mes "你真的要繼續嗎？";
				next;
				if(select("是:否") == 2) {
					mes "[蘇恩比]";
					mes "真是胡鬧。你浪費了我寶貴的時間。";
					mes "滾開，小子。";
					close;
				}
			}
		}
		if (countitem(.@item_req) > 0 && Zeny > .@price) {
			delitem .@item_req,1;
			set Zeny, Zeny - .@price;

			// anti-hack
			if (callfunc("F_IsEquipIDHack", .@part, getarg(4)) ||
				callfunc("F_IsEquipRefineHack", .@part, getarg(5)) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3])) {
				mes "[蘇恩比]";
				emotion ET_FRET;
				mes "等一下……";
				mes "你當我是笨蛋嗎？！";
				mes "你趁我沒注意的時候換了道具！給我滾出去！";
				close;
			}

			return;
		}
		mes "[蘇恩比]";
		mes "你就只有這些嗎？";
		mes "非常抱歉，材料不齊我什麼都做不了。況且，我的工作也該拿到報酬吧？";
		close;
	}
	mes "[蘇恩比]";
	mes "就算你不高興，我也沒辦法……";
	close;
}
