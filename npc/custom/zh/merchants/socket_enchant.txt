function	script	Func_Socket	{
	.@item_id = getarg(0);
	.@zeny_req = getarg(4);
	setarray .@item_req[0], getarg(5), getarg(7,0);
	setarray .@req_amount[0], getarg(6), getarg(8,0);

	disable_items;
	mes "[塞亞布萊姆]";
	mes "你想為 "+ getitemname(.@item_id) +"?";
	mes "好的，請帶來 ^FF0000"+ .@req_amount[0] +" "+ getitemname(.@item_req[0]) + (.@req_amount[0] > 1 && .@item_req[0] != 999 ? "寶石":"") +"^000000, "+ (.@item_req[1] != 0 && .@req_amount[1] != 0 ? ("^FF0000"+ .@req_amount[1] +" "+ getitemname(.@item_req[1]) + (.@req_amount[1] > 1 && .@item_req[1] != 999 ? "寶石":"") + "^000000 ") : "") + "以及我的 ^FF0000" + (.@zeny_req >= 1000 ? (.@zeny_req /1000 +",000") : .@zeny_req) +",000 Zeny^000000 服務費。";
	mes "啊，還有別忘了帶上 "+ getitemname(.@item_id) +"!";
	next;
	mes "[塞亞布萊姆]";
	if (getiteminfo(.@item_id, ITEMINFO_LOCATIONS) & EQP_HAND_R) // weapon
	{
		mes "如果你帶齊了所需道具與 Zeny，我現在就可以嘗試加孔。";
		mes "不過，你應該知道我有可能失敗。";
		mes "因此，我必須先向你提出警告……";
		next;
		mes "[塞亞布萊姆]";
		mes "如果這次為你的武器加孔失敗，^FF0000武器^000000、^FF0000以及上面鑲嵌的所有卡片都會被摧毀^000000。";
	}
	else // armor
	{
		mes "如果你已備齊所有所需材料、我的 Zeny 服務費以及防具，我們就可以開始加孔嘗試了。";
		mes "不過在那之前，我必須先提醒你風險。";
		next;
		mes "[塞亞布萊姆]";
		mes "如果這次為你的防具加孔失敗，^FF0000防具^000000、^FF0000它的強化等級^000000 ^FF0000以及上面鑲嵌的所有卡片都會被摧毀^000000。";
	}
	mes "你還想嘗試加孔嗎？";
	next;
	switch(select("嘗試加孔:取消"))
	{
		case 1:
			if ((Zeny >= .@zeny_req*1000) && countitem(.@item_req[0]) >= .@req_amount[0] && countitem(getarg(7,512)) >= .@req_amount[1] && countitem(.@item_id) > 0)// if there's no getarg(7) and getarg(8) provided, check itemcount against 0 (we still need a valid item ID though)
			{
				mes "[塞亞布萊姆]";
				mes "那好，開始工作吧！";
				mes "你最好祈禱能成功。";
				next;
				set .@a, rand(1,100);
				if ((.@a > getarg(2)) && (.@a < getarg(3)))
				{
					specialeffect getarg(3) == 51 ? EF_LORD : EF_SANCTUARY; // only S Class items have Lord of Vermillion effect
					mes "[塞亞布萊姆]";
					mes "太好了，看起來成功了。";
					mes "做得相當漂亮。恭喜！";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					getitem getarg(1),1;
					next;
					mes "[塞亞布萊姆]";
					mes "下次見，夥伴！";
					close;
				}
				else
				{
					specialeffect EF_SUI_EXPLOSION;
					mes "[塞亞布萊姆]";
					mes "哇！……非常抱歉，失敗了。";
					mes "不過，這件事與我完全無關。";
					mes "這是你的運氣，是神註定的，好嗎？";
					mes "別太失望，下次再試試。";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					next;
					mes "[塞亞布萊姆]";
					mes "祝你下次好運！";
					close;
				}
			}
			else
			{
				mes "[塞亞布萊姆]";
				mes "我很想進行這次加孔嘗試，但你還缺少一些東西。";
				mes "你確定已經帶齊裝備、所需材料與 Zeny 了嗎？";
				close;
			}

			close;
	}
