//============================================================
//= Main Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//============================================================
function	script	refinemain	{
	disable_items;
	.@npc_name$ = getarg(0);
	.@features = getarg(1);
	mes "["+ .@npc_name$ +"]";
	mes "我是武器鍛造師。";
	mes "各種武器、防具和裝備我都能精煉，所以";
	mes "告訴我你想讓我精煉什麼吧。";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<getarraysize(.@indices); ++.@i) {
		if(getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "我想我無法精煉你擁有的任何道具……";
		close;
	}
	.@part = .@indices[select(.@menu$)];

	if(!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "你身上沒有穿戴";
		mes "任何";
		mes "我能精煉的東西。";
		emotion ET_FRET;
		close;
	}
	//Check if the item is refinable...
	if(!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "我想我沒辦法";
		mes "精煉這個道具了……";
		close;
	}
	//Check to see if the items is already +10
	if(getequiprefinerycnt(.@part) >= 10) {
		mes "["+ .@npc_name$ +"]";
		mes "這個我沒辦法再";
		mes "精煉了。這已經";
		mes "是精煉的極限了！";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 10;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 7;
				break;
			case 2:
				.@safe = 6;
				break;
			case 3:
				.@safe = 5;
				break;
			case 4:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 40;
					break;
				case 2:
					.@price = .@price * 50;
					break;
				case 3:
					.@price = .@price * 2;
					break;
				case 4:
					.@price = .@price * 2;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else{
		.@safe = 4;
	}

	// TODO: What was this? Maybe shadow?
	// case 5: .@price = .@price * 10; break;

	if(.@features != 1) {
		mes "["+ .@npc_name$ +"]";
		mes "要精煉這個，我需要";
		mes "一個 ^003366"+getitemname(.@material)+"^000000 以及";
		mes "手續費 " + .@price + " Zeny。";
		mes "你真的要繼續嗎？";
		next;
		if(select("是:否") == 2){
			mes "["+ .@npc_name$ +"]";
			mes "是啊……";
			mes "不用";
			mes "急，慢慢來就好。";
			close;
		}
		if(getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "糟了！如果我繼續";
			mes "精煉這個，它有可能會";
			switch(.@material) {
			case 985:
				mes "被摧毀！這就表示 ^FF0000這件裝備^000000，以及這件防具上的 ^FF0000任何卡片^000000 或特殊屬性，^FF0000都會消失^000000。";
				break;
			default:
				mes "被摧毀，你就會 ^FF0000失去這把武器^000000，武器上的 ^FF0000任何卡片^000000，";
				mes "以及所有附加的特殊屬性。";
				break;
			}
			next;
			mes "["+getarg(0)+"]";
			mes "我已經說得不能再清楚了。";
			mes "武器一旦被摧毀，";
			mes "就再也拿不回來了。";
			mes "你確實有可能";
			mes "^FF0000永遠失去這把武器^000000。";
			mes "你還要精煉嗎？";
			next;
			if(select("是:否") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "我完全同意……";
				mes "我或許是出色的精煉師，但有時候連我也會失手。";
				close;
			}
		}
		if((countitem(.@material) < 1) || (Zeny < .@price)) {
			mes "["+ .@npc_name$ +"]";
			mes "你好像沒有";
			mes "足夠的 Zeny，或是 "+getitemname(.@material)+"...";
			mes "去多準備一些吧，我會";
			mes "整天都在這裡，需要時再來找我。";
			close;
		}
		Zeny = Zeny-.@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "等一下……";
			mes "你當我是笨蛋嗎？！";
			mes "你趁我沒注意的時候換了道具！給我滾出去！";
			close;
		}

		if(getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			mes "["+ .@npc_name$ +"]";
			emotion (!rand(5))?ET_MONEY:ET_HUK;
			.@lose = rand(1,3);
			if (.@lose == 1) {
				mes "天啊！我的天啊！";
				mes "該死！又來了！";
				mes "真的很抱歉，不過你也知道，熟能生巧嘛。";
				mes "嗯，對吧？嘿嘿……";
			} else if(.@lose == 2) {
				mes "不要啊啊啊！";
				mes "壞掉了！";
				mes "對、對不起！";
			} else {
				mes "糟糕！";
				mes "它承受不住";
				mes "再多一次的淬鍊了！";
				mes "很抱歉這樣……";
			}
			close;
		}
		mes "["+getarg(0)+"]";
		successrefitem .@part;
		emotion ET_SMILE;
		.@win = rand(1,3);
		if (.@win == 1) {
			mes "完美！";
			mes "嘿嘿！";
			mes "又一次，";
			mes "完美無瑕的作品";
			mes "出自大師之手～";
		} else if(.@win == 2) {
			mes "成功了……！";
			mes "我那驚人的";
			mes "才華又一次";
			mes "在今天閃耀光芒。";
		} else {
			mes "嘿嘿！";
			mes "全部完成了。";
			mes "毫無疑問，我的作品";
			mes "一定能讓你滿意。";
			mes "完美至極～";
		}
		close;
	}

// New Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "我可以把它精煉到安全上限，或是精煉到你想要的次數。由你決定。";
		next;
		.@menu2 = select("請精煉到安全上限。","我自己決定次數。","我改變主意了……");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "你想讓我把道具精煉幾次？";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 10) {
			mes "["+ .@npc_name$ +"]";
			mes "這個道具我沒辦法精煉那麼多次。";
			close;
		}
		if(.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
			mes "這將嘗試把裝備精煉超過安全上限 " + .@refinecheck + " 次。你的裝備可能會被摧毀……這樣可以嗎？";
			next;
			if(select("是的……","不……") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "你都這麼說了……那就這樣吧。";
				close;
			}
		}
		break;
	case 3:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "你都這麼說了……那就這樣吧。";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "這將會花費你 " + .@refinecnt + " " + getitemname(.@material) + " 與 " + .@fullprice + " Zeny。這樣可以嗎？";
	next;
	if(select("是","不……") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "你都這麼說了……那就這樣吧。";
		close;
	}
	if(countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
		mes "你就只有這些嗎？很遺憾，我不能用更低的價格幫你做事。換位思考一下吧。";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "喂……你身上根本沒有穿戴任何道具……";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
				callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt) || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "鏘……不，你竟然以為我有這麼笨嗎？！";
			mes "你改了……";
			mes "在我用錘子把你打暈之前，給我滾出去！！";
			close;
		} 
		mes "鏘！鏘！鏘！";
		if(.@menu2 == 2 && getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			emotion ET_HUK;
			mes "["+ .@npc_name$ +"]";
			mes "哇啊啊啊！！我真的很抱歉……我早就警告過你會發生這種事了……";
			.@refinecnt = .@refinecnt - 1;
			if(.@refinecnt == 0) close;
			mes "這是沒用到的 Zeny 和材料，還給你……";
			getitem .@material,.@refinecnt;
			.@fullprice = .@refinecnt * .@price;
			Zeny = Zeny + .@fullprice;
			close;
		}
		successrefitem .@part;
		emotion ET_BEST;
		.@refinecnt = .@refinecnt - 1;
		.@refinerycnt = getequiprefinerycnt(.@part);
		next;
	}
	mes "["+ .@npc_name$ +"]";
	mes "全部完成了……歡迎再來。";
	close;
}

function	script	phramain	{
	if (checkweight(1201,1) == 0) {
		mes "- 等一下！！ -";

		mes "- 道具太多了。 -";

	mes "武器，還有 ^007777強化武器金屬-級數二^000000";
	mes "用於二級武器。";
	next;
	switch(select("強化武器金屬-級數一 - 200 Zeny:強化武器金屬-級數二 - 1000 Zeny:詢問其他金屬")) {
	case 1:
		.@material = 1010;
		.@price = 200;
		break;
	case 2:
		.@material = 1011;
		.@price = 1000;
		break;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "其他金屬？";
		mes "嗯，要強化高等級的武器或任何防具，你會需要特殊的金屬。不過你也知道，神之金屬和鋁真的";
		mes "很難找到……";
		close;
	}
	mes "["+ .@npc_name$ +"]";
	mes "那麼，你想買幾個？";
	mes "如果不想買，請輸入數字 '0'。";
	next;
	while(1) {
		input .@input;
		if (.@input == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "交易已經";
			mes "取消了。";
			close;
		}
		else if (.@input < 0 || .@input > 500) {
			mes "["+ .@npc_name$ +"]";
			mes "好吧，你最多";
			mes "只能購買 500 個。";
			mes "不能再多了，";
			mes "聽清楚了嗎？很好。";
			next;
		}
		else {
			break;
		}
	}
	.@sell = .@input * .@price;
	if (Zeny < .@sell) {
		mes "["+ .@npc_name$ +"]";
		mes "呃……";
		mes "你沒有";
		mes "足夠的 Zeny 來買";
		mes ""+ .@input +" 個。";
		close;
	}
	if (checkweight(.@material,.@input) == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "嗯……";

// Equipment Repair Function
//============================================================
function	script	repairmain	{
	.@repairprice = 5000;
	.@npc_name$ = getarg(0);
	mes "["+ .@npc_name$ +"]";
	mes "嘿，你好！";
	mes "你要我";
	mes "幫你修理道具嗎？";
	mes "交給我吧，";
	mes "道具修理我最在行！";
	next;
	switch(select("其實，我確實有些道具……:目前沒有。")) {
	case 1:
		.@checkitem = 1;
		while (1) {
			if (getbrokenid(.@checkitem) == 0) {
				break;
			}
			.@checkitem = .@checkitem+1;
		}
		.@checkitem = .@checkitem-1;
		if (!.@checkitem) {
			mes "["+ .@npc_name$ +"]";
			mes "哇，這太不可思議了！";
			mes "你一定很愛惜你的東西。你的道具一件都沒有損壞！";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "如果大家都像你一樣，我就要失業了！！哈哈～！";
			close;
		}
		mes "["+ .@npc_name$ +"]";
		mes "嗯……";
		mes "我看看……";
		mes "你的道具中，";
		mes "" + .@checkitem + " 件已經損壞。";
		mes "要修理嗎？";
		next;
		.@totalcost = .@repairprice*.@checkitem;
		mes "["+ .@npc_name$ +"]";
		mes "每次修理要花 " + .@repairprice + " Zeny。修理所有損壞的道具總共要花 " + .@totalcost + " Zeny！要修理這些道具嗎？";
		next;
		switch(select("是:否")) {
		case 1:
			if (Zeny < .@totalcost) {
				mes "["+ .@npc_name$ +"]";
				mes "哇哇……";
				mes "收修理費之前，先看看你的錢包吧！你的 Zeny 不夠，我什麼都修不了。";
				close;
			}
			.@checkitem2 = 1;
			while (1) {
				if (getbrokenid(.@checkitem2) == 0) {
					break;
				}
				.@checkitem2 = .@checkitem2+1;
			}
			.@checkitem2 = .@checkitem2-1;
			if (.@checkitem == .@checkitem2) {
				Zeny = Zeny-.@totalcost;
				while (.@checkitem) {
					repair(.@checkitem);
					.@checkitem = .@checkitem-1;
				}
				mes "["+ .@npc_name$ +"]";
				mes "好了！全部修好了。以後小心一點，道具也是有生命的，你知道的。";
				close;
			}
			else {
				mes "["+ .@npc_name$ +"]";
				mes "嗯？有點不對勁。等等……先把需要修理的道具裝備上，再回來找我。";
				close;
			}
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "好吧，這跟我沒關係，不過道具壞著不修可不好。你最好盡快去修理！";
			close;
		}
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "呵呵呵……";
		mes "你沒有";
		mes "有什麼事要找我嗎？";
		mes "如果你沒有任何";
		mes "需要修理的道具，就別來打擾我了。";
		close;
	}
}
