payon,173,238,5	script	傑德#pay::Jade#pay	754,{
	if (checkweight(1201,1) == 0) {
		mes "^3355FF等一下！你現在攜帶的道具太多了。請把部分道具存入卡普拉倉庫後再來。^000000";
		close;
	}
	mes "[傑德]";
	mes "帶兩個";
	mes "同色的寶石";
	mes "來，我就會";
	mes "把它們換成寶石";
	mes "不同顏色的。";
	next;
	switch(select("藍色寶石換成紅色寶石！:紅色寶石換成黃色寶石！:黃色寶石換成藍色寶石！")) {
	case 1: callsub S_TradeGems,717,716;
	case 2: callsub S_TradeGems,716,715;
	case 3: callsub S_TradeGems,715,717;
	}

S_TradeGems:
	.@item_req = getarg(0);
	.@item_id = getarg(1);

	if (countitem(.@item_req) < 2) {
		mes "[傑德]";
		mes "哈……！";
		mes "你在開玩笑吧？";
		mes "我沒辦法為你提供這項";
		mes "服務，除非你";
		mes "至少給我";
		mes "2 "+ getitemname(.@item_req) +"些寶石！";
		close;
	}
	else {
		.@gems = countitem(.@item_req) /2;
		mes "[傑德]";
		mes "我相信我能製作出";
		mes "總計 "+ .@gems +" "+ getitemname(.@item_id) +"寶石";
		mes "使用 "+ getitemname(.@item_req) +"寶石";
		mes "你目前擁有的寶石。";
		mes "你想做什麼？";
		next;
		switch(select("給我你能給的全部。:我要自己設定數量。:我不做了。")) {
		case 1:
			delitem .@item_req,.@gems * 2;
			getitem .@item_id,.@gems;
			break;
		case 2:
			mes "[傑德]";
			mes "那麼你要";
			mes "要多少？";
			mes "數量上限";
			mes "是 100 個。";
			next;
			while(1) {
				input .@input,0,101;
				if (.@input == 0) {
					mes "[傑德]";
					mes "一個都沒有？";
					mes "看來你";
					mes "改變主意了……";
					close;
				}
				else if (.@input > 100) {
					mes "[傑德]";
					mes "呃……";
					mes "我要你輸入的";
					mes "數量不能超過";
					mes "100，記住了嗎……？";
					next;
				}
				else if (.@gems < .@input) {
					// Custom dialogue
					mes "[傑德]";
					mes "呃……";
					mes "你沒有那麼";
					mes "多寶石可以交換……";
					next;
				}
				else break;
			}
			delitem .@item_req,.@input * 2;
			getitem .@item_id,.@input;
			break;
		case 3:
			mes "[傑德]";
			mes "好的，沒問題。";
			mes "隨時歡迎再來。";
			close;
		}
		mes "[傑德]";
		mes "給你。";
		mes "歡迎隨時";
		mes "再來。";
		mes "嗯，你那是什麼表情？";
		mes "我臉上有什麼東西嗎？";
		close;
	}
}
