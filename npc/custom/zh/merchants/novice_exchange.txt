prontera,123,102,5	script	商人#pron::Merchant#pron	85,{
	if (MaxWeight - Weight < 6301 || checkweight(1201,1) == 0) {
		mes "[商人]";
		mes "哈哈！";
		mes "您是超人嗎？";
		mes "您帶了這麼多東西！";
		mes "您最好把一些東西";
		mes "存進卡普拉倉庫~";
		close;
	}
	mes "[商人]";
	mes "您好！";
	mes "您有收集到任何";
	mes "像貝殼或毛之類的道具嗎？";
	mes "果凍碎片怎麼樣？喔對了，";
	mes "我有需要用到它們。";
	next;
	mes "[商人]";
	mes "當然，我不會要您";
	mes "白白給我那些東西。如果我用一瓶紅色藥水換 5 個貝殼、10 撮毛，";
	mes "或 10 個果凍碎片？";
	next;
	switch(select("好啊！:我覺得這是坑人！:不用了，謝謝。")) {
	case 1:
		mes "[商人]";
		mes "那您想拿";
		mes "哪一種道具來給我？";
		mes "貝殼、毛，還是果凍碎片？";
		next;
		switch(select("貝殼:果凍碎片:毛:取消")) {
		case 1: callfunc "F_PotExchange",935,5;
		case 2: callfunc "F_PotExchange",909,10;
		case 3: callfunc "F_PotExchange",914,10;
		case 4:
			mes "[商人]";
			mes "好的~";
			mes "沒問題。";
			close;
		}
	case 2:
		mes "[商人]";
		mes "坑人……？";
		mes "如果您看看市場價值，";
		mes "交易的道具，其實是我吃虧";
		mes "啊。";
		close;
	case 3:
		mes "[商人]";
		mes "好吧，";
		mes "沒問題。";
		mes "不過如果您改變主意，";
		mes "請再回來找我。";
		close;
	}
}

function	script	F_PotExchange	{
	.@item_req = getarg(0);
	.@req_amount = getarg(1);

	if (countitem(.@item_req) < .@req_amount) {
		mes "[商人]";
		mes "嘿，你答應我的";
		mes "那些 "+ getitemname(.@item_req) +"";
		mes "東西去哪了？";
		mes "給我 "+ getitemname(.@item_req) +"!";
		close;
	}
	else {
		mes "[商人]";
		mes "好，我來查看";
		mes "有多少 "+ getitemname(.@item_req) +" 您";
		mes "身上帶著。嗯……";
		next;
		mes "[商人]";
		mes "您有";
		mes "總計 "+ countitem(.@item_req) +" "+ getitemname(.@item_req) +".";
		mes "我可以給您總共";
		mes "共 "+ (countitem(.@item_req)/.@req_amount) +" 瓶紅色藥水作為交換。";
		next;
		mes "[商人]";
		mes "您覺得如何？";
		mes "成交嗎？";
		next;
		if (select("成交。:不成交。") == 1) {
			mes "[商人]";
			mes "您知道確切的";
			mes "紅色藥水數量";
			mes "您想要收到";
			mes "那些 "+ getitemname(.@item_req) +"，對吧？";
			next;
			mes "[商人]";
			mes "現在，我可以跟您交換";
			mes "最少 1 瓶紅色藥水";
			mes "最多 100 瓶紅色藥水";
			mes "一次交換。如果您改變";
			mes "主意，輸入 '0' 即可取消。";
			next;
			while(1) {
				input .@input,0,101;
				if (.@input == 0) {
					mes "[商人]";
					mes "什麼……？";
					mes "你到底為什麼";
					mes "還要來跟我說話？";
					mes "您還真優柔寡斷，不是嗎？";
					close;
				}
				else if (.@input > 100) {
					mes "[商人]";
					mes "我不能給您超過";
					mes "100 瓶紅色藥水";
					mes "一次。我們再試一次";
					mes "吧。";
					next;
				}
				else {
					break;
				}
			}
			.@put_out_item = .@input * .@req_amount;
			if (countitem(.@item_req) < .@put_out_item) {
				mes "[商人]";
				mes "糟糕，您輸入的數字似乎不對。您最好再確認一次能交換的紅色藥水數量。";
				close;
			}
			mes "[商人]";
			mes "給你！";
			mes "請確認您拿到的紅色藥水數量";
			mes "我已經給您了，應該沒問題。謝謝，這筆交易很划算~";
			delitem .@item_req,.@put_out_item; //Shell
			getitem 501,.@input; //Red_Potion
			close;
		}
		mes "[商人]";
		mes "咦~";
		mes "好吧。";
		mes "不過不是";
		mes "紅色藥水更有用嗎";
		mes "對像您這樣的冒險者來說？";
		close;
	}
}
