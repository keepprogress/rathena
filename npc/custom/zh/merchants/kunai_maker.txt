que_ng,72,29,3	script	飛刀商人卡欣::Kunai Merchant Kashin	83,{
	if(BaseClass == Job_Ninja) {
		mes "[卡欣]";
		mes "我是卡欣，忍者飛刀";
		mes "的經銷商。請";
		mes "隨意看看，";
		mes "如果您有興趣，";
		mes "就告訴我您想買的商品。";
		next;
		switch(select("10 劇毒飛刀:10 垂冰飛刀:10 強風飛刀:10 黑土飛刀:10 火焰飛刀:取消")) {
		case 1: callfunc "Kunai_Trade",13250,20,7524,1,13259; break;
		case 2: callfunc "Kunai_Trade",13251,8,7522,2,13255; break;
		case 3: callfunc "Kunai_Trade",13252,4,7523,2,13257; break;
		case 4: callfunc "Kunai_Trade",13253,2,7524,1,13256; break;
		case 5: callfunc "Kunai_Trade",13254,1,7521,2,13258; break;
		case 6:
			mes "[卡欣]";
			mes "那麼，謝謝您";
			mes "光臨我的店。";
			mes "需要購買";
			mes "一些";
			mes "飛刀時請再來找我。再見~";
			close;
		}
	}
	mes "[卡欣]";
	mes "我是卡欣，忍者飛刀";
	mes "忍者飛刀的經銷商。如果您";
	mes "有任何是";
	mes "忍者的朋友，那麼您可以";
	mes "跟他們提起我。";
	close;
}

function	script	Kunai_Trade	{
	setarray .@item_req[0], getarg(0), getarg(2);
	setarray .@req_amount[0], getarg(1), getarg(3);
	.@item_id = getarg(4);

	mes "[卡欣]";
	mes "您可以用";
	mes .@req_amount[0] +" "+ getitemname(.@item_req[0]) +" 和";
	mes .@req_amount[1] +" "+ getitemname(.@item_req[1]) +" 換每";
	mes "組 10 "+ getitemname(.@item_id) +".";
	next;
	mes "[卡欣]";
	mes "我一次最多只能給您 500 組飛刀。";
	mes "如果您想取消交易，請輸入 ''0.'' 您要";
	mes "多少組飛刀？";
	next;
	input .@amount;
	if(.@amount < 1 || .@amount > 500) {
		mes "[卡欣]";
		mes "咦？很抱歉，";
		mes "我無法給您";
		mes "那麼多組飛刀。請";
		mes "輸入小於 500 的數字。";
		close;
	}
	if (countitem(.@item_req[0]) < .@amount*.@req_amount[0] || countitem(.@item_req[1]) < .@amount*.@req_amount[1]) {
		mes "[卡欣]";
		mes "嗯，您沒有";
		mes "足夠的道具";
		mes "來進行飛刀兌換。請";
		mes "再確認一次您的道具。";
		close;
 	}
	if (checkweight(.@item_id, .@amount*10) == 0) {
		mes "[卡欣]";
		mes "嗯，您的";
		mes "背包沒有";
		mes "足夠空間來放";
		mes "更多道具。您最好";
		mes "先騰出一些空間。";
		close;
	}
	mes "[卡欣]";
	mes "太好了，一切";
	mes "都就緒，那我們開始";
	mes "完成這筆交易吧。";
	mes "我相信您會";
	mes "對";
	mes "這些飛刀非常滿意。";
	delitem .@item_req[0],.@req_amount[0]*.@amount;
	delitem .@item_req[1],.@req_amount[1]*.@amount;
	getitem .@item_id,10*.@amount;
	close;
}
