alberta_in,16,28,4	script	藥劑師::Pharmacist	61,{
	if (checkweight(1201,1) == 0) {
		mes "- 請稍等！ -";
		mes "- 你目前攜帶的 -";
		mes "- 道具太多了。 -";
		mes "- 請稍後再來 -";
		mes "- 等你把部分道具存入卡普拉倉庫後 -";
		close;
	}
	mes "[老藥師]";
	mes "嗯嗯嗯……";
	mes "您來這裡有什麼事嗎……？";
	next;
	switch(select("製作藥水:聊聊:調配資訊:取消")) {
	case 1:
		if (MaxWeight - Weight < 5000) {
			mes "[老藥師]";
			mes "您怎麼帶這麼多！";
			mes "別貪心，只帶您需要的量！";
			next;
			mes "[老藥師]";
			mes "您帶太重了，無法接受我的藥水……";
			mes "請先把一些道具存進倉庫！";
			close;
		} else {
			mes "[老藥師]";
			mes "您的材料都準備好了吧？您想要哪一種？";
			next;
			switch(select("紅色藥水。:柳橙藥水。:黃色藥水。:白色藥水。:藍色藥水。:綠色藥水。:其實我什麼都不要。")) {
			case 1:
				callsub L_Making,507,3,501;
			case 2:
				mes "[老藥師]";
				if (countitem(507) < countitem(713)) {
					set .@max,countitem(507);
				}
				else if (countitem(508) < countitem(713)) {
					set .@max,countitem(508);
				}
				else {
					set .@max,countitem(713);
				}
				if ((countitem(507) < 1) || (countitem(508) < 1) || (countitem(713) == 0)) {
					mes "你這個混蛋！你以為會怎樣？！空手而來。嘖！";
					mes "滾開！";
					close;
				}
				if (Zeny < 3) {
					mes "你這個混蛋！你以為會怎樣？！空手而來。嘖！";
					mes "滾開！";
					close;
				}
				mes "要製作幾個？";
				next;
				switch(select("盡量製作:我要自己選數量:其實我什麼都不要。")) {
				case 1:
					if ((countitem(507) < .@max) || (countitem(508) < .@max) || (countitem(713) < .@max) || (Zeny < .@max*3)) {
						mes "[老藥師]";
						mes "你這小子！材料都還沒備齊，就想叫我幫你做藥水？！";
						close;
					}
					set Zeny,Zeny-(.@max*5);
					delitem 507,.@max; //Red_Herb
					delitem 508,.@max; //Yellow_Herb
					delitem 713,.@max; //Empty_Bottle
					getitem 502,.@max; //Orange_Potion
					break;
				case 2:
					mes "[老藥師]";
					mes "那就選個 100 以下的數字吧。不想要的話，輸入 '0' 就好。以你現有的材料，大約可以製作 "+.@max+" 瓶藥水。";
					next;
					input .@amount,0,101;
					if (.@amount == 0) {
						mes "[老藥師]";
						mes "你到底想好了沒有？！";
						close;
					}
					if (.@amount > 100) {
						mes "[老藥師]";
						mes "你聾了嗎？我說的是 100 以下！";
						close;
					}
					if ((countitem(507) < .@amount) || (countitem(508) < .@amount) || (countitem(713) < .@amount) || (Zeny < .@amount*3)) {
						mes "[老藥師]";
						mes "你這小子！材料都還沒備齊，就想叫我幫你做藥水？！";
						close;
					}
					next;
					set Zeny,Zeny-(.@amount*5);
					delitem 507,.@amount; //Red_Herb
					delitem 508,.@amount; //Yellow_Herb
					delitem 713,.@amount; //Empty_Bottle
					getitem 502,.@amount; //Orange_Potion
					break;
				case 3:
					mes "[老藥師]";
					mes "什麼？！";
					mes "哼……";
					mes "呸！";
					mes "滾開！";
					close;
				}
				mes "[老藥師]";
				mes "給你。已經做好了，拿去吧。不過要記住！藥物亂用可不好。";
				close;
			case 3:
				callsub L_Making,508,10,503;
			case 4:
				callsub L_Making,509,20,504;
			case 5:
				callsub L_Making,510,30,505;
			case 6:
				callsub L_Making,511,3,506;
			case 7:
				mes "[老藥師]";
				mes "什麼？！";
				mes "哼……";
				mes "呸！";
				mes "滾開！";
				close;
			}
		}
	case 2:
		mes "[老藥師]";
		mes "藥物確實能提升人的恢復能力，但也只到某個程度為止。*嘆氣* 我開始懷念年輕的時候了，看來我真的老了。";
		next;
		mes "[老藥師]";
		mes "總之，藥水就只是藥水，沒什麼特別的。";
		close;
	case 3:
		mes "[老藥師]";
		mes "嗯嗯……";
		mes "你們這些年輕人真是煩人。不過既然你問了，我就跟你說明。";
		next;
		mes "[老藥師]";
		mes "藥草單獨使用就有效，但如果用我的獨門技術把它們製成藥水，效果會好上許多許多。";
		next;
		mes "[老藥師]";
		mes "如果你誠心有禮地來求我，我就幫你做。不過可不是白做的……別擔心，我只收一點小手續費，不會太貴。";
		next;
		mes "[老藥師]";
		mes "紅色藥水 - ^0098E52 紅色藥草、1 個空瓶，手續費 2 Zeny。^000000";
		mes "赤色藥水 - ^0098E51 紅色藥草、1 黃色藥草、1 個空瓶，手續費 5 Zeny。^000000";
		mes "黃色藥水 - ^0098E52 黃色藥草、1 個空瓶，10 Zeny。^000000";
		next;
		mes "[老藥師]";
		mes "白色藥水 - ^0098E52 白色藥草、1 個空瓶，手續費 20 Zeny。^000000";
		mes "藍色藥水 - ^0098E52 藍色藥草、1 個空瓶，手續費 30 Zeny。^000000";
		mes "綠色藥水 - ^0098E52 綠色藥草、1 個空瓶，手續費 3 Zeny。^000000";
		close;
	case 4:
		mes "[老藥師]";
		mes "真是個無聊的人。有話要說就直說啊！";
		close;
	}
	
L_Making:
	.@item_req = getarg(0);
	.@req_amount = getarg(1);

	mes "[老藥師]";
	if ((countitem(.@item_req)/2) < countitem(713)) {
		.@max = countitem(.@item_req) /2;
	} else {
		.@max = countitem(713);
	}
	if (countitem(.@item_req) < 2 || countitem(713) == 0) {
		mes "你這個混蛋！你以為會怎樣？！空手而來。嘖！";
		mes "滾開！";
		close;
	}
	if (Zeny < 3) {
		mes "你這個混蛋！你以為會怎樣？！空手而來。嘖！";
		mes "滾開！";
		close;
	}
	mes "要製作幾個？";
	next;
	switch(select("盡量製作:我要自己選數量:其實我什麼都不要。")) {
	case 1:
		if ((countitem(.@item_req) < .@max*2) || countitem(713) < .@max || (Zeny < .@max*.@req_amount)) {
			mes "[老藥師]";
			mes "你這小子！材料都還沒備齊，就想叫我幫你做藥水？！";
			close;
		}
		Zeny = Zeny - (.@max*.@req_amount);
		delitem .@item_req,.@max*2;
		delitem 713,.@max; //Empty_Bottle
		getitem getarg(2),.@max; 
		break;
	case 2:
		mes "[老藥師]";
		mes "那就選個 100 以下的數字吧。不想要的話，輸入 '0' 就好。以你現有的材料，大約可以製作 "+.@max+" 瓶藥水。";
		next;
		input .@amount;
		if (.@amount == 0) {
			mes "[老藥師]";
			mes "你到底想好了沒有？！";
			close;
		}
		if (.@amount > 100) {
			mes "[老藥師]";
			mes "你聾了嗎？我說的是 100 以下！";
			close;
		}
		if ((countitem(.@item_req) < .@amount*2) || countitem(713) < .@amount || (Zeny < .@amount*.@req_amount)) {
			mes "[老藥師]";
			mes "你這小子！材料都還沒備齊，就想叫我幫你做藥水？！";
			close;
		}
		Zeny = Zeny - (.@max*.@req_amount);
		delitem .@item_req,.@amount*2;
		delitem 713,.@amount; //Empty_Bottle
		getitem getarg(2),.@amount; 
		break;
	case 3:
		mes "[老藥師]";
		mes "什麼？！";
		mes "哼……";
		mes "呸！";
		mes "滾開！";
		close;
	}
	mes "[老藥師]";
	mes "給你。已經做好了，拿去吧。不過要記住！藥物亂用可不好。";
	close;
}
