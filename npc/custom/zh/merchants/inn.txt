		mes "謝謝你~";
		mes "希望你享受";
		mes "您在";

		mes "[飯店員工]";

//======================= Inn Function ==============================
function	script	F_InnMaid	{
	.@npc_name$ = getarg(0);

	mes .@npc_name$;
	mes "歡迎來到";
	mes getarg(1) + ".";
	mes "有什麼可以為您效勞的嗎？";
	next;
	switch(select("存檔:休息 -> 5000 Zeny:取消")) {
	case 1:
		mes .@npc_name$;
		mes "您的重生點";
		mes "已存檔。";
		mes "謝謝您，";
		mes "歡迎再次光臨。";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close;
	case 2:
		mes .@npc_name$;
		if(Zeny < 5000){
			mes "很抱歉，住宿費為 5,000 Zeny。下次請確認您有足夠的錢辦理入住，好嗎？";
			close;
		}
		mes "謝謝。";
		mes "希望您";
		mes "祝您休息愉快~";
		close2;
		set Zeny,Zeny - 5000;
		percentheal 100,100;
		return;
	case 3:
		close;
	}
}
