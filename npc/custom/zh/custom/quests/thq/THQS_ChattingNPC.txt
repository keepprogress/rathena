	mes "嗯，抱歉，要是你連會員都不是，我可不想浪費時間跟你說話！";
	close;

	set @TEMP,rand(1,2);
	if (@TEMP == 1) goto N_Chat1;
	if (@TEMP == 2) goto N_Chat2;

	mes "嗯，除非你在看程式碼 :)";
	close;

	close;

	close;
}
