umbala,126,129,4	script	鬍子男#bandit::Bearded Man#bandit	120,{
	if(BEARD_QUEST == 4 ) goto L_MAKE;
	if(BEARD_QUEST >= 5 ) goto L_DONE;
	if(BEARD_QUEST >= 1 ) goto L_THREAD;

	mes "[鬍子男]";
	mes "鬍子！快來買您喜愛的鬍子！";
	mes "您想要鬍子嗎？";
	next;
	menu "好，我想要一些臉上的鬍鬚！",-,"不用了，好噁心！",M_NO;

	mes "[鬍子男]";
	mes "呵呵~ 所以您想要鬍鬚呀。";
	emotion ET_SMILE;
	mes "其實我本來想送您我的鬍子，但它們可不便宜。";
	mes "說實話，我只剩下一個了……";
	mes "不過我不賣它。";
	mes "但別生氣！我可以幫您做一個。";
	next;
	set BEARD_QUEST,1;
L_THREAD:
	mes "[鬍子男]";
	mes "我需要一些 ^8080FF線^000000 來做鬍子。";
	mes "去依斯魯得找我的朋友拿一些 ^8080FF線^000000 吧。";
	next;

	mes "[鬍子男]";
	mes "另外，我還需要您幫我拿一些做鬍子的材料。";
	mes "材料如下:";
L_LIST:
	mes "^8080FF橡皮筋^000000";
	mes "^8080FF5 張動物外皮^000000";
	mes "^8080FF黑色染料^000000";
	mes "^8080FF白鬍鬚^000000";
	mes "^8080FF100 個粘稠液體^000000";
	mes "^8080FF中和劑^000000";
	mes " 以及一顆 ^8080FF2克拉鑽石^000000";
	close;

M_NO:
	mes "[鬍子男]";
	mes "好吧，隨您便！";
	emotion ET_ANGER;
	close;

L_DONE:
	mes "[鬍子男]";
	mes "抱歉，每人只能做一個。";
	mes "我們現在不希望鬍子太多了，對吧？";
	emotion ET_KIK;
	close;

L_MAKE:
	mes "[鬍子男]";
	mes "來幫您做鬍子吧！";
	next;
	if(countitem(7200) < 1 || countitem(919) < 1 ||	countitem(983) < 1 || countitem(2241) < 1 ||
	countitem(938) < 100 || countitem(973) < 1 || countitem(731) < 1) goto L_NOITEMS;
	delitem 7200, 1;
	delitem 919, 1;
	delitem 983, 1;
	delitem 2241, 1;
	delitem 938, 100;
	delitem 973, 1;
	delitem 731, 1;
	mes "^8080FF~他正在剪裁縫合鬍子~^000000";
	next;
	mes "^8080FF~他把完成的鬍子交給您~^000000";
	next;
	getitem 2237, 1;
	set BEARD_QUEST, 5;
	mes "[鬍子男]";
	mes "祝您有愉快的一天！";
	emotion ET_KIK;
	close;

L_NOITEMS:
	mes "[鬍子男]";
	mes "搞什麼！沒有材料我做不出鬍子！";
	mes "以下再次列出清單:";
	emotion ET_HUK;
	goto L_LIST;
}

alberta,65,123,6	script	男人#bandit::Man#bandit	85,{
	if(BEARD_QUEST == 3 && $@beardMobD) goto L_KILLED;
	if(BEARD_QUEST > 3) goto L_DONE;

	mes "[線商人]";
	mes "啊啊啊！我的貨永遠送不到艾爾帕蘭！";
	mes "我一直被一群蟲子圍攻。";
	mes "牠們偷走了我所有的補給和 ^8080FF線^000000。";
	next;
	mes "[線商人]";

	if(BEARD_QUEST != 3) mes "What am I to do?";
	if(BEARD_QUEST != 3) close;

	if($@beardmob > 0) mes "Kill! Kill them already!!!";
	if($@beardmob > 0) close;
	mes "您是來幫我的嗎？";
	emotion ET_QUESTION;
	next;
	menu "是",-,"否",M_NO;

	mes "[線商人]";
	mes "感謝神明！";
	mes "只要您能擊殺那些蟲子，我就能把貨送出去了。";
	next;
	mes "[線商人]";
	mes "糟了！牠們來了！！";
	emotion ET_PROFUSELY_SWEAT;
	donpcevent "BRDQ_MOBS::OnStart";
	close;

L_KILLED:
	set $@beardMobD,0;
	mes "[線商人]";
	mes "感謝您擊殺了盜蟲。";
	mes "作為感謝，這裡有一整箱 ^8080FF線^000000 送給您。";
	set BEARD_QUEST,4;
	close;

L_DONE:
	mes "[線商人]";
	mes "再次感謝您！";
	emotion ET_THANKS;
	close;

M_NO:
	mes "[線商人]";
	mes "該怎麼辦……";
	close;
}
