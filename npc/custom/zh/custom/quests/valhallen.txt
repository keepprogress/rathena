	mes "[蘭妮絲]";
	mes "我從瓦爾哈拉來協助你。";
	mes "我可以為你煉成許多瓦爾哈拉的道具，";
	mes "雖然製作它們需要許多稀有元素。";
	next;
	switch(select("雷神之錘:史雷普尼爾之靴:女神頸鍊:魔法鐵鏈:雷神腰帶:阿斯富力咖:布林喜德:海爾伯樂神盾:耶夢加得法杖:神奇金屬:符文之精華:關閉")) {
		case 1:
			mes "[蘭妮絲]";
			mes "^3355FF雷神之錘^000000是索爾的強大神錘。";
			setarray .@Items[0],1530,1522,2,984,20,985,5,969,40,7074,2,7075,4,7078,5,7091,5,7089,5;
			break;
		case 2:
			mes "[蘭妮絲]";
			mes "^3355FF史雷普尼爾之靴^000000是仿照奧丁的戰馬所打造的靴子。";
			setarray .@Items[0],2410,2406,2,969,20,985,10,7076,3,7079,5,7083,3,7086,4,7087,3;
			break;
		case 3:
			mes "[蘭妮絲]";
			mes "^3355FF女神頸鍊^000000是美之女神芙蕾雅的魔法項鍊。";
			setarray .@Items[0],2630,2603,1,726,2,722,3,727,10,724,5,969,20,7073,4,7077,4,7088,3,7090,3,7092,3;
			break;
		case 4:
			mes "[蘭妮絲]";
			mes "^3355FF魔法鐵鏈^000000是一條輕巧卻堅韌的繩子，是製作^3355FF雷神腰帶^000000所需的材料。";
			setarray .@Items[0],7058,7080,4,7081,5,7082,4,7084,3,7085,3;
			break;
		case 5:
			mes "[蘭妮絲]";
			mes "^3355FF雷神腰帶^000000是索爾的強大腰帶。";
			setarray .@Items[0],2629,7058,1,2627,1,969,20,726,10,984,10;
			break;
		case 6:
			mes "[蘭妮絲]";
			mes "^3355FF阿斯富力咖^000000是諸神的光輝。";
			setarray .@Items[0],2541,7835,1,7836,1,7837,1,7838,1,2513,1,7063,100,7291,10,7293,10,985,20;
			break;
		case 7:
			mes "[蘭妮絲]";
			mes "^3355FF布林喜德^000000是戰士的鎧甲。";
			setarray .@Items[0],2383,7830,1,7831,1,7832,1,7833,1,7834,1,7510,100,2357,1,969,10,985,20;
			break;
		case 8:
			mes "[蘭妮絲]";
			mes "^3355FF海爾伯樂神盾^000000是以替戰士帶來勝利的女武神來命名的。";
			setarray .@Items[0],2181,6595,2,6596,1,6594,4,6597,3,6602,1,6604,1,6605,1,2115,1;
			break;
		case 9:
			mes "[蘭妮絲]";
			mes "^3355FF耶夢加得法杖^000000以能毒害敵人而聞名。";
			setarray .@Items[0],2020,6603,4,6599,1,6598,1,6601,4,6600,1,6604,1,6605,1,1473,1;
			break;
		case 10:
			mes "[蘭妮絲]";
			mes "^3355FF神奇金屬^000000是製作^3355FF耶夢加得法杖^000000和^3355FF海爾伯樂神盾^000000所需的材料。";
			setarray .@Items[0],6605,984,50,985,50;
			break;
		case 11:
			mes "[蘭妮絲]";
			mes "^3355FF符文之精華^000000是製作^3355FF耶夢加得法杖^000000和^3355FF海爾伯樂神盾^000000所需的材料。";
			setarray .@Items[0],6604,12735,200,12736,100;
			break;
		default:
			close;
	}
	
	mes "^FF0000我需要的道具如下:^000000";
	set .@itemmissing,0;
	for(set .@i,1; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		set .@amount,.@Items[.@i+1];
		set .@id,.@Items[.@i];
		mes " ~ "+.@amount+"x "+((countitem(.@id)<.@amount)?("^dd3355"):("^33dd55"))+getitemname(.@id)+((getitemslots(.@id)>0)?" ["+getitemslots(.@id)+"]":"")+"^000000";
		if (countitem(.@id) < .@amount) 
			set .@itemmissing,1;
	}
	if (.@itemmissing) {
		next;
		mes "[蘭妮絲]";
		mes "去把這些道具全部找來吧。";
		mes "我會在這裡等著幫你。";
		close;
	}
	next;
	mes "[蘭妮絲]";
	mes "你想要 ^3355FF"+getitemname(.@Items[0])+"^000000?";
	next;
	if(select("否","是")==1) {
		mes "[蘭妮絲]";
		mes "如果你改變心意，請再回來。";
		close;
	}
	mes "[蘭妮絲]";
	mes "這是你的 ^3355FF"+getitemname(.@Items[0])+"^000000.";
	mes "願它為你好好效力。";
	//second check if the player tries to bypass the first
	for(set .@i,1; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		set .@amount,.@Items[.@i+1];
		set .@id,.@Items[.@i];
		if (countitem(.@id) < .@amount) 
			close;
	}
	// Delete the items _after_ the second check
	for(set .@i,1; .@i<getarraysize(.@Items); set .@i,.@i+2) {
		set .@amount,.@Items[.@i+1];
		set .@id,.@Items[.@i];
		delitem .@id, .@amount;
	}
	getitem .@Items[0],1;
	close;
}
