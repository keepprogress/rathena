		break;

		mes "["+@name$+"]";
		mes "我正在等你的伴侶接受你，好開始婚禮。";
		close;
	case 3:
		SF_RetrieveRingM();
		break;

		break;
	default:

		mes "請現在不要打擾我。";
		close;
	}

function SF_RetrieveRingM {
	mes "["+@name$+"]";
	if ($@wed_ring && countitem($@wed_ring) < 1) {
		mes "你的 "+getitemname($@wed_ring)+"怎麼了？你不會是弄丟了吧……？我們需要它才能繼續婚禮！";
		close;
	}
	if ($wed_bride_sex)
		set @item, 2634;	//Groom's wedding ring
	else
		set @item, 2635;	//Bride's wedding ring
	if (getnameditem(@item,$wed_groom$) == 0) {
		mes "你的背包似乎沒有足夠的空間放戒指……請先騰出一些空間，再回來領取你伴侶的戒指。";
		close;
	}
	mes "這是給新娘的婚戒。";
	if ($@wed_ring) delitem $@wed_ring,1;
	set $wed_groom_progress,4;
	
	if ($wed_bride_progress == 4)
		SF_StartCeremony();
	else {
		mes "新娘領取戒指後，婚禮就會開始。";
		close;
	}
}

function SF_StartCeremony {
	mes "現在我將開始婚禮，並宣布你們正式成為夫妻。";
	set $wed_bride_progress,5;
	set $wed_groom_progress,5;
	set $@msg$,$wed_groom$;
	if (strcharinfo(0) == $wed_groom$)
		set $@msg$,$wed_bride$;
	if (marriage($@msg$) == 0) {
		next;
		mes "["+@name$+"]";
		mes "請問 "+$@msg$+"？？少了一方，我沒辦法為你們證婚……";
		close;
	}
	set $wed_bride_progress,6;
	set $wed_groom_progress,6;
	initnpctimer;
	close;
}

//Subfunction: Checks that the groom/bride is still wearing their stuff.
function SF_equip_check {
	if (Sex == SEX_MALE && getequipid(EQI_ARMOR) != 7170) {
		mes "["+@name$+"]";
		mes "孩子，你把你的 "+getitemname(7170)+"?";
		emotion ET_THINK;
		return 0;
	}
	if (Sex == SEX_FEMALE && getequipid(EQI_ARMOR) != 2338) {
		mes "["+@name$+"]";
		mes "孩子，你應該要隨時戴著 "+getitemname(2338)+" 直到婚禮結束……";
		emotion ET_THINK;
		return 0;
	}
	if (Sex == SEX_FEMALE && $@wed_veil && getequipid(EQI_HEAD_TOP) != 2206) {
		mes "["+@name$+"]";
		mes "孩子，你還不能把你的 "+getitemname(2206)+" 拿下來……";
		emotion ET_THINK;
		return 0;
	}
	return 1;
}

function SF_Register {
	if ($@wed_allow) { //Role select
		mes "["+@name$+"]";
		mes "好的，你想以哪一種身分登記呢？";
		next;
		set @submenu, select("新郎","新娘","取消");
	} else if (Sex == SEX_MALE) { //Groom
		mes "["+@name$+"]";
		mes "好的，你要以新郎的身分登記嗎？";
		next;
		if (select("是","我改變主意了。")==1)
			set @submenu, 1;
		else
			set @submenu, 3;
	} else { //Bride
		mes "["+@name$+"]";
		mes "好的，你要以新娘的身分登記嗎？";
		next;
		if (select("是","我改變主意了。")==1)
			set @submenu, 2;
		else
			set @submenu, 3;
	}
	switch (@submenu) {
	case 1: //Groom
		SF_TryRegister(0);
		set $wed_progress,1;
		mes "["+@name$+"]";
		mes "你已登記為新郎。";
		mes "請盡快請你的新娘來登記。";
		emotion ET_SCRATCH;
		initnpctimer;
		close;
	case 2: //Bride
		SF_TryRegister(1);
		set $wed_progress,1;
		mes "["+@name$+"]";
		mes "你已登記為新娘。";
		mes "請盡快請你的新郎來登記。";
		emotion ET_SCRATCH;
		initnpctimer;
		close;
	default: //Cancel
		mes "["+@name$+"]";
		mes "準備好了再回來吧。";
		close;
	}
}

//Subfunction SF_TryRegister (int bride)
function SF_TryRegister {
	set @bride, getarg(0);
	set @type$, "groom";
	if (@bride)
		set @type$, "bride";
		
	mes "["+@name$+"]";
	mes "在登記成為 "+@type$+"之前，讓我先確認你是否符合所有條件……";
	next;
	if (Upper == 2) {
		mes "["+@name$+"]";
		mes "哎呀，你還太年輕，不該想這種事情！";
		emotion ET_SURPRISE;
		close;
	}
	if (Sex == SEX_MALE)
		set @item, 7170;
	else
		set @item, 2338;
	
	if (getequipid(EQI_ARMOR) != @item) {
		mes "["+@name$+"]";
		mes "想結婚的話，你應該要戴著 "+getitemname(@item)+" 才行。";
		close;
	}
	if (Sex == SEX_FEMALE && $@wed_veil && getequipid(EQI_HEAD_TOP) != 2206) {
		mes "["+@name$+"]";
		mes "你的 "+getitemname(2206)+"在哪裡？那是搭配禮服不可或缺的飾品。";
		close;
	}
	if ($@wed_ring && countitem($@wed_ring) < 1) {
		mes "["+@name$+"]";
		mes "戒指在哪裡？你需要一個 "+getitemname($@wed_ring)+" 來交換戒指，親愛的。";
		close;
	}
	if (@bride)
		set @cost, $@wed_bride_reg;
	else 
		set @cost, $@wed_groom_reg;

	if (Zeny < @cost) {
		mes "["+@name$+"]";
		mes "很抱歉，你的 Zeny 不足以支付登記費。";
		mes "等你收集到 "+@cost+"z。"; 
		close;
	}
	set Zeny,Zeny-@cost;
	sc_start SC_Wedding,3600000,1; //Start Wedding Effect (SC_WEDDING)
	if (@bride) {
		set $wed_bride_progress,1;
		set $wed_bride$,strcharinfo(0);
		set $wed_bride_sex, Sex;
		if ($@wedding_effect == 1) //Store account id for effect.
			set $wedding_effect_id, getcharid(3);
	} else {
		set $wed_groom_progress,1;
		set $wed_groom$,strcharinfo(0);
		set $wed_groom_sex, Sex;
		if ($@wedding_effect == 2) //Store account id for effect.
			set $wedding_effect_id, getcharid(3);
	}
}
