//===== Description: =========================================
//= Black Jack card game.  Gameplay based off standard casino
//= black jack rules.  Dealer must have at least 17 to stay and will
//= automatically stay at 17 and up.  Player must have at least
//= 13 to stay.  Aces counted as 11 or 1.  Option to "Double Down".
//= Currently does not allow for "insurance", or "splitting"
//= of pairs.
//===== Additional Comments: =================================
//= 1.0 First version.
//============================================================

cmd_in02,188,89,1	script	21點莊家::Black Jack Dealer	57,{
	mes "[莊家]";
	mes "您好！要來玩一局 21 點嗎？";
M_Menu:
	next;
	menu "規則是什麼？",M_0, "我想玩。",L_Play, "不用了，謝謝。",M_End;

M_0:
	mes "[莊家]";
	mes "你想知道什麼？";
sM_Menu0:
	next;
	menu "基本規則。",sM_0a, "勝負判定。",sM_0b, "「加倍」。",sM_0c,
	"A 的點數。",sM_0d, "沒有了。",M_Menu;

sM_0a:
	mes "[莊家]";
	mes "21 點是一種紙牌遊戲，目標是拿到 21 點，或是盡可能";
	mes "接近 21 點，但不能超過 21 點。";
	next;
	mes "[莊家]";
	mes "遊戲開始前，玩家必須先下注。下注完成後，";
	mes "莊家與玩家都會發到兩張牌";
	mes "。依照手上的牌，您可以選擇 ^5533FF「停牌」^000000，";
	mes "或是選擇 ^5533FF「要牌」^000000。";
	next;
	mes "[莊家]";
	mes "選擇「停牌」時，代表您告訴莊家您";
	mes "不需要再拿牌。這讓莊家可以";
	mes "視情況決定是否要牌。若要停牌，您必須至少有 ^FF3355 13 點^000000。";
	mes "莊家只有在點數達到 ^FF3355 17 點或以上^000000 時才能停牌。";
	next;
	mes "[莊家]";
	mes "選擇「要牌」時，代表您告訴莊家您想要";
	mes "再拿一張牌。藉由要更多牌，您可以增加點數";
	mes "總和。只要您的點數 ^FF3355低於 21 點^000000，就能再要一張";
	mes "牌（從牌堆抽取）。";
	goto sM_Menu0;
sM_0b:
	mes "[莊家]";
	mes "21 點的勝負判定有三種方式。";
	next;
	mes "[莊家]";
	mes "1.) 一局結束時，若您的點數";
	mes "高於莊家，您就贏得該局。反之，若您的點數";
	mes "低於莊家，您就輸了該局。";
	next;
	mes "[莊家]";
	mes "2.) 若您要的牌讓點數超過 21 點，您";
	mes "將自動輸掉該局。這稱為 ^5533FF「爆牌」^000000。若";
	mes "莊家爆牌，您就贏得該局。";
	next;
	mes "[莊家]";
	mes "3.) 若您前兩張牌的點數為 21 點，您";
	mes "將自動贏得該局。這稱為 ^5533FF「黑傑克」^000000";
	mes "，發生於您拿到「A」與「10 點牌」時。若";
	mes "莊家拿到黑傑克，就會自動贏得該局。";
	next;
	mes "[莊家]";
	mes "4.) 除了勝負之外，您也可能與莊家平手。若";
	mes "一局結束時，您與莊家的點數相同，";
	mes "就會平手，雙方都不贏也不輸，";
	mes "不增減金錢。";
	next;
	mes "[莊家]";
	mes "這稱為與莊家的 ^5533FF「平手」^000000。這也";
	mes "同樣適用於您與莊家同時拿到黑傑克時，";
	mes "同樣算平手。";
	goto sM_Menu0;
sM_0c:
	mes "[莊家]";
	mes "「加倍」選項讓您將目前的賭注加倍，";
	mes "但缺點是您只能再要";
	mes "一張牌。這個選項只能在";
	mes "每局一開始時使用。";
	next;
	mes "[莊家]";
	mes "加倍的好時機，舉例來說，是在";
	mes "您前兩張牌的點數為 11 點時。您有很大";
	mes "機率能用下一張牌拿到 21 或 20 點。這";
	mes "是很適合加倍的好牌。";
	goto sM_Menu0;
sM_0d:
	mes "[莊家]";
	mes "「A」在 21 點中是獨特的牌，因為";
	mes "它有兩種點數。A 可以算作 11 點，";
	mes "或只算 1 點。例如您拿到 A 和 4，這樣";
	mes "就會得到 15 點或 5 點。";
	next;
	mes "[莊家]";
	mes "若您決定停牌，A 會自動算作";
	mes "11 點，使您的總點數為 15 點。";
	next;
	mes "[莊家]";
	mes "若您決定要牌並拿到 9，A 會";
	mes "自動算作 1 點，使您的總點數為 14";
	mes "點。若 A 算作 11 點，您就會";
	mes "使點數超過 21 點而輸掉該局。";
	next;
	mes "[莊家]";
	mes "正是因為「A」的彈性，讓它成為";
	mes "遊戲中最強大的一張牌。";
	goto sM_Menu0;
M_End:
	mes "[莊家]";
	mes "歡迎隨時再來";
	close;

//================
L_Play:
	mes "[莊家]";
	mes "請下注……";
	next;
	menu "2z",M_1a, "10z",M_1b, "20z",M_1c, "100z",M_1d, "賭注太高，我玩不起……",M_End;

M_1a:
	if(Zeny < 2) goto sL_NotEnuf;
	set @bet, 2;
	goto L_Cont0;
M_1b:
	if(Zeny < 10) goto sL_NotEnuf;
	set @bet, 10;
	goto L_Cont0;
M_1c:
	if(Zeny < 20) goto sL_NotEnuf;
	set @bet, 20;
	goto L_Cont0;
M_1d:
	if(Zeny < 100) goto sL_NotEnuf;
	set @bet, 100;
	goto L_Cont0;

sL_NotEnuf:
	mes "[莊家]";
	mes "很抱歉，您的 Zeny 不足以下這個賭注。";
	close;

L_Cont0:
	mes "（正在發牌）";
	next;
	deletearray $@card[0],13;
	set @dealerTurn, 0;
	set @numP, 0;
	set @numD, 0;
	set @pAce, 0;
	set @dAce, 0;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;

//==============
L_Start:
	callsub sF_GetTot;

	mes "- 以下是 ^FF5533莊家^000000 的牌:";
	if (@numD==2) callsub sF_D2cards;
	if (@numD==3) callsub sF_D3cards;
	if (@numD==4) callsub sF_D4cards;
	if (@numD==5) callsub sF_D5cards;
	mes "    莊家點數: ^FF5533"+@dealTot+"^000000";
	mes " ";
	mes "- 以下是 ^5533FF您的^000000 牌:";
	if (@numP==2) callsub sF_P2cards;
	if (@numP==3) callsub sF_P3cards;
	if (@numP==4) callsub sF_P4cards;
	if (@numP==5) callsub sF_P5cards;
	if(@pAce != 1 || @playTot == 21) mes "    YOU have: ^5533FF" +@playTot+ "^000000";
	if(@pAce == 1 && @playTot != 21) mes "    You have: ^5533FF" +@playTot+ "^000000, or ^5533FF" +(@playTot-10)+ "^000000";
	next;
	if(@playTot==21 && @dealTot==21) goto sL_Push;
	if(@numP==2 && @playTot == 21) goto sL_Win;
	if(@numD==2 && @dealTot == 21) goto sL_Lose;
	if(@playTot > 21) goto sL_Lose;
	if(@dealTot > 21) goto sL_Win;
	if(@numP==2 && @dealerTurn==0) menu "Hit me(pull)",M_Hit, "Stay",M_Stay, "Double Down",M_Double;
	if(@dealerTurn == 0) menu "Hit me(pull)",M_Hit, "Stay",M_Stay;

M_Stay:
	mes "[莊家]";
	if(@playTot < 13) goto sL_PlayToLow;
	if(@dealTot > 16) mes "The Dealer stays.";
	if(@dealTot > 16 || @numD == 5) goto L_Check;
	mes "莊家將會要牌";
	next;
	callsub sF_GetCards, @numD, @dealCard[@numD], @dealCard$[@numD], @dAce;
	set @dealerTurn, 1;
	goto L_Start;

sL_PlayToLow:
	mes "很抱歉，您的點數不夠高，無法停牌。您必須要牌。";
	next;
	goto M_Hit;

M_Hit:
	if(@numP == 5) goto M_Stay;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	goto L_Start;
M_Double:
	mes "[莊家]";
	mes "玩家選擇了加倍。您目前的賭注將會";
	mes "加倍，且您只能再要一張牌。";
	next;
	set @dealerTurn, 1;
	set @bet, @bet*2;
	callsub sF_GetCards, @numP, @playCard[@numP], @playCard$[@numP], @pAce;
	goto L_Start;

//=============
L_Check:
	next;
	if(@playTot < @dealTot) goto sL_Lose;
	if(@playTot == @dealTot) goto sL_Push;

sL_Win:
	mes "[莊家]";
	mes "恭喜，您贏了！";
	next;
	set Zeny, Zeny + @bet;
	goto L_Play;
sL_Lose:
	mes "[莊家]";
	mes "很抱歉，您輸了。";
	set Zeny, Zeny - @bet;
	next;
	goto L_Play;
sL_Push:
	mes "[莊家]";
	mes "平手。您與莊家的點數相同。";
	next;
	goto L_Play;


//==================================
	// Sub function for dealing/pulling the cards
sF_GetCards:
	set @rnd, rand(1,13);
	if($@card[@rnd] == 4) goto sF_GetCards;
	set $@card[@rnd], $@card[@rnd] + 1;
	set getarg(1), @rnd;
	if(getarg(1) > 10) set getarg(1), 10;
	if(getarg(1) == 1 && getarg(3) < 1) set getarg(1), 11;
	if(getarg(1) == 11) set getarg(3), 1;
	set getarg(2), " " + getarg(1) + " ";
	if(@rnd == 10) set getarg(2), getarg(1);
	if(@rnd == 1) set getarg(2), " A ";
	if(@rnd == 11) set getarg(2), " J ";
	if(@rnd == 12) set getarg(2), " Q ";
	if(@rnd == 13) set getarg(2), " K ";
	set getarg(0), getarg(0) + 1;
	return;

//==================================
	// Sub function for getting the total score for each hand
sF_GetTot:
	set @i, 0;
	set @dealTot, 0;
	set @playTot, 0;

GetDealTot:
	set @dealTot, @dealTot + @dealCard[@i];
	set @i, @i + 1;
	if(@i < @numD) goto GetDealTot;
	if(@dAce == 1 && @dealTot > 21) set @dealTot, @dealTot - 10;
	set @i, 0;
GetPlayTot:
	set @playTot, @playTot + @playCard[@i];
	set @i, @i + 1;
	if(@i < @numP) goto GetPlayTot;
	if(@pAce == 1 && @playTot > 21) set @pAce, 2;
	if(@pAce > 0 && @playTot > 21) set @playTot, @playTot - 10;
	return;

//=======================================
	// Sub function for displaying the Cards
sF_D2cards:
	mes "    .-----.   .-----. ";
	mes "    | "+@dealCard$[0]+" |   | "+@dealCard$[1]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_P2cards:
	mes "    .-----.   .-----. ";
	mes "    | "+@playCard$[0]+" |   | "+@playCard$[1]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_D3cards:
	mes "    .-----.   .-----.   .-----. ";
	mes "    | "+@dealCard$[0]+" |   | "+@dealCard$[1]+" |   | "+@dealCard$[2]+" | ";
	mes "    '-----'   '-----'    '-----' ";
	return;
sF_P3cards:
	mes "    .-----.   .-----.   .-----. ";
	mes "    | "+@playCard$[0]+" |   | "+@playCard$[1]+" |   | "+@playCard$[2]+" | ";
	mes "    '-----'   '-----'    '-----' ";
	return;
sF_D4cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@dealCard$[0]+" |  | "+@dealCard$[1]+" |  | "+@dealCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.  ";
	mes "    | "+@dealCard$[3]+" |";
	mes "    '-----'   ";
	return;
sF_P4cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@playCard$[0]+" |  | "+@playCard$[1]+" |  | "+@playCard$[2]+" |";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.  ";
	mes "    | "+@playCard$[3]+" |";
	mes "    '-----'   ";
	return;
sF_D5cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@dealCard$[0]+" |  | "+@dealCard$[1]+" |  | "+@dealCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.   .-----. ";
	mes "    | "+@dealCard$[3]+" |   | "+@dealCard$[4]+" | ";
	mes "    '-----'   '-----' ";
	return;
sF_P5cards:
	mes " .-----.  .-----.  .-----. ";
	mes " | "+@playCard$[0]+" |  | "+@playCard$[1]+" |  | "+@playCard$[2]+" | ";
	mes " '-----'  '-----'   '-----' ";
	mes "    .-----.   .-----. ";
	mes "    | "+@playCard$[3]+" |   | "+@playCard$[4]+" | ";
	mes "    '-----'   '-----' ";
	return;
}
