//===== Description: =========================================
//= Players buy ninjas to assassinate other players
//===== Additional Comments: =================================
//= 1.1.1 Changed all gmcommand to atcommand as Poki#3 suggested. [Vicious]
//= 1.1.2 Updated WoE Check. [Paradox924X]
//============================================================

morocc,148,86,5	script	狡詐刺客::Shifty Assassin	725,{
	set .@ninja_price,250000;

	// STARTS THE MENU //
M_Start:
	mes "[狡詐刺客]";
	mes "您想做什麼？";
	next;
	menu "購買忍者",M_Buy,"暗殺某人",M_Kill,"查看您的忍者",M_Check, ( getgmlevel() > 90 ? "增加忍者" : "" ),-,"取消",M_Exit;

	// GM MENU TO ADD NINJAS //
	mes "[狡詐刺客]";
	mes "您想要開放多少名忍者？";
	next;
	input .@add;
	set $ninja_avail, $ninja_avail+.@add;
	mes .@add +" 名忍者已新增。";
	close;

	// BUY NINJAS //
M_Buy:
	mes "[狡詐刺客]";
	mes "您想購買多少名忍者？";
	mes "目前有 ^0000FF" + $ninja_avail + "^000000 名忍者可供購買。";
	mes "每名忍者的費用為 ^0000FF" + .@ninja_price + " Zeny ^000000。";

	input .@buy;
	next;
	if ($ninja_avail < 1) goto NoNinjas;
	if ($ninja_avail < .@buy) goto NotEnoughNinjas;
	set .@price, .@buy*.@ninja_price;
	if (Zeny < .@price) goto NoZeny;

	mes "[狡詐刺客]";
	mes "這將花費您 ^0000FF"+ .@price +" Zeny^000000。";
	next;
	menu "繼續",-,"取消",M_Exit;

	set Zeny, Zeny - .@price;
	set #ninjas, #ninjas + .@buy;
	set $ninja_avail, $ninja_avail - .@buy;

	mes "[狡詐刺客]";
	mes "謝謝。";
	close;

	// ASSASSINATE SOMEBODY //
M_Kill:
	if (agitcheck()) goto M_Busy;
	mes "[狡詐刺客]";
	mes "請輸入目標的名稱。";
	mes "^FF0000請正確輸入名稱，否則我無法找到目標。^000000";
	next;
	menu "繼續",-,"取消",M_Exit;
	input .@name$;
	if ( !getcharid( 3,.@name$ ) ) {
		mes "[狡詐刺客]";
		mes .@name$ +" 目前不在線上。";
		close;
	}
	next;
	mes "[狡詐刺客]";
	mes "活動中的忍者: "+#ninjas;
	mes "休息中的忍者: "+#ninjasr;
	mes "您想派出多少名？";
	input .@number;
	if (.@number < 1) goto NoNinjasSent;
	if (.@number > #ninjas) goto NotEnoughNinjas1;
	if (.@number > 10) goto TooManyNinjas;
	set .@chance, rand(1,12);
	set #ninjas,#ninjas-.@number;
	set #ninjas,#ninjas+#ninjasr;
	set #ninjasr,0;
	if (.@number < .@chance) goto M_Failure;

	// SUCCESSFUL ATTACK //
	mes "現在派出忍者。";
	next;
	mes "[狡詐刺客]";
	set .@ninjasurvived, rand(1,.@number);
	set #ninjasr,.@number-.@ninjasurvived;
	mes "您的攻擊成功，但只有 ^FF0000" + #ninjasr + "^000000 名忍者存活。";
	set .@acc_id, getcharid( 3,.@name$ );
	if ( .@acc_id )
		unitkill .@acc_id;
	announce .@name$ +" 已遭到 " + strcharinfo(0) +"的忍者暗殺。",bc_npc;
	close;

	// FAILED ATTACK //
M_Failure:
	mes "現在派出忍者。";
	next;
	mes "[狡詐刺客]";
	set .@ninjasurvived, rand(1,.@number);
	set #ninjasr, .@number-.@ninjasurvived;
	mes "您的攻擊失敗，只有 ^FF0000" + #ninjasr + "^000000 名忍者存活。";

	announce .@name$+" 在 " + strcharinfo(0) +"的忍者攻擊中存活下來。",8;
	close;

	// NINJAS BUSY FOR WOE //
M_Busy:
	mes "[狡詐刺客]";
	mes "抱歉，我的忍者們都在忙著參加攻城戰。";
	close;

	// CHECK YOUR NINJAS //
M_Check:
	mes "[狡詐刺客]";
	mes "您目前有:";
	mes "^FF0000" + #ninjas + "^000000 名活動中的忍者。";
	mes "^0000FF" + #ninjasr + "^000000 名休息中的忍者。";
	next;
	goto M_Start;


	// LIMIT //
NoNinjasSent:
	mes "[狡詐刺客]";
	mes "沒有忍者，您就無法擊殺任何人。";
	next;
	goto M_Start;

TooManyNinjas:
	mes "[狡詐刺客]";
	mes "最多只能派出 10 名忍者。";
	next;
	goto M_Start;

NoZeny:
	mes "[狡詐刺客]";
	mes "您的 Zeny 不足。";
	close;

NotEnoughNinjas:
	mes "[狡詐刺客]";
	mes "可供購買的忍者數量不足。";
	next;
	goto M_Start;

NoNinjas:
	mes "[狡詐刺客]";
	mes "已沒有可供購買的忍者了。";
	close;

NotEnoughNinjas1:
	mes "[狡詐刺客]";
	mes "您沒有那麼多忍者。";
	next;
	goto M_Start;

M_Exit:
	mes "[狡詐刺客]";
	mes "再見。";
	close;

// TIMER DELAY NINJA ADDER //
OnClock0600:
OnClock1200:
OnClock1500:
OnClock1900:
OnClock2000:
OnClock0000:
	set $ninja_avail,$ninja_avail+2;
	end;
OnClock1800:
	set $ninja_avail,$ninja_avail+3;
	end;
OnInit:
	set $ninja_avail,$ninja_avail+1;
	end;
}
