//Counts down and starts race
pvp_y_1-5,145,269,5	script	起跑員::Starter	733,{
	if ($@race != 0) goto L_Started;
	if ($@counting != 0) goto L_Started;
	if ($@racecount == 1) goto L_Started;
L_Menu:
	mes "[起跑員]";
	mes "請站在我的東邊。";
	menu "開始比賽",L_Count,"取消",-;
	close;
L_Count:
	set $@counting,1;
	mes "倒數中……";
	addtimer 1000, "Starter::OnCount1000";
	addtimer 2000, "Starter::OnCount2000";
	addtimer 3000, "Starter::OnCount3000";
	addtimer 4000, "Starter::OnCount4000";
	announce strcharinfo(0) + "已開始倒數",1;
	announce "準備好比賽了！",1;
	close;

OnCount1000:
	announce "[3]",1;
	end;
OnCount2000:
	announce "[2]",1;
	end;
OnCount3000:
	announce "[1]",1;
	end;
OnCount4000:
	emotion ET_GO;
	specialeffect EF_CHIMTO;
	announce "[出發！]",1;
	set $@race,1;
	set $@position,0;
	set $@counting,0;
	set $@raceid,rand(100000,999999);
	end;

L_Started:
	if ((getmapusers("pvp_y_1-5") < 3) && ($@position > 0)) goto L_Menu;
	if ($@position > 2) goto L_Menu;
	mes "[起跑員]";
	mes "比賽進行中";
	close;

OnInit:
	set $@race,0;
	set $@position,0;
	set $@racecount,0;
	end;
}

//Checkpoint 1
pvp_y_1-5,144,262,5	script	檢查點1::Check Point 1	111,0,5,{
	end;
OnTouch:
	if (@raceid != $@raceid) goto L_Started;
	if (@race == 6) goto L_Finished;
	if ($@race == 1) goto L_Started;
	mes "比賽還沒開始，請往後退。";
	close;
L_Started:
	set @race,1;
	set @raceid,$@raceid;
	end;
L_Finished:
	mes "你已經完成比賽了。";
	close;
}

//Finish Line
pvp_y_1-5,174,244,5	script	終點線::Finish Line	111,6,6,{
	end;
OnTouch:
	if (@raceid != $@raceid) goto L_WrongRace;
	if (@race != 5) goto L_Miss;
	set @race,6;
	set $@position,$@position+1;
	announce "[" + strcharinfo(0) +"] 已抵達終點線！[名次: " + $@position + "]",1;
	end;
L_Miss:
	mes "你錯過了一個檢查點，請回頭。";
	close;
L_WrongRace:
	mes "你不在這場比賽中。";
	close;
}
