//===== Description: =========================================
//= Click the chicken and try retrieve the item at a low 
//= chance. If you fail he will nuke, freeze, stone, 
//= stun, or make you fall asleep.
//= The prize is configurable and triggered by the NPC.
//===== Additional Comments: =================================
//= 1.0 First release.
//= 1.1 Using 'switch rand' instead.

prontera,156,219,4	script	咯咯雞::Cluckers	800,{

	set .@GMAccess,80;	// GM level required to access NPC

	if (.startcluck) {
		specialeffect2 EF_HIT3;
		switch(rand(15)) {
			case 0: npctalk "Cluckers : CLUUUUUUCK!!!"; unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; emotion ET_HUK; break;
			case 1: npctalk "Cluckers : Cluuuuuck!~"; break;
			case 2: unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; break;
			case 3: sc_start SC_Freeze,10000,0; break;
			case 4: npctalk "Cluckers : CLUUUUUUUUUCK!!!"; unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; emotion ET_HUK; break;
			case 5: sc_start SC_Sleep,10000,0; break;
			case 6: sc_start SC_Stone,10000,0; emotion ET_KIK; break;
			case 7: npctalk "Cluckers : CLUUUUUUCK!!!"; unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; emotion ET_HUK; break;
			case 8: npctalk "Cluckers : Cluck! CLUUUCK!!"; unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; emotion ET_HUK; break;
			case 9: sc_start SC_Stun,10000,0; break;
			case 10: sc_start SC_Sleep,10000,0; emotion ET_KIK; break;
			case 11: npctalk "Cluckers : Cluck! Cluck!"; break;
			case 12: sc_start SC_Stun,10000,0; break;
			case 13: unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; break;
			default:
				if (rand(50) < 4) {
					npctalk "咯咯雞 : 汪！...........";
					specialeffect2 EF_SPHERE;
					announce "[咯咯咯！轟！] " + strcharinfo(0) + " 擠出了獎品！做得好！",0;
					getitem $cluck_item_id,$cluck_item_amount;
					set .startcluck,0;
				} else {
					npctalk "咯咯雞 : 咯咯！咯咯咯！！";
					unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1;
				}
				break;
		}
		end;
	}
	mes "[咯咯雞]";
	if (getgmlevel() >= .@GMAccess) {
		mes "咯咯！咯咯咯？ ^FF0000~嗨 GM " + strcharinfo(0) + "，今天要來玩嗎？~^000000";
		mes "咯咯……咯咯！ ^FF0000~快告訴我要做什麼！~^000000";
		next;
		while(1) switch(select("開始活動:查看獎品:設定獎品:今天不玩了，咯咯雞")) {
			case 1:
				mes "[咯咯雞]";
				mes "咯咯！ ^FF0000~沒問題！~^000000";
				emotion ET_OK;
				close2;
				announce "[咯咯咯！轟！] 即將在普隆德拉開始！",bc_blue;
				initnpctimer;
				end;
			case 2:
				mes "[咯咯雞]";
				mes "咯咯，咯咯咯……咯咯！ ^FF0000~目前的獎品是^000000 ^008000"+ $cluck_item_amount +" "+ getitemname($cluck_item_id) +".^000000";
				next;
				break;
			case 3:
				mes "[咯咯雞]";
				mes "咯咯？咯咯？？？ ^FF0000~贏家的獎品要設定什麼？請輸入道具 ID。~^000000";
				input $cluck_item_id;
				next;
				mes "[咯咯雞]";
				mes "咯咯？咯咯？ ^FF0000~這個道具要發出幾個？~^000000";
				input $cluck_item_amount;
				next;
				mes "[咯咯雞]";
				mes "咯咯……？咯咯。 ^FF0000~所以，獎品是^000000 ^008000"+ $cluck_item_amount +" "+ getitemname($cluck_item_id) +"^000000？ ^FF0000很好。~^000000";
				emotion ET_OK;
				next;
				break;
			case 4:
				mes "[咯咯雞]";
				mes "咯咯咯……";
				close;
		}
	}
	mes "咯咯！咯咯咯？";
	mes "咯咯……";
	close;

OnTimer10000:
	announce "想跟瘋狂的雞一起玩，就快到噴水池後面來！",bc_blue;
	end;
OnTimer20000:
	announce "咯咯雞吃掉了我的一個道具！我太害怕了，不敢去把它拿回來！",bc_blue;
	end;
OnTimer30000:
	announce "點擊發狂的雞，試著把道具擠出來，運氣好就能贏！準備好了嗎？",bc_blue;
	end;
OnTimer40000:
	announce "開始！點擊雞隻就能拿到獎品！",bc_blue;
	stopnpctimer;
	if (!$cluck_item_id) set $cluck_item_id,512;
	if (!$cluck_item_amount) set $cluck_item_amount,1;
	set .startcluck,1;
	end;
}
