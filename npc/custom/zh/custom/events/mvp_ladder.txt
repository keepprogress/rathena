prontera,164,171,3	script	MvP Ladder Warper	56,{
	mes "[MVP天梯傳送員]";
	mes "嗯……你想玩 MVP 天梯遊戲嗎？";
	next;
	switch(select("好，開始吧！:說明。:給我看最佳紀錄。:不要。")) {
	case 1:
		break;
	case 2:
		mes "[MVP天梯傳送員]";
		mes "在這個遊戲中，你的隊伍必須依照由弱到強的順序，逐一擊殺每一隻 MVP 怪物。";
		if ( .finish_item_amount )
			mes "如果你的隊伍能完成 MVP 天梯遊戲，每位成員都會獲得 "+ callfunc("F_InsertPlural", .finish_item_amount, getitemname( .finish_item_id )) +".";
		if ( .register_cost )
			mes "但入場費是 "+ callfunc( "F_InsertComma", .register_cost ) +" Zeny。";
		next;
		mes "[MVP天梯傳送員]";
		mes "若無法在 "+ .timeout +" 分鐘內完成，或全隊陣亡，就算輸掉遊戲。";
		mes "祝你好運！";
		close;
	case 3:
		mes "[MVP天梯傳送員]";
		if ( !$mvpladdderparty_time ) {
			mes "目前還沒有人完成這個遊戲。";
			close;
		}
		mes "目前最佳紀錄為";
		mes "[ "+( $mvpladdderparty_time / 60 )+" 分 "+( $mvpladdderparty_time % 60 )+" 秒 ]";
		mes "由隊伍 ^FF0000"+ $mvpladdderparty_name$ +"^000000.";
		.@size = getarraysize( $mvpladderparty_member$ );
		for ( .@i = 0; .@i < .@size; .@i++ )
			mes "^000000"+ ( .@i +1 ) +"。 ^0000FF"+ $mvpladderparty_member$[.@i];
		if ( getgmlevel() < .gmlvlreset ) close;
		next;
		if ( select( "關閉。", "重設紀錄。" ) == 1 ) close;
		if ( select( "算了。", "我真的想重設它。" ) == 1 ) close;
		$mvpladdderparty_time = 0;
		$mvpladdderparty_name$ = "";
		deletearray $mvpladderparty_member$[.@i];
		mes "[MVP天梯傳送員]";
		mes "紀錄已成功重設。";
		close;
	case 4:
		mes "[MVP天梯傳送員]";
		mes "等你的實力足以完成遊戲時，再回來吧。";
		close;
	}
	if ( !getcharid(1) ) {
		mes "[MVP天梯傳送員]";
		mes "你必須組成隊伍才能參加。";
		close;
	}
	if ( is_party_leader() == false ) {
		mes "[MVP天梯傳送員]";
		mes "只有隊長可以報名。";
		close;
	}
	.@origin = getcharid(3);
	getpartymember getcharid(1), 1;
	getpartymember getcharid(1), 2;
	for ( .@i = 0; .@i < $@partymembercount; .@i++ ) {
		if ( isloggedin( $@partymemberaid[.@i], $@partymembercid[.@i] ) ) {
			attachrid $@partymemberaid[.@i];
			if ( strcharinfo(3) == strnpcinfo(4) )
				.@online++;
		}
	}
	attachrid .@origin;
	if ( $@partymembercount != .register_min ) {
		mes "[MVP天梯傳送員]";
		mes "你必須組成恰好 "+ .register_min +" 名成員的隊伍才能參加。";
		close;
	}
	else if ( .@online != .register_min )  {
		mes "[MVP天梯傳送員]";
		mes "你的隊伍必須有 "+ .register_min +" 名成員在線上，且位於地圖 '"+ strnpcinfo(4) +"'.";
		close;
	}
	else if ( .register_cost && Zeny < .register_cost ) {
		mes "[MVP天梯傳送員]";
		mes "你的Zeny不夠。等你的Zeny足夠了再回來。";
		close;
	}
	else if ( .party_id ) {
		mes "[MVP天梯傳送員]";
		mes "很抱歉，目前有隊伍正在進行遊戲，請稍候，等他們結束後再來。";
		mes "謝謝。";
		close;
	}
	Zeny -= .register_cost;
	announce "隊伍 ["+ strcharinfo(1) +"] 已開始 MVP 天梯遊戲。", bc_all;
	set .party_id, getcharid(1);
	set .@time_enter, gettimetick(2);
	for ( .@i = 0; .@i < $@partymembercount; .@i++ ) {
		if ( isloggedin( $@partymemberaid[.@i], $@partymembercid[.@i] ) ) {
			attachrid $@partymemberaid[.@i];
			if ( strcharinfo(3) == strnpcinfo(4) ) {
				announce "你有 "+ .timeout +" 分鐘內完成 "+ .totalround +" 回合。", bc_self;
				.@name$[.@c] = strcharinfo(0);
				.@c++;
			}
		}
	}
	cleanmap .eventmap$;
	warpparty .eventmap$, 0,0, .party_id, strnpcinfo(4);
	donpcevent strnpcinfo(0)+"::OnMvpDead";
	sleep .timeout * 60000;
	if ( .round == .totalround +1 ) {
		getpartymember .party_id, 1;
		getpartymember .party_id, 2;
		mapannounce .eventmap$, "恭喜……你成功擊敗了所有 MVP！", bc_map;
		for ( .@i = 0; .@i < $@partymembercount; .@i++ ) {
			if ( isloggedin( $@partymemberaid[.@i], $@partymembercid[.@i] ) ) {
				attachrid $@partymemberaid[.@i];
				if ( strcharinfo(3) == .eventmap$ )
					{ getitem .finish_item_id, .finish_item_amount; #CASHPOINTS += 50; dispbottom "獲得 50 商城點數，目前共 " + #CASHPOINTS + " 點。"; }
			}
		}
		set .@timeused, gettimetick(2) - .@time_enter;
		if ( .bonus_item_amount && .@timeused < .bonus_time * 60 ) {
			mapannounce .eventmap$, "你在時限內完成了天梯，獲得額外獎勵道具！時限 "+ .bonus_time +" 分鐘。", bc_map;
			for ( .@i = 0; .@i < $@partymembercount; .@i++ ) {
				if ( isloggedin( $@partymemberaid[.@i], $@partymembercid[.@i] ) ) {
					attachrid $@partymemberaid[.@i];
					if ( strcharinfo(3) == .eventmap$ )
						{ getitem .bonus_item_id, .bonus_item_amount; #CASHPOINTS += 30; dispbottom "獲得 30 商城點數，目前共 " + #CASHPOINTS + " 點。"; }
				}
			}
		}

		if ( !$mvpladdderparty_time || .@timeused < $mvpladdderparty_time ) {
			mapannounce .eventmap$, "而且你打破了紀錄！[ "+( .@timeused / 60 )+" 分 "+( .@timeused % 60 )+" 秒 ]", bc_map;
			set $mvpladdderparty_time, .@timeused;
			set $mvpladdderparty_name$, getpartyname( .party_id );
			copyarray $mvpladderparty_member$, .@name$, .register_min;
		}
		else
			mapannounce .eventmap$, "使用時間 [ "+( .@timeused / 60 )+" 分 "+( .@timeused % 60 )+" 秒 ]", bc_map;
		sleep 10000;
		announce "隊伍 ["+ getpartyname( .party_id ) +"] 已完成 MVP 天梯遊戲！", bc_all;
	}
	else
		announce "隊伍 ["+ getpartyname( .party_id ) +"] 未能完成 MVP 天梯遊戲。", bc_all;
	mapwarp .eventmap$, .map$, .x, .y;
	killmonsterall .eventmap$;
	.party_id = .round = 0;
	end;

OnMvpDead:
	getpartymember .party_id, 1;
	getpartymember .party_id, 2;
	.round++;
	if ( .round >= 2 && .round != .totalround +1 && .round_item_amount ) {
		for ( .@i = 0; .@i < $@partymembercount; .@i++ ) {
			if ( isloggedin( $@partymemberaid[.@i], $@partymembercid[.@i] ) ) {
				attachrid $@partymemberaid[.@i];
				if ( strcharinfo(3) == .eventmap$ )
					{ getitem .round_item_id, .round_item_amount; #CASHPOINTS += 5; dispbottom "獲得 5 商城點數，目前共 " + #CASHPOINTS + " 點。"; }
			}
		}
	}
	if ( .round == .totalround +1 ) {
		awake strnpcinfo(0);
		end;
	}
	else if ( .round == .totalround )
		mapannounce .eventmap$, "最後一回合將於 "+ .delay +" 秒後開始……", bc_map;
	else
		mapannounce .eventmap$, "開始第 "+ .round +" 回合，在 "+ .delay +" 秒後開始……", bc_map;
	sleep .delay * 1000;
	if ( .mvpid[.round] == 1646 )  // pick random Bio3 MVP
		.mvpid[.round] = rand(1646,1651);
	monster .eventmap$,0,0, "--ja--", .mvpid[.round], 1, strnpcinfo(0)+"::OnMvpDead";
	mapannounce .eventmap$, getmonsterinfo( .mvpid[.round], MOB_NAME ) +" 已出現！", bc_map|bc_blue;
	end;

OnPCLogoutEvent:
	if ( hp > 0 )
		.@less_one = 1;
	else
		end;
OnPCDieEvent:
	if ( strcharinfo(3) != .eventmap$ || !getcharid(1) ) end;
	if ( getcharid(1) != .party_id ) end;
	getpartymember .party_id, 1;
	getpartymember .party_id, 2;
	for ( .@i = 0; .@i < $@partymembercount; .@i++ ) {
		if ( isloggedin( $@partymemberaid[.@i], $@partymembercid[.@i] ) ) {
			attachrid $@partymemberaid[.@i];
			if ( strcharinfo(3) == .eventmap$ && hp > 0 )
				.@alive++;
		}
	}
	if ( .@less_one )
		.@alive--;
	if ( !.@alive ) {
		mapannounce .eventmap$, "隊伍全滅！", bc_map;
		sleep 10000;
		awake strnpcinfo(0);
	}
	end;

OnInit:
//	Configurations -----------------------------------------------------

	// Time limit (in minutes)
	// When time runs out, all players inside the room will be kicked out.
	// Do NOT set this to zero!
	set .timeout, 60;

	// entrance fee (in Zeny)
	set .register_cost, 100000;

	// exact amount of party members needed to start the game
	set .register_min, 2;

	// id of each mvp. you can add more
	setarray .mvpid[1],
		1086,//	Golden Thief Bug	64
		1115,//	Eddga				65
		1150,//	Moonlight Flower	67
		1159,//	Phreeoni			69
		1112,//	Drake				70
		1583,//	Tao Gunka			70
		1492,//	Incantation Samurai	71
		1046,//	Doppelgangger		72
		1252,//	Garm				73
		1418,//	Evil Snake Lord		73
		1059,//	Mistress			74
		1190,//	Orc Lord			74
		1087,//	Orc Hero			77
		1251,//	Knight of Windstorm	77
		1038,//	Osiris				78
		1658,//	Ygnizem				79
		1272,//	Dark Lord			80
		1871,//	Fallen Bishop		80
		1039,//	Baphomet			81
		1147,//	Maya				81
		1785,//	Atroce				82
		1389,//	Dracula				85
		1630,//	Bacsojin			85
		1885,//	Gorynych			85
		1623,//	RSX 0806			86
		1511,//	Amon Ra				88
		1688,//	Lady Tanee			89
		1768,//	Gloom Under Night	89
		1719,//	Datale				90
		1734,//	Kiel D-01			90
		1157,//	Pharaoh				93
		1373,//	Lord of Death		94
		1312,//	Turtle General		97
		1779,//	Ktullanux			98
		1874,//	Beelzebub			98
		1646,// Bio3 placeholder    99
		1708,//	Thanatos			99
		1751,//	Valkyrie Randgris	99
		1832;//	Ifrit				99

	// number of rounds (default: 39)
	set .totalround, getarraysize(.mvpid) -1;

	// item reward for completing each round
	set .round_item_id, 512;
	set .round_item_amount, 1;

	// item reward for completing the entire ladder
	set .finish_item_id, 501;
	set .finish_item_amount, 10;

	// bonus reward if ladder completed within a certain time (in minutes)
	set .bonus_time, 45; // if completed within 45 minutes, this reward is given
	set .bonus_item_id, 504;
	set .bonus_item_amount, 10;

	// time delay between rounds, in seconds (default: 3)
	set .delay, 5;

	// minimum GM level to reset the best record
	set .gmlvlreset, 99;

	// event map
	set .eventmap$, "guild_vs2-2";

	// mapflag configuration
	setarray .@mapflag,
		mf_nowarp,
		mf_nowarpto,
		mf_nosave,
		mf_nomemo,
		mf_noteleport,
//		mf_nopenalty, // disable exp loss
		mf_noreturn,
//		mf_nobranch,
//		mf_nomobloot, // disable monster drop loots,
//		mf_nomvploot, // 2 of these
		mf_partylock;

//	Config Ends --------------------------------------------------------------

	mapannounce .eventmap$, "管理員已重新啟動伺服器，請重新報名。造成您的不便，敬請見諒。", bc_map;
	getmapxy .map$, .x, .y, BL_NPC;
	mapwarp .eventmap$, .map$, .x, .y;
	killmonsterall .eventmap$;
	.@size = getarraysize( .@mapflag );
	for ( .@i = 0; .@i < .@size; .@i++ )
		setmapflag .eventmap$, .@mapflag[.@i];
	end;
}
