//===== rAthena Script =======================================
//= Reset NPC
//===== Description: =========================================
//= Resets skills, stats, or both.
//===== Additional Comments: =================================
//= 1.0 First Version
//= 1.1 Optimized for the greater good. [Kisuka]
//= 1.2 Cleaning [Euphy]
//= 1.3 All statuses removed upon skill reset. [Euphy]
//= 1.4 Compressed Script, Added limit use option [Stolao]
//=	Changed set -> setarray, Improved text with F_InsertPlural
//= 1.5 Added sc_end_class to reset related status changes [sader1992]
//============================================================
prontera,150,193,4	script	重置少女::Reset Girl	124,{
	//	 		Skills,	Stats,	Both,	Limit
	setarray .@Reset,	5000,	5000,	9000,	0;
	mes "[重置少女]";
	if(.@Reset[3] && reset_limit > .@Reset[3]) {
		mes "很抱歉，你最多只能重置 "+callfunc("F_InsertPlural",.@Reset[3],"time")+" 。";
		close;
	}
	mes "我是重置少女。";
	mes "重置能力值: "+ callfunc("F_InsertComma",.@Reset[1]) +"z";
	mes "重置技能: "+ callfunc("F_InsertComma",.@Reset[0]) +"z";
	mes "全部重置: "+ callfunc("F_InsertComma",.@Reset[2]) +"z";
	if(.@Reset[3]) mes "You may only reset "+callfunc("F_InsertPlural",.@Reset[3],"time")+", so use "+((.@Reset[3]>1)?"them":"it")+" wisely.";
	mes "請選擇你要的服務:";
	next;
	set .@i,(select("^FF3355重置技能:重置能力值:全部重置^000000:取消"));
	if(.@i > 3) close;
	mes "[重置少女]";
	if (Zeny < .@Reset[.@i-1]) {
		mes "很抱歉，你的 Zeny 不足。";
		close;
	}
	if(.@Reset[3]){
		mes "你最多只能重置 "+callfunc("F_InsertPlural",.@Reset[3],"time")+" ，確定要重置嗎？";
		if(select("讓我想想:沒問題") == 1) close;
	}
	set Zeny, Zeny-.@Reset[.@i-1];
	if(.@i&1){
		sc_end_class;
		ResetSkill;
	}
	if(.@i&2) ResetStatus;
	mes "給你！";
	if(.@Reset[3]) set reset_limit,reset_limit + 1;
	close;
}
