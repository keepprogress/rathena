				mes "[傭兵管理員]";
				mes "很抱歉，你的基本";
				mes "等級還不夠高";

				mes "達到基本等級 " + .@BaseLevel + ".";

				mes "很抱歉，您";

		case 4:

			mes "十級傭兵是";
			mes "我們最好的傭兵，";

-	script	Mercenary Merchant#dummy::MercMerchant	-1,{
	mes "[傭兵道具商人]";
	mes "你好，我販售";
	mes "傭兵能使用的";
	mes "道具。你有特別";
	mes "想找的";
	mes "東西嗎？";
	next;
	setarray .@item, 12184,	12185,	12241,	12242,	12243;
	setarray .@cost, 2500,	5000,	800,	1500,	3000;
	for (.@size = getarraysize(.@item); .@i < .@size; .@i++)
		.@menu$ = .@menu$ + getitemname(.@item[.@i]) + ":";
	.@m = select(.@menu$)-1;
	.@str$ = .@cost[.@m] + "";
	.@len = getstrlen(.@str$);
	.@str_cost$ = (.@len <= 3)? .@str$ : insertchar(.@str$,",",.@len-3);
	mes "[傭兵道具商人]";
	mes getitemname(.@item[.@m]);
	mes "每個售價 " + .@str_cost$ + " Zeny。";
	mes "你要買多少？";
	next;
	input .@input;
	if (!.@input) {
		mes "[傭兵道具商人]";
		mes "你改變主意了嗎？";
		mes "好的，隨時回來";
		mes "找我，";
		mes "想買傭兵藥水都可以。";
		close;
	}
	else if (.@input < 0 || .@input > 10000) {
		mes "[傭兵道具商人]";
		mes "很抱歉，您";
		mes "最多一次只能購買";
		mes "10,000 瓶這種藥水。";
		mes "請輸入";
		mes "1 到 10,000 之間的數字。";
		close;
	}
	else if (set(.@total_po, .@input * .@cost[.@m]) > Zeny) {
		mes "[傭兵道具商人]";
		mes "很抱歉，您";
		mes "你的 Zeny 不足以";
		mes "買這麼多藥水。嗯，";
		mes "我會在這裡，";
		mes "等你準備好要";
		mes "為傭兵購買東西。";
		close;
	}
	else if (!checkweight(.@item[.@m], .@input)) {
		mes "[傭兵道具商人]";
		mes "如果我把這麼多";
		mes "藥水交給你，你";
		mes "將無法全部隨身攜帶。";
		mes "請先";
		mes "在背包裡騰出一些空間，";
		mes "再回來。";
		close;
	}
	mes "[傭兵道具商人]";
	if (.@input == 1) 
		mes "這是你的 " + getitemname(.@item[.@m]) + ".";
	else {
		mes "給你，就是這個";
		mes callfunc("F_InsertPlural",.@input,getitemname(.@item[.@m])) + ".";
	}
	mes "謝謝，請";
	mes "在需要更多";
	mes "傭兵藥水時再來。";
	Zeny = Zeny - .@total_po;
	getitem .@item[.@m], .@input;
	close;
}

// Mercenary Switch NPCs
//============================================================
-	script	Mercenary_Switch	-1,{

	setarray .@name$, "Spear","Sword","Bow";
	.@npc$ = strnpcinfo(2);
	for (.@size = getarraysize(.@name$); .@type < .@size; .@type++)
		if (.@npc$ == .@name$[.@type]) break;

	mes "[檢查員]";
	mes "請輸入密碼。";
	next;
	.@i = callfunc("F_GM_NPC",1854,0,0,10000);
	mes "[檢查員]";
	if (.@i == -2)
		mes "錯誤。";
	else if (.@i == 0)
		mes "數字錯誤";
	else if (.@i == 1) {
		mes "請選擇。";
		next;
		switch( select( "關閉傭兵 NPC","開啟傭兵 NPC" ) ) {
		case 1:
			mes "NPC 已關閉。";
			disablenpc "Mercenary Manager#" + .@name$[.@type];
			disablenpc "Mercenary Merchant#" + .@name$[.@type];
			break;
		case 2:
			mes "NPC 已開啟。";
			enablenpc "Mercenary Manager#" + .@name$[.@type];
			enablenpc "Mercenary Merchant#" + .@name$[.@type];
			break;
		}
	}
	close;
}
