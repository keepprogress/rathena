	mes "[健身達人]";
	mes "嘿，你好。現在的人";
	mes "已不像從前那樣";
	mes "經常運動了。就算你是靠戰鬥吃飯的，";
	mes "你的身體可能";
	mes "在某些部位";
	mes "變得鬆軟無力。你懂我的意思吧？";
	next;
	mes "[健身達人]";
	mes "嘿，跟我一起訓練，我保證";
	mes "你能";
	mes "舉起並搬運更多你的";
	mes "東西。只要把你的";
	mes "^FF0000健身房通行證^000000 每次都要給，";
	mes "我們就可以開始了。";
	next;
	mes "[健身達人]";
	mes "但別太興奮:";
	mes "不管我帶你訓練";
	mes "多少，你都";
	mes "可能會過度訓練。你聽說過";
	mes "有人練得太壯嗎？";
	mes "那是因為他們已經死了。懂了吧？";
	next;
	mes "[健身達人]";
	mes "我可以跟你說，這樣很安全";
	mes "你只要認真";

	mes "道具負重能力 ^FF000010 次^000000。";

	next;
	switch(select("是:否:嗯，我的鍛鍊效果已經消失了。")) {
	case 1:
		if (gympassmemory < 10) {
			set .@add_carry,gympassmemory + 1;
			set .@remain_carry,10 - .@add_carry;
			if (countitem(7776) > 0) {
				mes "[健身達人]";
				mes "哦，太好了，我看到你";
				mes "帶來了你的健身房通行證。";
				mes "好的，照我做的做，";
				mes "試著感受那股燒灼感。";
				mes "準備好了嗎？我們開始吧。";
				next;
				specialeffect2 EF_EARTHSPIKE;
				next;
				specialeffect2 EF_DEVIL;
				next;
				specialeffect2 EF_COIN;
				next;
				specialeffect2 EF_SIGHTRASHER;
				next;
				mes "[健身達人]";
				mes "好了，你現在應該能";
				mes "帶更多東西了。";
				mes "我們來看看，可以提升";
				mes "你的道具負重能力";
				mes "^FF00000" + .@remain_carry + "^000000 次，如果我們繼續";
				mes "像這樣一起訓練。";
				delitem 7776,1; //Max_Weight_Up_Scroll
				set gympassmemory,.@add_carry;
				skill "ALL_INCCARRY",.@add_carry,SKILL_PERM_GRANT;
				close;
			}
			else {
				mes "[健身達人]";
				mes "老兄，我跟你說過什麼？";

				mes "道具負重肌肉。";
				close;
			}
		}
		else {
			mes "[健身達人]";
			mes "老兄，我覺得我們沒辦法";
			mes "再增強你的道具負重";

		mes "[健身達人]";

			mes "發生什麼事了？";

			mes "肌肉就這樣萎縮了？幸好";

			specialeffect2 EF_EARTHSPIKE;
			next;
			specialeffect2 EF_DEVIL;
			next;
			specialeffect2 EF_COIN;
			next;
			specialeffect2 EF_SIGHTRASHER;
			next;
			mes "[健身達人]";
			mes "怎麼樣？";
			mes "你的道具負重";
			mes "肌肉長回來了，";
			mes "就是這樣！盡量別";
			mes "再退縮了，好嗎？";
			skill "ALL_INCCARRY",gympassmemory,SKILL_PERM_GRANT;
			close;
		}
		else {
			mes "[健身達人]";
			mes "呃……";
			mes "我們之前沒有";
			mes "一起鍛鍊過。";
			mes "我很確定。";
			close;
		}
	}
}
