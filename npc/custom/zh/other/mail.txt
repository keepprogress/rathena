// Floating mailbox npc that all NPCs duplicate from.
//============================================================
-	script	Mailbox#dummy::MailBox	-1,{
	mes "[信箱]";
	mes "要使用信箱服務，";
	mes "你需要支付 130 Zeny。";
	mes "你要使用這項服務嗎？";
	next;
	switch(select("是。:否。")) {
	case 1:
		mes "[信箱]";
		if (Zeny < 130) {
			mes "很抱歉，你的金錢不足。";
			mes "要使用信箱服務，";
			mes "你需要支付 130 Zeny。";
			close;
		}
		mes "謝謝，歡迎再來。";
		set Zeny, Zeny - 130;
		close2;
		openmail;
		end;
	case 2:
		mes "[信箱]";
		mes "謝謝，歡迎再來。";
		close;
	}
}
