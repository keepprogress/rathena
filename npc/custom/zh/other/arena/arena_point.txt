//= Point exchanger: Turbo track points -> Arena Points
//===== Additional Comments: =================================

//============================================================

prt_are_in,103,11,3	script	競技點數管理員::Arena Point Manager	732,{
	mes "[競技點數管理員]";
	mes "希望您";
	mes "在競技場玩得愉快。若您已";
	mes "於";
	mes "阿爾德巴蘭的競速場地獲得競速場點數，我能";
	mes "將其兌換為競技點數。";
	next;
	switch(select("點數查詢:兌換點數:^660000兌換資訊^000000")) {
	case 1:
		mes "[競技點數管理員]";
		mes "" + strcharinfo(0) + ",";
		mes "您目前擁有";
		mes "" + arena_point + " 競技點數";
		mes "和 " + tt_point + " 競速場點數。";
		close;
	case 2:
		mes "[競技點數管理員]";
		mes "" + strcharinfo(0) + ",";
		mes "您目前擁有";
		mes "" + arena_point + " 競技點數";
		mes "和 " + tt_point + " 競速場點數。";
		next;
		mes "[競技點數管理員]";
		mes "請從以下";
		mes "競速場點數兌換競技點數的";
		mes "兌換方案中選擇。請注意";
		mes "若您一次兌換超過10";
		mes "點競速場點數，只能以^4D4DFF10的倍數^000000兌換。";
		next;
		switch(select("2 TP -> 1 AP:4 TP -> 2 AP:6 TP -> 3 AP:8 TP -> 4 AP:10 TP以上:取消")) {
		case 1:
			callsub S_ExchangePoints,28999,2,1;
		case 2:
			callsub S_ExchangePoints,28998,4,2;
		case 3:
			callsub S_ExchangePoints,28997,6,2;
		case 4:
			callsub S_ExchangePoints,28996,8,4;
		case 5:
			mes "[競技點數管理員]";
			mes "請輸入";
			mes "您想兌換的次數";
			mes "10點競速場點數為";
			mes "競技點數。可輸入的最大";
			mes "值為20。";
			mes "若要取消，請輸入''^3355FF0^000000''。";
			next;
			input .@input;
			if (.@input <= 0) {
				mes "[競技點數管理員]";
				mes "您有";
				mes "已取消";
				mes "此服務。";
				close;
			}
			else if (.@input > 20) {
				mes "[競技點數管理員]";
				mes "您的要求超過";
				mes "上限。請";
				mes "輸入不大於20的數值。";
				close;
			}
			else {
				set .@want_point1,10 * .@input;
				set .@want_point,5 * .@input;
				set .@my_arena_all,arena_point + .@want_point;
				set .@my_turbo_all,tt_point - .@want_point1;
				if (.@my_arena_all > 28999) {
					mes "[競技點數管理員]";
					mes "您將超過";
					mes "最大數量";
					mes "競技點數上限，若我們繼續";
					mes "進行您此次的";
					mes "競速場點數兌換，您將無法";
					mes "擁有超過29,000點競技點數。";
					next;
					mes "[競技點數管理員]";
					mes "請先花費部分";
					mes "競技點數，再";
					mes "再次使用此服務。";
					mes "感謝您的惠顧。";
					close;
				}
				if (.@my_turbo_all < 0) {
					mes "[競技點數管理員]";
					mes "很抱歉，";
					mes "您沒有足夠的";
					mes "競速場點數";
					mes "來進行此次競技";
					mes "點數兌換。";
					close;
				}
				else {
					mes "[競技點數管理員]";
					mes "你已兌換";
					mes "10點競速場點數";
					mes "為5點競技點數 " + .@input + " 次。";
					mes "總計 " + .@want_point1 + " 點競速場點數已兌換為";
					mes "" + .@want_point +" 點競技點數。";
					set tt_point, .@my_turbo_all;
					set arena_point, .@my_arena_all;
					next;
					mes "[競技點數管理員]";
					mes "" + strcharinfo(0) + ",";
					mes "你現在擁有";
					mes "^4682B4" + arena_point + "^000000 競技點數";
					mes "與^00688B" + tt_point + "^000000 點競速場點數。";
					mes "感謝您的惠顧。";
					close;
				}
			}
		case 6:
			mes "[競技點數管理員]";
			mes "您有";
			mes "已取消";
			mes "此服務。";
			close;
		}
	case 3:
		// Again. no dialog for this option.
		close;
	}
	close;

S_ExchangePoints:
	if (arena_point > getarg(0) ) {
		mes "[競技點數管理員]";
		mes "您將超過";
		mes "最大數量";
		mes "競技點數上限，若我們繼續";
		mes "進行您此次的";
		mes "競速場點數兌換，您將無法";
		mes "擁有超過29,000點競技點數。";
		next;
		mes "[競技點數管理員]";
		mes "請先花費部分";
		mes "競技點數，再";
		mes "再次使用此服務。";
		mes "感謝您的惠顧。";
		close;
	}
	.@track_point = getarg(1);
	if (tt_point >= .@track_point) {
		.@arena_point = getarg(2);
		mes "[競技點數管理員]";
		mes "" + strcharinfo(0) + ",";
		mes "你已兌換 "+ .@track_point +" 競速場";
		mes "點數為 "+ .@arena_point +" 點競技點數。";
		tt_point = tt_point - .@track_point;
		arena_point = arena_point + .@arena_point;
		next;
		mes "[競技點數管理員]";
		mes "" + strcharinfo(0) + ",";
		mes "你現在擁有";
		mes "^4682B4" + arena_point + "^000000 競技點數";
		mes "與^00688B" + tt_point + "^000000 點競速場點數。";
		mes "感謝您的惠顧。";
		close;
	}
	else {
		mes "[競技點數管理員]";
		mes "很抱歉，您身上沒有";
		mes "足夠的競速場點數。";
		mes "你需要至少2點競速場點數，才能使用此兌換服務。";
		close;
	}
}
