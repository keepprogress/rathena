	} else if (.@reward > 1500) {
		mes "[吉伍]";
		mes "您已超過最大容量。";
		close;
	}
	if (arena_point < (.@reward * .@arena_point)) {
		mes "[吉伍]";
		mes "您的競技場點數不足。";
		mes "請確認您擁有的競技場點數總數。";
		close;
	}
	arena_point = arena_point - (.@reward * .@arena_point);
	getitem .@item_id,.@reward * .@amount;
	mes "[吉伍]";
	mes "謝謝，歡迎再來。";
	close;
}
