function	script	F_AuctionWarper	{
	mes "[拍賣場嚮導]";
	mes "您好，您是否";
	mes "想進入";
	mes "拍賣場嗎？";
	next;
	if (select("是:否") == 1) {
		.@num = getarg(0);
		if (.@num == 1 || .@num == 4) {
			mes "[拍賣場嚮導]";
			mes "太好了！那麼，";
			mes "希望您玩得愉快，";
			mes "並享受拍賣的樂趣~";
		} else {
			mes "[拍賣場嚮導]";
			mes "祝您拍賣愉快。";
		}
		close2;
		switch(.@num) {
		case 1:
			warp "auction_01",179,53; end;
		case 2:
			warp "auction_01",21,43; end;
		case 3:
			warp "auction_02",151,23; end;
		case 4:
			warp "auction_02",43,24; end;
		}
	}
	mes "[拍賣場嚮導]";
	mes "那好吧，";
	mes "下次見。";
	mes "如果您改變";
	mes "心意，請再";
	mes "來享受拍賣的樂趣~";
	close;
}
