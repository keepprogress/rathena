e_tower,81,105,0	script	塔之守護石::Tower Protection Stone	406,{

	set .@party_id,getcharid(1);
	set .@md_name$,"Endless Tower";

	set .@etower_timer,checkquest(60200,PLAYTIME); // 1 week
	set .@etower_timer2,checkquest(60201,PLAYTIME); // 4 hours

	if (!instance_check_party(.@party_id,2)) {
		mes "請組成或加入一個超過 1 人的隊伍後再試一次。";
		close;
	}
	switch(.@etower_timer) {
	case -1:
		if (is_party_leader() == true) {
			mes "已確認隊伍成立。要預約進入無盡之塔嗎？";
			next;
			switch(select("生成副本 "+.@md_name$+":進入副本:返回艾爾貝塔:取消")) {
			case 1:
				if (instance_create(.@md_name$) < 0) {
					mes "隊伍名稱: "+ getpartyname(.@party_id);
					mes "隊長: "+strcharinfo(0);
					mes "^0000ff"+.@md_name$+" ^000000- 預約失敗！";
					close;
				}
				mes "^0000ff"+.@md_name$+"^000000 - 嘗試預約";
				mes "完成預約後，你必須與後方的 NPC 對話，並選擇「進入副本」選單才能進入副本。";
				close;
			case 2:
				callsub L_Enter,0,1;
			case 3:
				mes "我會把你送到艾爾貝塔。";
				close2;
				warp "alberta",223,36;
				end;
			case 4:
				close;
			}
		}
		switch(select("進入 "+.@md_name$+":返回艾爾貝塔:取消")) {
		case 1:
			callsub L_Enter,1,1;
		case 2:
			mes "我會把你送到艾爾貝塔。";
			close2;
			warp "alberta",223,36;
			end;
		case 3:
			end;
		}
	case 0:
	case 1:
		if (.@etower_timer2 < 2 && getcharid(1) == etower_partyid) {
			mes "如果副本已經生成，你就可以進入。 ";
			next;
			switch(select("進入 "+.@md_name$+":返回艾爾貝塔:取消")) {
			case 1:
				callsub L_Enter,0,0;
			case 2:
				mes "我會把你送到艾爾貝塔。";
				close2;
				warp "alberta",223,36;
				end;
			case 3:
				close;
			}
		} else {
			set .@dun_lim_time,etower_timer+604800; // 1 week
			// set .@dun_lim_time2,etower_timer+14400; // 4 hours
			set .@dun_cur_time,gettimetick(2);
			set .@dun_ent_t,(.@dun_lim_time - .@dun_cur_time);
			set .@dun_h,(.@dun_ent_t / 3600);
			set .@dun_m,(.@dun_ent_t - (.@dun_h * 3600)) / 60;
			set .@dun_s,.@dun_ent_t - ((.@dun_h * 3600) + (.@dun_m * 60));

			mes "由於塔的後遺症，你現在無法進入副本， " + .@dun_h + "小時 " + .@dun_m + "分鐘 " + .@dun_s + "秒後才能進入下一個副本。";
			next;
			mes "這裡很危險。我會把你送到艾爾貝塔。";
			close2;
			warp "alberta",223,36;
			end;
		}
	case 2:
		set etower_timer,0;
		erasequest 60200;
		erasequest 60201;
		mes "^0000ff與無盡之塔相關的紀錄與後遺症已被清除。你可以再次生成並進入無盡之塔。^000000";
		close;
	}

L_Enter:
	switch(instance_enter("Endless Tower")) {
	case IE_OTHER:
		mes "發生了未知的錯誤。";
		close;
	case IE_NOINSTANCE:
		mes "紀念副本無盡之塔不存在。";
		mes "隊長尚未生成副本。";
		close;
	case IE_NOMEMBER:
		mes "組成隊伍後才能進入副本。";
		close;
	case IE_OK:
		mapannounce "e_tower", strcharinfo(0) +" 隊伍中的 "+ getpartyname( getcharid(1) ) +"，正在進入無盡之塔副本。",bc_map,"0x00ff99",FW_NORMAL,12;
		if (getarg(1)) {
			set etower_timer,gettimetick(2);
			set etower_partyid, getcharid(1);
			setquest 60200;
			setquest 60201;
		}
		//warp "1@tower",52,354;
		if (getarg(0) == 0) close;
		else end;
	}
}
