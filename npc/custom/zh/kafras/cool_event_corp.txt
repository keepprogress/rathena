//callfunc "F_CoolEventCorp","<menu>","<location>","<save map>",<x>,<y>,"<warp map>";
function	script	F_CoolEventCorp	{
	cutin "zonda_01",2;
	mes "[酷活動公司 員工]";
	mes "歡迎來到酷活動公司。";
	mes "我們的員工一直努力";
	mes "超越您的期待";
	mes "，提供優質服務。請問";
	mes "今天有什麼能為您效勞的嗎？";
	next;
	switch(select(getarg(0))) {
	case 1:
		mes "[酷活動公司 員工]";
		mes "您的重生點";
		mes "已透過";
		mes getarg(1)+".";
		mes "感謝您使用";
		mes "酷活動公司的服務~";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close2;
		break;
	case 2:
		mes "[酷活動公司 員工]";
		if(!callfunc("F_CanOpenStorage")){
			mes "很抱歉，您";
			mes "需要初學者的";
			mes "基本技能等級 6 才能";
			mes "使用倉庫服務。";
			close2;
			break;
		}
		if(Zeny < 40){
			mes "很抱歉，您";
			mes "足夠的 Zeny 來使用";
			mes "倉庫服務。我們的";
			mes "倉庫存取費用為 40 Zeny。";
			close2;
			break;
		}
		set Zeny, Zeny - 40;
		mes "讓我為您開啟";
		mes "個人倉庫，馬上就好。";
		mes "感謝您支持酷";
		mes "活動公司，透過使用我們的";
		mes "服務。祝您有美好的一天~";
		callfunc("F_CheckKafCode");	//check your storage password, if set
		close2;
		openstorage;
		break;
	case 3:
		mes "請選擇";
		mes "您的目的地。";
		next;
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE))
			set .@cost,4400;
		else
			set .@cost,2200;
		.@destination$ = getarg(5);
		if (select(.@destination$ +" -> "+ .@cost +" z:取消") == 1) {
			if ((countitem(7060) != 0) || (Zeny >= .@cost)) {
				if (countitem(7060) != 0) {
					delitem 7060,1; //Warp_Free_Ticket
				}
				else {
					set Zeny,Zeny-.@cost;
				}
				if (.@destination$ == "Veins") warp "veins",205,101;
				else if (.@destination$ == "Rachel") warp "rachel",115,125;
				cutin "",255;
				end;
			}
			else {
				mes "[酷活動公司 員工]";
				mes "很抱歉，您身上沒有";
				mes "足夠的 Zeny 支付傳送";
				mes "服務。傳送費用";
				mes "至 "+ .@destination$ +" 為 "+.@cost+" Zeny。";
			}
		}
		close2;
		break;
	case 4:
		mes "[酷活動公司 員工]";
		if(BaseClass != Job_Merchant){
			mes "很抱歉，";
			mes "手推車租借服務";
			if (checkre(0))
				mes "僅限商人職業可使用。";
			else {
				mes "僅限商人、";
				mes "鐵匠、白鐵匠、";
				mes "鍊金術師與創造者。";
			}
			close2;
			break;
		}
		if(getskilllv("MC_PUSHCART") == 0){
			mes "您必須在";
			mes "學會手推車使用技能後，才能租借手推車。";
			close2;
			break;
		}
		if(checkcart() == 1){
			mes "您已經裝備了";
			mes "一台手推車。";
			mes "很抱歉，我們無法";
			mes "同時為每位顧客租借";
			mes "超過一台手推車。";
			close2;
			break;
		}
		if(countitem(7061)){
			delitem 7061,1; //Cart_Free_Ticket
			setcart;
			close2;
			break;
		}
		mes "手推車租借";
		mes "費用為 800 Zeny。您想要";
		mes "想要租借手推車嗎？";
		next;
		if(select("租借手推車:取消") == 1){
			if(Zeny < 800){
				mes "[酷活動公司 員工]";
				mes "很抱歉，您";
				mes "您身上沒有足夠的";
				mes "足夠的 Zeny 來支付手推車";
				mes "租借費用 800 Zeny。";
			} else {
				set Zeny, Zeny - 800;
				setcart;
			}
		}
		close2;
		break;
	case 5:
		callfunc("F_SetKafCode","[Cool Event Corp. Staff]","Cool Event Corp.");
		//the function closes msg and removes cutins
	case 6:
	default:
		mes "[酷活動公司 員工]";
		mes "酷活動公司一直";
		mes "努力提供最好的";
		mes "服務給我們的顧客。";
		mes "請協助我們成為最好的，";
		mes "提供我們您寶貴的意見，";
		mes "以及誠實的回饋。感謝您。";
		close2;
	}
	cutin "", 255;
	end;
}

lighthalzen,94,248,3	script	Cool Event Corp Staff::CoolEventCorpStaffLighthalzen	4_M_ZONDAMAN,{
	callfunc "F_CoolEventCorp",
		"Save:Use Storage::Rent a Pushcart:Storage Password Service:Cancel",
		"in the town of Lighthalzen","lighthalzen",95,240;
	end;
}
