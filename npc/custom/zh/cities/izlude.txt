-	script	::Sailor_izlude	-1,{
	mes "[水手]";
	mes "嘿，大家！";
	mes "注意，注意！";
	mes "快來乘著風，";
	mes "搭上迷人的船！";
	mes "快來，快來！";
	next;
	switch(select("拜歐蘭島 -> 150 Zeny.:阿爾貝塔港口  -> 500 Zeny.:取消.")) {
	case 1:
		if (Zeny < 150) {
			mes "[水手]";
			mes "150 Zeny！";
			mes "只要 150 Zeny 就能搭乘！";
			close;
		}
		set Zeny, Zeny - 150;
		warp "izlu2dun",107,50;
		end;
	case 2:
		if (Zeny < 500) {
			mes "[水手]";
			mes "500 Zeny！";
			mes "只要 500 Zeny 就能搭乘！";
			close;
		}
		set Zeny, Zeny - 500;
		warp "alberta",188,169;
		end;
	case 3:
		close;
	}
}
