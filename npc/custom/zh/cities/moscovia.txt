-	script	::PR_Officer_moscovia	-1,{
	mes "[莫斯科公關員]";
	mes "莫斯科維亞！你知道莫斯科維亞嗎？";
	mes "那片延伸在無盡大海上的樂園……";
	mes "歡迎來到莫斯科維亞";
	mes "那裡充滿冒險與神秘。";
	next;
	switch(select("關於莫斯科維亞……:前往莫斯科維亞。:取消")) {
		case 1:
			mes "[莫斯科公關員]";
			mes "莫斯科維亞是一個美麗的王國，";
			mes "位於盧恩-";
			mes "米德迦爾特以北的島嶼上。";
			next;
			mes "[莫斯科公關員]";
			mes "我相信你一定會";
			mes "深深著迷";
			mes "於莫斯科維亞的美麗景色";
			mes "與華麗的宮殿。";
			next;
			mes "[莫斯科公關員]";
			mes "既然我們漫長的冬天已經";
			mes "過去了，";
			mes "我很高興現在能向你展示";
			mes "我們美麗的故鄉。";
			next;
			mes "[莫斯科公關員]";
			mes "如果你想造訪莫斯科維亞，";
			mes "現在就是好機會！";
			mes "我會協助你享受愉快的旅程";
			mes "前往莫斯科維亞！";
			close;

		case 2:
			mes "[莫斯科公關員]";
			mes "好，那我們現在就出發吧。";
			mes "你需要支付我 10,000 Zeny";
			mes "才能前往莫斯科維亞。";
			mes "但等你回來的時候，";
			mes "就不需要付費了。";
			next;
			mes "[莫斯科公關員]";
			mes "現在可以出發了嗎？";
			next;
			if (select("出發吧！:取消") == 2) {
				mes "[莫斯科公關員]";
				mes "如果你現在太忙，";
				mes "隨時想去都可以再跟我說。";
				mes "我隨時都準備好帶任何人前往莫斯科維亞。";
				close;
			}
			if (Zeny < 10000) {
				mes "[莫斯科公關員]";
				mes "很抱歉，你的";
				mes "Zeny 現在不夠";
				mes "你需要 10,000 Zeny";
				mes "才能前往莫斯科維亞";
				mes "謝謝。";
				close;
			} else {
				mes "[莫斯科公關員]";
				mes "好，我們現在出發了。";
				close2;
				set Zeny,Zeny - 10000;
				warp "moscovia",163,55;
				end;
			}
		case 3:
			mes "[莫斯科公關員]";
			mes "如果你現在太忙，";
			mes "隨時想去都可以再跟我說。";
			mes "我隨時都準備好帶任何人前往莫斯科維亞。";
			close;
	}
}
