		next;
		mes "[艾巴克辛]";
		mes "對於渴望冒險的人來說，有一則古老的故事，關於薩明虎這種邪惡生物，牠盤踞在我們的神殿遺跡中。";
		next;
		mes "[艾巴克辛]";
		mes "呼！哈！呼！哈！";
		mes "世上沒有任何人能抗拒";
		mes "這片美麗土地的呼喚。";
		mes "米德加爾人";
		mes "永遠受歡迎！";
		close;
	case 2:
		mes "[艾巴克辛]";
		mes "啊，您必須支付10,000 Zeny";
		mes "才能前往哎喲泰雅。";
		mes "如果您準備好了，我們可以馬上";
		mes "出發~";
		next;
		if (select("我準備好了，出發吧！:不要。") == 1) {
			if (Zeny > 9999) {
				mes "[艾巴克辛]";
				mes "呼！哈！呼！哈！";
				mes "出發吧！回到我的";
				mes "美麗的哎喲泰雅！";
				close2;
				set Zeny, Zeny-10000;
				warp "ayothaya",149,71;
				end;
			}
			mes "[艾巴克辛]";
			mes "很抱歉，您沒有足夠的10,000 Zeny可以前往哎喲泰雅。與等待您的體驗相比，這點費用根本不算什麼！";
			close;
		}
	case 3:
		mes "[艾巴克辛]";
		mes "你是否曾夢見";
		mes "一個充滿";
		mes "靈性寧靜的美麗之地？你";
		mes "一定是在作夢";
		mes "夢見哎喲泰雅了，我的朋友。";
		close;
	}
}
