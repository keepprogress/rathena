	else if (.@item_type == 1)

	else if (.@item_type == 2) {

		mes "等級高於80的玩家可以裝備的防具為:^ff0000"+ getitemname(.@item_id) +"^000000.";
	}
	else if (.@item_type == 3) {
		mes "你已選擇'量產型 "+ callfunc("F_GetArmorType",.@item_id) +"'.";
		mes "只有一件量產型 "+ callfunc("F_GetArmorType",.@item_id) +": ^ff0000"+ getitemname(.@item_id) +"^000000.";
	}
	else if (.@item_type == 4) {
		mes "你已選擇'普及型 "+ callfunc("F_GetArmorType",.@item_id) +"'.";
		mes "只有一件普及型 "+ callfunc("F_GetArmorType",.@item_id) +": ^ff0000"+ getitemname(.@item_id) +"^000000.";
	}

	if (!.@req_setting)
		mes "需要^0000ff2,000^000000 KVM點數，而 "+ .@name$ +"，你的點數為:^580080"+ kvm_point +"^000000.";
	else if (.@req_setting == 4)
		mes "需要^0000ff1,200^000000 KVM點數，而 "+ .@name$ +"，你的點數為:^580080"+ kvm_point +"^000000.";
	else
		mes "需要^0000ff"+ .@prices[.@req_setting] +"^000000 KVM點數，而 "+ .@name$ +"，你的點數為:^580080"+ kvm_point +"^000000.";

	mes "你確定要這件道具嗎？";
	next;
	switch(select("否，我不購買。:是，我要購買。")) {
	case 1:
		mes "[後勤]";
		mes "你已選擇'不購買'。";
		mes "購買道具時請留意，恕不退款。";
		break;
	case 2:
		if (kvm_point >= .@prices[.@req_setting]) {
			kvm_point = kvm_point -.@prices[.@req_setting];
			getitem .@item_id,1;
			mes "[後勤]";
