bat_room,133,178,5	script	弗拉維斯軍官#01a::Flavius Officer#01a	418,{
	if (getbattleflag("feature.bgqueue"))
		end;
	if (checkweight(1201,1) == 0) {
		mes "- 等一下！！ -";
		mes "- 你目前攜帶的 -";
		mes "- 道具太多了。 -";
		mes "- 請再試一次 -";
		mes "- 等你減輕一些負重後。 -";
		close;
	}
	mes "[紀堯姆軍官]";
	mes "讓那些臭克魯瓦人見識一下紀堯姆軍的實力吧！";
	next;
	switch(select("我想加入你們的軍隊！:結束對話")) {
	case 1:
		if ((BaseJob == Job_Novice) || (BaseJob == Job_Super_Novice)) {
			mes "[紀堯姆軍官]";
			mes "很高興你想加入我們，與克魯瓦軍團作戰，但很抱歉:我不能讓你這樣的新兵在殘酷的戰場上送死。";
		}
		else {
			if (BaseLevel < 80) {
				mes "[紀堯姆軍官]";
				mes "很高興你想加入我們，與克魯瓦軍團作戰，但很抱歉:我不能讓你這樣的新兵在殘酷的戰場上送死。";
			}
			else {
				set .@chk_urtime,checkquest(2070,PLAYTIME);
				if ((.@chk_urtime == 0)|| (.@chk_urtime == 1)) {
					mes "[紀堯姆軍官]";
					mes "你看起來才剛從戰場上回來。";
					mes "你還不能回去。去休息吧，敵人就交給我們！";
				}
				else {
					if (.@chk_urtime == 2)
						erasequest 2070;
					set .@mapcount2,getmapusers("bat_b01");
					if (.@mapcount2 > 0) {
						mes "[紀堯姆軍官]";
						mes "我收到報告，已有精銳部隊被派往戰場，正在與克魯瓦人作戰。";
						mes "你何不先等一下？我建議你在此之前磨利武器，準備好補給。";
					}
					else {
						if ($@FlaviusBG1 > 0) {
							mes "[紀堯姆軍官]";
							mes "精銳部隊已準備好隨時派往戰場。";
							mes "你何不先等一下？我建議你在此之前磨利武器，準備好補給。";
						}
						else {
							mes "[紀堯姆軍官]";
							mes "你看起來已經準備好作戰了！";
							mes "去讓克魯瓦人見識真正的恐懼吧！";
							mes "今天，我們的勝利吶喊將響徹整個戰場！";
							close2;
							warp "bat_room",85,223;
							end;
						}
					}
				}
			}
		}
		break;
	case 2:
		mes "[紀堯姆軍官]";
		mes "今天，我們必將獲勝！";
		break;
	}
	close;
}
