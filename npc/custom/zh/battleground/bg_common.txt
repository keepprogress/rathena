		mes "[艾倫德克]";
		if (.@j == 3) {
			mes "你取消了兌換。";
			break;
		}
		set .@cost, ((.@j==1)?7828:7829);
		if (countitem(.@cost) >= .@Items[.@i+1]) {
			mes "感謝你的兌換。";
			delitem .@cost, .@Items[.@i+1];
			getitem .@Items[.@i],1;
		}
		else mes "You do not have enough "+getitemname(.@cost)+"s.";
		break;
	}
	close;
}

// Badge Reward Function
//============================================================
function	script	F_BG_Badge	{
	.@team$ = getarg(1);
	.@bg_type$ = getarg(2);

	if (.@bg_type$ == "Tierra") {
		set .@badge,7828; //BF_Badge1
		set .@amount_win,3;
		set .@amount_lose,1;
	} else if (.@bg_type$ == "Flavius") {
		set .@badge,7829; //BF_Badge2
		set .@amount_win,9;
		set .@amount_lose,3;
	} else
		return;

	if (getarg(0) == 1) {
		set .@amount, .@amount_win;
		if (.@team$ == "Guillaume") {
			mes "[艾克索·羅斯]";
			mes "吉約姆祝福你！";
		} else if (.@team$ == "Croix") {
			mes "[斯旺德里]";
			mes "克魯瓦祝福你！";
		}
		mes "讓我們享受這光榮的勝利吧！";
		mes strcharinfo(0)+"，這是勝利的象徵。";
	} else {
		set .@amount, .@amount_lose;
		if (.@team$ == "Guillaume") {
			mes "[艾克索·羅斯]";
			mes "你雖然輸了，但你對這場戰鬥全力以赴。";
			mes "這是吉約姆·馬洛洛為了表彰你的努力所給予的獎勵！";
			mes "就把這次的失敗當作教訓吧，下次你一定會贏的。";
		} else if (.@team$ == "Croix") {
			mes "[斯旺德里]";
			mes "哦， "+strcharinfo(0)+" 不要難過。";
			mes "雖然我們沒有贏，但我們已經盡力了。";
			mes "這是克魯瓦王子賜予的皇家禮物，請別忘了這場戰鬥。下一場我們一定會贏。";
		}
	}
	close2;

	// If the VIP system is enabled, VIP players receive an extra 2 badges.
	if (VIP_SCRIPT && vip_status(VIP_STATUS_ACTIVE))
		set .@amount, .@amount+2;

	set .@medal_gap, 500 - countitem(.@badge);
	if (.@medal_gap >= .@amount)
		getitem .@badge,.@amount;
	else
		getitem .@badge,.@medal_gap;
	return;
}
