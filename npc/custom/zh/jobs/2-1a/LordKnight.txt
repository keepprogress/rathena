valkyrie,44,39,5	script	騎士領主#Valkyrie::Lord Knight#Valkyrie	56,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[騎士領主]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[騎士領主]";
		mes "我們騎士肩負著一份";
		mes "重大的責任……";
		mes "守護與效忠。";
		next;
		mes "[騎士領主]";
		mes "即使付出";
		mes "自己的性命，";
		mes "我們也必須守護";
		mes "夥伴們的安危。";
		close;
	}
	else if (ADVJOB == Job_Lord_Knight && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[騎士領主]";
		mes "你的時機已到！";
		mes "世界依然需要你。";
		mes "請以";
		mes "嶄新的姿態，繼續作為英雄而活。";
		next;
		mes "[騎士領主]";
		mes "你想要";
		mes "想成為騎士領主嗎？";
		next;
		if (select("不。:是的。") == 1) {
			mes "[騎士領主]";
			mes "準備好的時候，";
			mes "隨時歡迎再回來。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[騎士領主]";
			mes "你仍然可以學習更多技能。請先使用";
			mes "剩餘的技能點數，再回來找我。";
			close;
		}
		jobchange Job_Lord_Knight;
		set ADVJOB,0;
		mes "[騎士領主]";
		mes "恭喜！";
		mes "身為騎士領主，";
		mes "我希望你能站";
		mes "在戰場的最前線，";
		mes "帶領你的夥伴贏得勝利！";
		close;
	}
	mes "[騎士領主]";
	mes "歡迎";
	mes "前往瓦爾哈拉，";
	mes "榮譽大廳。";
	next;
	mes "[騎士領主]";
	mes "請您";
	mes "放輕鬆，";
	mes "在此期間請隨意休息。";
	mes "向勇士致敬！";
	close;
}
