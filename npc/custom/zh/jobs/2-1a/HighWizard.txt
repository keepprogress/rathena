valkyrie,44,47,5	script	大巫師#Valkyrie::High Wizard#Valkyrie	735,{
	if (ADVJOB == 0 ||  Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[大巫師]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[大巫師]";
		mes "我們大巫師";
		mes "肩負著";
		mes "運用毀滅性魔法的";
		mes "責任，必須用於正當的目的。";
		next;
		mes "[大巫師]";
		mes "需要畢生的修練";
		mes "，才能成為";
		mes "大巫師。你能想像嗎？";
		mes "若我們的力量";
		mes "落入錯誤的人手中會怎樣？！";
		close;
	}
	else if (ADVJOB == Job_High_Wizard && Class == Job_Mage_High && JobLevel > 39) {
		mes "[大巫師]";
		mes "時候到了。";
		mes "米德加爾";
		mes "需要能夠";
		mes "駕馭最強大魔法的人……";
		next;
		mes "[大巫師]";
		mes "您想要";
		mes "成為大巫師嗎？";
		next;
		if (select("不。:是的。") == 1) {
			mes "[大巫師]";
			mes "準備好的時候，";
			mes "隨時歡迎再回來。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[大巫師]";
			mes "你仍然可以學習更多技能。請先使用";
			mes "剩餘的技能點數，再回來找我。";
			close;
		}
		jobchange Job_high_Wizard;
		set ADVJOB,0;
		mes "[大巫師]";
		mes "恭喜。";
		mes "身為大巫師，";
		mes "我希望你能善用";
		mes "你的力量，";
		mes "為受壓迫者帶來和平。";
		close;
	}
	mes "[大巫師]";
	mes "歡迎";
	mes "前往瓦爾哈拉，";
	mes "榮譽大廳。";
	next;
	mes "[大巫師]";
	mes "請您";
	mes "放輕鬆，";
	mes "在此期間請隨意休息。";
	mes "向勇士致敬！";
	close;
}
