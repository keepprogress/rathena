valkyrie,44,55,5	script	狙擊手#Valkyrie::Sniper#Valkyrie	727,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[狙擊手]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[狙擊手]";
		mes "一箭出手。";
		mes "一擊必殺。";
		mes "這並不難，";
		mes "只要你培養出";
		mes "那種風格的";
		mes "戰鬥眼光。";
		close;
	}
	else if (ADVJOB == Job_Sniper && Class == Job_Archer_High && JobLevel > 39) {
		mes "[狙擊手]";
		mes "世界正";
		mes "需要強大的弓箭手";
		mes "像你這樣的人。你準備好迎接";
		mes "這份重大的責任了嗎？";
		next;
		mes "[狙擊手]";
		mes "你願意";
		mes "踏出下一步，";
		mes "成為狙擊手嗎？";
		next;
		if (select("不。:是的。") == 1) {
			mes "[狙擊手]";
			mes "準備好的時候，";
			mes "隨時歡迎再回來。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[狙擊手]";
			mes "你仍然可以學習更多技能。請先使用";
			mes "剩餘的技能點數，再回來找我。";
			close;
		}
		jobchange Job_Sniper;
		set ADVJOB,0;
		mes "[狙擊手]";
		mes "恭喜！";
		mes "身為狙擊手，我希望";
		mes "邪惡的爪牙";
		mes "將永遠無處可躲";
		mes "只要還在";
		mes "你的視野之中！";
		close;
	}
	mes "[狙擊手]";
	mes "歡迎";
	mes "前往瓦爾哈拉，";
	mes "榮譽大廳。";
	next;
	mes "[狙擊手]";
	mes "請您";
	mes "放輕鬆，";
	mes "在此期間請隨意休息。";
	mes "向勇士致敬！";
	close;
}
