valkyrie,44,42,5	script	高僧#Valkyrie::High Priest#Valkyrie	60,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[高僧]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[高僧]";
		mes "藉由神聖的力量，";
		mes "願我們";
		mes "尋得和平與力量";
		mes "與庇護。請";
		mes "拯救我們脫離邪惡勢力……";
		close;
	}
	else if (ADVJOB == Job_High_Priest && Class == Job_Acolyte_High && JobLevel > 39) {
		mes "[高僧]";
		mes "我們的世界正";
		mes "需要擁有";
		mes "才能與信念的人。";
		mes "請繼續你的";
		mes "善行，成為";
		mes "更偉大的神聖英雄……";
		next;
		mes "[高僧]";
		mes "你想要";
		mes "想成為高僧嗎？";
		next;
		if (select("不。:是的。") == 1) {
			mes "[高僧]";
			mes "準備好的時候，";
			mes "隨時歡迎再回來。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[高僧]";
			mes "你仍然可以學習更多技能。請先使用";
			mes "剩餘的技能點數，再回來找我。";
			close;
		}
		jobchange Job_High_Priest;
		set ADVJOB,0;
		mes "[高僧]";
		mes "恭喜。";
		mes "身為高僧，";
		mes "我希望你能引導";
		mes "他人走上";
		mes "神聖之路……";
		close;
	}
	mes "[高僧]";
	mes "歡迎";
	mes "前往瓦爾哈拉，";
	mes "榮譽大廳。";
	next;
	mes "[高僧]";
	mes "請您";
	mes "放輕鬆，";
	mes "在此期間請隨意休息。";
	mes "向勇士致敬！";
	close;
}
