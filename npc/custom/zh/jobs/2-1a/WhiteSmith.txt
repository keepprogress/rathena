valkyrie,44,50,5	script	鍛造大師#Valkyrie::MasterSmith#Valkyrie	731,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[鍛造大師]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[鍛造大師]";
		mes "啊……";
		mes "工藝的";
		mes "極致。";
		mes "那是";
		mes "鍛造大師的作品。";
		next;
		mes "[鍛造大師]";
		mes "當你成為";
		mes "最頂尖中的頂尖，";
		mes "你將開始懷著";
		mes "發掘";
		mes "更佳新方法的";
		mes "crafting...";
		close;
	}
	else if (ADVJOB == Job_WhiteSmith && Class == Job_Merchant_High && JobLevel > 39) {
		mes "[鍛造大師]";
		mes "時候到了！";
		mes "我們的世界需要勇敢、";
		mes "勤奮的冒險者，";
		mes "像你這樣的……";
		next;
		mes "[鍛造大師]";
		mes "您想要";
		mes "成為鍛造大師嗎？";
		next;
		if (select("不。:是的。") == 1) {
			mes "[鍛造大師]";
			mes "準備好的時候，";
			mes "隨時歡迎再回來。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[鍛造大師]";
			mes "你仍然可以學習更多技能。請先使用";
			mes "剩餘的技能點數，再回來找我。";
			close;
		}
		jobchange Job_Whitesmith;
		set ADVJOB,0;
		mes "[鍛造大師]";
		mes "恭喜！";
		mes "身為鍛造大師，";
		mes "我希望你能鍛造出";
		mes "通往更光明的";
		mes "米德加爾未來的道路。";
		close;
	}
	mes "[鍛造大師]";
	mes "歡迎";
	mes "前往瓦爾哈拉，";
	mes "榮譽大廳。";
	next;
	mes "[鍛造大師]";
	mes "請您";
	mes "放輕鬆，";
	mes "在此期間請隨意休息。";
	mes "向勇士致敬！";
	close;
}
