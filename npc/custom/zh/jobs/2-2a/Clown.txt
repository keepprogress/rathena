valkyrie,53,54,3	script	吟遊詩人#Valkyrie::Minstrel#Valkyrie	741,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[吟遊詩人]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[吟遊詩人]";
		mes "想不想";
		mes "和我一起唱首歌？";
		mes "啦啦啦啦~";
		close;
	}
	else if (ADVJOB == Job_Clown && Class == Job_Archer_High && JobLevel > 39) {
		mes "[吟遊詩人]";
		mes "憂鬱的世界";
		mes "凡人們所需要的，";
		mes "是更多歡快的歌聲。";
		mes "你願意把它帶給他們嗎，";
		mes "並在";
		mes "對抗邪惡的戰役中扭轉乾坤？";
		next;
		mes "[吟遊詩人]";
		mes "你願意這麼做，";
		mes "為了米德加德……";
		mes "成為吟遊詩人嗎？";
		next;
		if (select("不。:是的。") == 1) {
			mes "[吟遊詩人]";
			mes "準備好的時候，";
			mes "隨時歡迎再回來。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[吟遊詩人]";
			mes "你仍然可以學習更多技能。請先使用";
			mes "剩餘的技能點數，再回來找我。";
			close;
		}
		jobchange Job_Clown;
		set ADVJOB,0;
		mes "[吟遊詩人]";
		mes "恭喜！";
		mes "身為吟遊詩人，你的";
		mes "歌聲將帶來";
		mes "希望給你的夥伴，並且";
		mes "絕望給你的敵人。";
		close;
	}
	mes "[吟遊詩人]";
	mes "歡迎";
	mes "前往瓦爾哈拉，";
	mes "榮譽大廳。";
	next;
	mes "[吟遊詩人]";
	mes "請您";
	mes "放輕鬆，";
	mes "在此期間請隨意休息。";
	mes "向勇士致敬！";
	close;
}
