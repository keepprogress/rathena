valkyrie,53,39,3	script	聖騎士#Valkyrie::Paladin#Valkyrie	752,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[聖騎士]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[聖騎士]";
		mes "別忘了";
		mes "聖戰";
		mes "即將來臨！";
		mes "我們必須做好準備！";
		mes "願正義之光";
		mes "永遠照亮我們的道路！";
		close;
	}
	else if (ADVJOB == Job_Paladin && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[聖騎士]";
		mes "聖戰將會";
		mes "在我們還來不及";
		mes "察覺之前降臨。比以往任何時候，";
		mes "我們都更需要強壯的男人";
		mes "和女人，為了";
		mes "正義與正確而戰。";
		next;
		mes "[聖騎士]";
		mes "你願意站在";
		mes "正義的一方";
		mes "成為聖騎士嗎？";
		next;
		if (select("不。:是的。") == 1) {
			mes "[聖騎士]";
			mes "準備好的時候，";
			mes "隨時歡迎再回來。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[聖騎士]";
			mes "你仍然可以學習更多技能。請先使用";
			mes "剩餘的技能點數，再回來找我。";
			close;
		}
		jobchange Job_Paladin;
		set ADVJOB,0;
		mes "[聖騎士]";
		mes "恭喜。";
		mes "身為聖騎士，我希望";
		mes "你會保護那些";
		mes "比你弱小的人，並帶來";
		mes "勝利給我們，在即將到來的";
		mes "善惡之戰中。";
		close;
	}
	mes "[聖騎士]";
	mes "歡迎";
	mes "前往瓦爾哈拉，";
	mes "榮譽大廳。";
	next;
	mes "[聖騎士]";
	mes "請您";
	mes "放輕鬆，";
	mes "在此期間請隨意休息。";
	mes "向勇士致敬！";
	close;
}
