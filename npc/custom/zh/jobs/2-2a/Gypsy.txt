valkyrie,53,56,3	script	吉普賽人#Valkyrie::Gypsy#Valkyrie	101,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[吉普賽人]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[吉普賽人]";
		mes "向左移，";
		mes "向右移~！";
		mes "然後踏步……！";
		mes "舞蹈可以是";
		mes "不只是一種嗜好。";
		mes "對我來說，那是一種生活方式~";
		close;
	}
	else if (ADVJOB == Job_Gypsy && Class == Job_Archer_High && JobLevel > 39) {
		mes "[吉普賽人]";
		mes "米德加德之地";
		mes "需要有才華的女性，";
		mes "微妙地改變平衡，";
		mes "在善與";
		mes "惡的戰鬥中。";
		next;
		mes "[吉普賽人]";
		mes "你準備好";
		mes "擔任這個角色，";
		mes "成為吉普賽人嗎？";
		next;
		if (select("不。:是的。") == 1) {
			mes "[吉普賽人]";
			mes "準備好的時候，";
			mes "隨時歡迎再回來。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[吉普賽人]";
			mes "你仍然可以學習更多技能。請先使用";
			mes "剩餘的技能點數，再回來找我。";
			close;
		}
		jobchange Job_Gypsy;
		set ADVJOB,0;
		mes "[吉普賽人]";
		mes "恭喜！";
		mes "身為吉普賽人，我知道";
		mes "你的表演";
		mes "能動搖所有人的心，";
		mes "那些觀看的人……";
		close;
	}
	mes "[吉普賽人]";
	mes "歡迎";
	mes "前往瓦爾哈拉，";
	mes "榮譽大廳。";
	next;
	mes "[吉普賽人]";
	mes "請您";
	mes "放輕鬆，";
	mes "在此期間請隨意休息。";
	mes "向勇士致敬！";
	close;
}
