valkyrie,53,58,3	script	追獵者#Valkyrie::Stalker#Valkyrie	747,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[追獵者]";
			mes "恭喜。";
			mes "向勇士致敬！";
			close;
		}
		mes "[追獵者]";
		mes "嘿嘿……";
		mes "很辛苦，";
		mes "當英雄";
		mes "又要陰險行事，";
		mes "讓人不信任，";
		mes "sneaky...";
		next;
		mes "[追獵者]";
		mes "但當";
		mes "情況艱難時，";
		mes "我的夥伴知道";
		mes "可以依靠我。";
		mes "我需要他們，";
		mes "他們也需要我。";
		close;
	}
	else if (ADVJOB == Job_Stalker && Class == Job_Thief_High && JobLevel > 39) {
		mes "[追獵者]";
		mes "這個世界需要";
		mes "更多英雄，他們";
		mes "願意走在";
		mes "秩序與無法之間。";
		next;
		mes "[追獵者]";
		mes "你準備好";
		mes "想加入";
		mes "最狡猾的";
		mes "戰士之列嗎？你準備好";
		mes "成為追獵者了嗎？";
		next;
		if (select("不。:是的。") == 1) {
			mes "[追獵者]";
			mes "準備好的時候，";
			mes "隨時歡迎再回來。";
			mes "向勇士致敬！";
			close;
		}
		if (SkillPoint) {
			mes "[追獵者]";
			mes "你仍然可以學習更多技能。請先使用";
			mes "剩餘的技能點數，再回來找我。";
			close;
		}
		jobchange Job_Stalker;
		set ADVJOB,0;
		mes "[追獵者]";
		mes "恭喜！";
		mes "身為追獵者，我希望";
		mes "你能刺向正確的人，";
		mes "從背後下手。驅逐";
		mes "邪惡，用它們自己的";
		mes "卑鄙手段！";
		close;
	}
	mes "[追獵者]";
	mes "歡迎";
	mes "前往瓦爾哈拉，";
	mes "榮譽大廳。";
	next;
	mes "[追獵者]";
	mes "請您";
	mes "放輕鬆，";
	mes "在此期間請隨意休息。";
	mes "向勇士致敬！";
	close;
}
