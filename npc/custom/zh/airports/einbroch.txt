airport,143,43,5	script	機場職員#airport1a::airport1	90,{
	mes "[機場職員]";
	mes "歡迎來到";
	mes "艾音布羅克機場，";
	mes "我們提供直飛";
	mes "航班前往";
	mes "朱諾、里希塔樂鎮與毀葛。";
	next;
	if (select("登上飛空艇:取消") == 1) {
		mes "[機場職員]";
		mes "飛空艇登機費";
		mes "為 1,200 Zeny，不過如果您";
		mes "持有飛空艇免費票，";
		mes "就能免除費用。要";
		mes "登上飛空艇嗎？";
		next;
		if (select("是:否") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airport",148,51;
				end;
			}
			mes "[機場職員]";
			mes "很抱歉，您";
			mes "持有飛空艇";
			mes "免費票，但您沒有";
			mes "足夠的 Zeny 支付";
			mes "登機費。記住，登機費";
			mes "為 1,200 Zeny。";
			close;
		}
	}
	mes "[機場職員]";
	mes "謝謝您，";
	mes "祝您有愉快的一天。";
	close;
}
