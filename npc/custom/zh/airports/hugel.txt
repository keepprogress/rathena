	end;

	mes "你要使用這項服務嗎？";
	next;
	if (select("是:否") == 1) {
		if (countitem(7311) > 0) {
			delitem 7311,1; //Free_Flying_Ship_Ticket
			warp "airplane",244,58;
			end;
		}
		if (Zeny >= 1200) {
			set Zeny, Zeny - 1200;
			warp "airplane",244,58;
			end;
		}
		mes "很抱歉，你的金錢不足。";
		mes "請記住，使用此服務需支付 1,200 Zeny。";
		close;
	}
