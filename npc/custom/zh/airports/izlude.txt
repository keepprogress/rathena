-	script	::Airship_Staff_izlude	-1,{
	mes "[飛行船職員]";
	mes "歡迎！您想要";
	mes "登上";
	mes "停靠朱諾與拉赫的";
	mes "飛空艇嗎？";
	next;
	if (select("登上前往朱諾/拉赫的飛空艇。:取消。") == 1) {
		mes "[飛行船職員]";
		mes "登機費為";
		mes "1,200 Zeny。不過，";
		mes "如果您使用";
		mes "飛空艇免費票，此費用就免除了。現在，";
		mes "您還要登機嗎？";
		next;
		if (select("是:否") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airplane_01",244,58;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airplane_01",244,58;
				end;
			}
			mes "[飛行船職員]";
			mes "很抱歉，您";
			mes "身上沒有 1,200 Zeny";
			mes "可以支付登機費。";
			close;
		}
	}
	mes "[飛行船職員]";
	mes "謝謝您，";
	mes "歡迎再次光臨。";
	mes "祝您有美好的一天~";
	close;
}
