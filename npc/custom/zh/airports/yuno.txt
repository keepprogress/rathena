y_airport,143,43,5	script	機場職員#y_air1a::y_airport1	90,{
	mes "[機場職員]";
	mes "歡迎來到朱諾機場，我們提供前往艾音布羅克、里希塔樂鎮與毀葛的國內航班，";
	mes "以及前往依斯魯得與拉赫的國際航班。";
	mes "需要什麼服務呢？";
	next;
	if (select("登上飛空艇。:取消。") == 1) {
		mes "[機場職員]";
		mes "所有航班的登機費";
		mes "皆為 1,200 Zeny。如果您";
		mes "使用飛空艇免費票，";
		mes "就能免除登機費。那麼您要出發嗎？";
		next;
		if (select("是:否") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "y_airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "y_airport",148,51;
				end;
			}
			mes "[機場職員]";
			mes "很抱歉，您";
			mes "身上沒有 1,200 Zeny";
			mes "可以支付登機費。";
			close;
		}
	}
	mes "[機場職員]";
	mes "謝謝您，";
	mes "祝您有愉快的一天。";
	close;
}
