// Guild Flags (Outside Castle)
//============================================================
-	script	Bamboo Grove Hill#1::OutsideFlagsPA5	722,{

	set .@GID, GetCastleData("payg_cas05",1);

	if (.@GID == 0) {
		mes "[ 神聖符文中土王國敕令 ]";
		mes " ";
		mes "1. 遵守神聖符文中土王國的律令， ";
		mes "我們宣告";
		mes "此城堡並無正式的主人。";
		mes " ";
		mes "2. 能夠 ";
		mes "克服所有考驗";
		mes "並摧毀華麗金屬，";
		mes "國王將把";
		mes "此城堡的所有權賜予該人。";
		close;
	}
	if (getcharid(2) == .@GID) {
		mes "[ 迴響之聲 ]";
		mes "勇者們……";
		mes "您想返回您尊貴的所在嗎？";
		next;
		if(select("返回公會城堡:離開") == 1) {
			close2;
			if (getcharid(2) == GetCastleData("payg_cas05",1)) {
					warp "payg_cas05",243,27;
				end;
			}
			end;
		}
		close;
	}
	mes "[ 神聖符文中土王國敕令 ]";
	mes " ";
	mes "1. 遵守神聖符文中土王國的律令， ";
	mes "我們認定此地";
	mes "私人持有：^ff0000"+GetGuildName(.@GID)+"^000000公會所有。";
	mes " ";
	mes "2. 公會會長為 ^ff0000"+GetGuildName(.@GID)+"^000000 公會，";
	mes "^ff0000"+GetGuildMaster(.@GID)+"^000000";
	mes "若有人對此有異議，";
	mes "請以手中的鋼刃證明你的實力與榮譽。";
	close;

OnRecvCastlePy05:
	FlagEmblem GetCastleData("payg_cas05",1);
	end;
}
