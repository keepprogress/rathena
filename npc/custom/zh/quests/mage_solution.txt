pay_arche,122,100,0	script	多爾修伊::Dollshoi	88,{
	mes "[魔法師公會]";
	mes "你要溶液？嗯，給我 50 Zeny，還要至少 1 個空試管。";
	next;
	if (select("好，成交。:算了。") == 1) {
		mes "[魔法師公會]";
		if (Zeny < 50) {
			mes "喂！你的錢不夠付我 50 Zeny 的費用。";
			close;
		}
		if (countitem(1092) == 0) {
			mes "溶液沒有瓶子可裝可不行！拿個空試管來給我。";
			close;
		}
		delitem 1092,1; //Empty_Cylinder
		set Zeny, Zeny-50;
		getitem 1089,1; //Payon_Potion
	}
	close;
}
