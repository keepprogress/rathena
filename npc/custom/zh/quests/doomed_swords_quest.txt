		next;
	}
	if (checkweight(getarg(0),1) == 0) {
		mes "[休伊]";
		mes "等等，你身上帶著";
		mes "身上東西太多了。";
		mes "你最好先把一些東西";

	}

	mes "[休伊]";
	if (.@num == 0) {
		mes "那、那是魔短劍 闇噬！";

		mes "五芒星陣的技術之後，";

		mes "真不敢相信！";
		mes "你……你被";

		mes "小心，別讓";
		mes "它的力量把你吞噬。";
		mes "只為正當的目的使用它……";

	else if (.@num == 2) {
		mes "那、那是……！";
		mes "死神之劍已經";

	mes "嗯，看來我的";

	mes "我的哥哥們應該會感到驕傲。";
	next;
	mes "[休伊]";
	mes "說到這個，";
	mes "請代我向莫夫和";
	mes "沃克問好。如果你";
	mes "剛好見到他們。那麼";
	mes "就請你多保重，使用那把劍時也要小心。";
