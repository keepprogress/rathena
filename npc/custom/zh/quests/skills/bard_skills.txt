function	script	F_BardSkillYhelle	{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
		mes "^3355FF站住！";
		mes "你身上的物品太多，背包空間不足，無法領取任何獎勵。請先騰出背包空間後再回來挑戰。";
		close;
	}
	emotion ET_HUK, playerattached();
	emotion ET_HUK;
	mes "[母雞伊海勒]";
	mes "咕咕咕咕？";
	mes "咯咯咯！";
	next;
	if (rand(1,3) == 2) {
		disablenpc "Yhelle#bard_chick"+getarg(0);
		donpcevent "Yhelle#bard_chick"+getarg(1)+"::OnEnable";
		mes "^3355FF感覺到你的";
		mes "氣息，母雞";
		mes "迅速逃走了。^000000";
		if (qskill_bard > 1 && qskill_bard < 7) {
			mes "^3355FF你找到了^3333331 顆蛋^3355FF，在";
			mes "伊海勒母雞";
			mes "棲息的地方。^000000";
			set qskill_bard,qskill_bard+1;
			getitem 574,1; //Egg
		}
	}
	else {
		disablenpc "Yhelle#bard_chick"+getarg(0);
		donpcevent "Yhelle#bard_chick"+getarg(1)+"::OnEnable";
		mes "^3355FF感覺到你的";
		mes "氣息，母雞";
		mes "迅速逃走了。^000000";
	}
	return;
}
