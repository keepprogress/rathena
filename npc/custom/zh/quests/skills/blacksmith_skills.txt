			mes "大師鐵匠，對吧？";
		else
			mes "鐵匠，對吧？";
		mes "噗，這是反問句啦。";
		next;
		mes "[古德曼]";
		mes "好消息。我負責";
		mes "教授這個技能，由";
		mes "鐵匠與商人";
		mes "公會共同研發，名為 ^FF0000貪婪^000000。要不要學，";
		mes "全由你決定，";
		mes "不過我建議你學。";
		next;
		mes "[古德曼]";
		mes "如果你想，我現在就能開始";
		mes "教你貪婪";
		mes "技能。那麼";
		mes "你怎麼說，小子？";
		next;
		if (select("好啊！:之後再說吧~") == 1) {
			mes "[古德曼]";
			mes "哈！明智的選擇。那麼，";
			mes "貪婪技能會自動";
			mes "撿取施法者周圍";
			mes "5*5 範圍內的所有物品。";
			mes "對於撿拾怪物掉落的一大堆物品非常方便。";
			next;
			mes "[古德曼]";
			mes "首先，我得";
			mes "測試你，看看你有沒有資格";
			mes "學習這個技能。去把你的背包裝滿任何物品，直到";
			mes "滿到無法再放入一件";
			mes "500 重量的物品為止。";
			next;
			mes "[古德曼]";
			mes "完成這項";
			mes "任務後，回來";
			mes "找我說話。好了，";
			mes "那麼，祝你好運~";
			BLACK_SK2 = 1;
			close;
		}
		mes "[古德曼]";
		mes "我知道了。忙碌是好事，";
		mes "雖然你還有其他事要忙，";
		mes "但記住，時間是 ";
		mes "一種商品，你應該";
		mes "總是把它花在";
		mes "最有價值的事情上。明白嗎？";
		close;
	}
	if (BLACK_SK2 == 1) {
		if (MaxWeight - Weight >= 5000) {
			mes "[古德曼]";
			mes "嗯。你回來得";
			mes "太早了。而且你沒有";
			mes "完成我";
			mes "交代給你的目標。仔細聽好。";
			next;
			mes "[古德曼]";
			mes "去把你的背包";
			mes "裝滿任何物品，直到它";
			mes "滿到放不進";
			mes "一件 500 重量的物品為止。";
			mes "完成這項任務後，";
			mes "回來向我報告。";
			close;
		}
		mes "[古德曼]";
		mes "這麼快就回來啦？";
		mes "好吧，讓我查一下";
		mes "你身上帶著的東西。";
		mes "我看看，背包、背包……";
		next;
		mes "[古德曼]";
		mes "做得好:你通過了。";
		mes "你現在有沒有意識到，";
		mes "在";
		mes "收集這些物品上花費的時間和經驗，";
		mes "比它們的 Zeny 價格更有價值？";
		next;
		mes "[古德曼]";
		if (Upper == 1)
			mes "我相信大師鐵匠";
		else
			mes "我相信鐵匠";
		mes "首先應該是";
		mes "工匠。富有與聲名";
		mes "只是其中的副產品而已。";
		if (Upper == 1)
			mes "尊貴的大師鐵匠們明白，價值存在於工藝之中……";
		else
			mes "尊貴的鐵匠們明白，價值存在於工藝之中……";
		next;
		if (Upper == 1) {
			mes "[古德曼]";
			mes "現在，如果你不會太";
			mes "在意物品";
			mes "以 Zeny 計算的價值，你就";
			mes "能更輕易地";
			mes "放手。看我示範。";
			mes "看到了嗎？現在你也會了。";
			next;
		}
		mes "[古德曼]";
		mes "你學會的技能";
		mes "叫做貪婪，但我希望";
		mes "你使用它的方式";
		mes "要跟它的名字不一樣。";
		mes "別忘了這點，也希望";
		mes "你能打造出真正的傑作。";
		skill "BS_GREED",1,SKILL_PERM;
		if (Upper == 1)
			BLACK_SK2 = 4;	// note: official script is using another variable when a transcendent character is doing the quest for the first time
		else
			BLACK_SK2 = 2;
		close;
	}
	mes "[古德曼]";
	mes "你最近都怎麼使用";
	mes "貪婪技能呢？希望";
	mes "它在你";
	mes "替夥伴們打造物品，";
	mes "讓大家都能得到";
	mes "更高品質的成品。";
	if (getskilllv("BS_GREED") == 0)	// debug
		skill "BS_GREED",1,SKILL_PERM;
	close;
}
