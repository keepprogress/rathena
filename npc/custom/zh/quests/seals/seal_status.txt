prontera,113,294,3	script	路標#god::Sign Post#god	837,{
	mes "======== 神之印記狀態 ========";
	setarray .@seal$[1], "Sleipnir","Megingjard","Brisingamen","Mjolnir";
	for(set .@i,1; .@i<=4; set .@i,.@i+1) {
		set .@val, getd("$God"+.@i);
		if (.@val == 0)
			set .@status$,"Unseen";
		else if (.@val < $@god_check1)
			set .@status$,"Active";
		else if (.@val < $@god_check2)
			set .@status$,"Appeared";
		else
			set .@status$,"Released";
		mes .@seal$[.@i]+" 印記: "+.@status$;
	}
	mes " ";
	mes "======= 你的印記狀態 ========";
	if (god_sl_1 < 51)
		mes "^ff0000史雷普尼爾之靴印記: 未完成^000000";
	else
		mes "^00ff00史雷普尼爾之靴印記: 已完成^000000";
	if (god_eremes < 28)
		mes "^ff0000雷神腰帶印記: 未完成^000000";
	else
		mes "^00ff00雷神腰帶印記: 已完成^000000";
	if (god_brising < 50)
		mes "^ff0000布里辛嘉曼項鍊印記: 未完成^000000";
	else
		mes "^00ff00布里辛嘉曼項鍊印記: 已完成^000000";
	if (god_mjo_0 < 11)
		mes "^ff0000雷神之錘印記: 未完成^000000";
	else
		mes "^00ff00雷神之錘印記: 已完成^000000";
	close;
}
