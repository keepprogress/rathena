		mes "自視甚高。^000000";

// Number Match Game :: gef_yagu
//============================================================
geffen,122,77,4	script	妮雅#yagu::Nia#yagu	700,{
	mes "[妮雅]";
	mes "你好，我是妮雅，第五位";
	mes "馬旺的弟子，";
	mes "遊戲大師。好了，";
	mes "你想不想來玩一場";
	mes "數字配對遊戲？";
	next;
	switch(select("數字配對遊戲？:練習數字配對:開始數字配對:最高紀錄:取消")) {
	case 1:
		mes "[妮雅]";
		mes "啊，嗯，這個";
		mes "數字配對遊戲的目標，就是";
		mes "猜中我心裡想的";
		mes "數字。當然，有幾個";
		mes "規則限制了數字";
		mes "我可以選的範圍。";
		next;
		mes "[妮雅]";
		mes "我想出的所有數字";
		mes "都必須是三位數，而且我不能";
		mes "在任何一位使用 0，";
		mes "也不能使用相同的";
		mes "數字在任何一位上";
		mes "重複出現。";
		next;
		mes "[妮雅]";
		mes "舉例來說，我不能";
		mes "使用 103、112、";
		mes "252 或 701。這些數字";
		mes "不是含有 0，就是";
		mes "在數位上";
		mes "重複使用同一個數字。懂了嗎？";
		next;
		mes "[妮雅]";
		mes "現在，你只有 5 次機會猜，";
		mes "每猜一次，我會告訴你";
		mes "你猜中了幾個數字，";
		mes "以及其中有幾個數字猜對了";
		mes "正確的位置。";
		next;
		mes "[妮雅]";
		mes "舉例來說，假設";
		mes "我想的數字是 168。";
		mes "如果你第一次猜 678，那你猜中了其中兩個數字，";
		mes "也就是 6 和 8，但是";
		mes "只有 8 的位置是正確的。";
		next;
		mes "[妮雅]";
		mes "你知道，最好是";
		mes "先試試練習模式，";
		mes "然後，等你比較";
		mes "熟悉了，再玩真正的";
		mes "數字配對遊戲。別忘了，速度會影響你的分數！";
		next;
		mes "[妮雅]";
		mes "啊，如果你恰好";
		mes "創下新紀錄，在";
		mes "數字配對遊戲中，你將會";
		mes "讓你的大名被記錄下來，";
		mes "千萬別錯過這個機會，";
		mes "名利雙收的機會喔~";
		close;
	case 2:
		mes "[妮雅]";
		mes "太好了，我們來玩一場";
		mes "數字配對遊戲的練習賽。請";
		mes "試著猜猜我的數字，好嗎？";
		mes "如果你想取消，請輸入大於 1,000 的數字。";
		set .@practice,1;
		break;
	case 3:
		mes "[妮雅]";
		mes "太好了，我們來玩數字配對吧！";
		mes "你會有 5 次機會猜出我心中的數字。";
		mes "如果你想取消，請輸入大於 1,000 的數字。";
		set .@practice,0;
		break;
	case 4:
		set .@name_record$, $050908_minus1_yagu$;
		if (.@name_record$ == "") //custom
			set .@name_record$,"(null)";
		set .@score_record, $050908_minus1_yagu;
		set .@score_min, .@score_record / 60;
		set .@score_sec, .@score_record % 60;
		mes "[妮雅]";
		mes "^ff0000" + .@name_record$ + "^000000 是";
		mes "數字配對遊戲的";
		mes "最快紀錄保持者，時間為";
		mes "^ff0000" + .@score_min + " 分 " + .@score_sec + " 秒^000000。";
		close;
	case 5:
		mes "[妮雅]";
		mes "一路保重，";
		mes "冒險者。如果你哪天";
		mes "想玩數字";
		mes "配對遊戲，就來找我說一聲";
		mes "好嗎？再見囉~";
		close;
	}
	next;

	// Generate 3 digits of a number.
	while(1) {
		set .@yagu100, rand(1,9);
		set .@yagu10, rand(1,9);
		set .@yagu1, rand(1,9);
		if ((.@yagu100 != .@yagu10) && (.@yagu100 != .@yagu1) && (.@yagu10 != .@yagu1)) {
			// Store the actual number (unofficial).
			set .@yagu, (100*.@yagu100) + (10*.@yagu10) + .@yagu1;
			break;
		}
	}

	// The official time calculation is based on ServerTime,
	// an int HHMMSS equivalent to atoi(gettimestr("%H%M%S",7)).
	// UNIX time is really a better way to do this.
	set .@start_time, gettimetick(2);

	while(1) {

		// Guess a number that meets all conditions.
		while(1) {
			if (input(.@input,100,999) != 0) {
				mes "[妮雅]";
				mes "你已取消了";
				mes "數字配對遊戲。";
				close;
			}
			set .@input100, .@input / 100;
			set .@input10, (.@input % 100) / 10;
			set .@input1, .@input % 10;
			if ((.@input100 > 0) && (.@input10 > 0) && (.@input1 > 0)) {
				if ((.@input100 != .@input10) && (.@input100 != .@input1) && (.@input10 != .@input1))
					break;
			}
			mes "[妮雅]";
			mes "你輸入的其中一個";
			mes "數字是 0，或是輸入了";
			mes "相同的數值在";
			mes "多個位數中重複出現。";
			mes "請再輸入一次猜測的數字。";
			next;
		}

		set .@retry, .@retry+1;
		mes "[妮雅]";
		mes "^ff0000"+strcharinfo(0)+"^000000,";
		mes "你猜的是 ^0000ff"+.@input+"^000000.";
		mes "請稍等一下，我";
		mes "來公布你的結果。";
		next;

		// Correct digits in correct locations.
		set .@strike,0;
		if (.@yagu100 == .@input100)
			set .@strike, .@strike+1;
		if (.@yagu10 == .@input10)
			set .@strike, .@strike+1;
		if (.@yagu1 == .@input1)
			set .@strike, .@strike+1;

		// Correct digits in incorrect locations.
		set .@ball,0;
		if ((.@yagu100 == .@input10) || (.@yagu100 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu10 == .@input100) || (.@yagu10 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu1 == .@input100) || (.@yagu1 == .@input10))
			set .@ball, .@ball+1;

		if (.@strike == 3) {
			if (.@practice) {
				mes "[妮雅]";
				mes "沒錯，我心中的數字是";
				mes "我想的是 ^ff0000"+.@yagu+"^000000!";
				mes "恭喜你，";
				mes "剛剛贏得了這場練習賽";
				mes "！現在你應該已經";
				mes "準備好迎接真正的挑戰了~";
				close;
			} else {

				// Calculate final time and check against the previous record.
				set .@end_time, gettimetick(2);
				if (.@end_time < .@start_time)
					set .@end_time, .@end_time + 3600;
				set .@total_time, .@end_time - .@start_time;
				set .@total_min, .@total_time / 60;
				set .@total_sec, .@total_time % 60;
				set .@name_record$, $050908_minus1_yagu$;
				if (.@name_record$ == "") //custom
					set .@name_record$,"(null)";
				set .@score_record, $050908_minus1_yagu;
				set .@score_min, .@score_record / 60;
				set .@score_sec, .@score_record % 60;

				if (.@score_record >= .@total_time || !.@score_record) { //custom (second condition)
					set $050908_minus1_yagu$, strcharinfo(0);
					set $050908_minus1_yagu, .@total_time;
					mes "[妮雅]";
					mes "太棒了！你猜中了";
					mes "我心中的數字，就是";
					mes "^ff0000" + .@yagu + "^000000。你花費了";
					mes "^ff0000" + .@total_min + " 分 " + .@total_sec + " 秒^000000，";
					mes "打破了先前的紀錄";
					mes "的 ^ff0000" + .@score_min + " 分 " + .@score_sec + " 秒^000000。";
					next;
					mes "[妮雅]";
					mes "^ff0000" + .@name_record$ + "^000000";
					mes "保持的舊紀錄，";
					mes "但現在你成了";
					mes "數字配對遊戲中";
					mes "新的挑戰對象。恭喜";
					mes "你刷新了紀錄！";
					close;
				}
				mes "[妮雅]";
				mes "做得好，你猜中了";
				mes "我的數字！答案，當然";
				mes "就是 ^ff0000" + .@yagu + "^000000。你成功";
				mes "猜對了這個數字，";
				mes "耗時 ^ff0000" + .@total_min + " 分 " + .@total_sec + " 秒^000000。";
				next;
				mes "[妮雅]";
				mes "^ff0000" + .@name_record$ + "^000000 所有，";
				mes "目前數字配對遊戲的";
				mes "最快紀錄保持者，時間為";
				mes "的 ^ff0000" + .@score_min + " 分 " + .@score_sec + " 秒^000000。";
				mes "祝你好運，希望你";
				mes "有朝一日能打破這個紀錄~";
				close;
			}
		} else {
			if (.@practice) {
				mes "[妮雅]";
				mes "你的猜測中有 ^ff0000" + .@ball + "^000000 個";
				mes "數字正確，^ff0000" + .@strike + "^000000 個";
				mes "位數中，有";
				mes "數字正確且";
				mes "位置也正確。";
				mes "好吧，再試一次吧~";
				next;
			} else {
				mes "[妮雅]";
				mes "你上次的猜測中有 ^ff0000" + .@ball + "^000000";
				mes "個數字正確，";
				mes "以及 ^ff0000" + .@strike + "^000000 個位數，";
				mes "數字正確且";
				mes "位置也正確。";
				switch(.@retry) {
					case 1: mes "You have four guesses left."; break;
					case 2: mes "You have three guesses left."; break;
					case 3: mes "You have two guesses left."; break;
					case 4: mes "You only have one guess left..."; break;
					case 5: mes "The answer was ^ff0000" + .@yagu + "^000000."; close;
				}
				next;
			}
		}
	}
}
