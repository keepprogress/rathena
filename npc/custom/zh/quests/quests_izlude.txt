// Edgar's Offer :: iz_npc
//============================================================
-	script	::Edgar_izlude	-1,{
	if (MISC_QUEST & 16) {
		mes "[艾德加]";
		mes "你又要去艾爾貝塔嗎？跟上次一樣，給你同樣的折扣，只收 250 Zeny。";
		mes "這樣如何？";
		next;
		if (select("好啦～！:你為什麼對我這麼好？！") == 1) {
			if (Zeny < 250) {
				mes "[艾德加]";
				mes "嗯……";
				mes "這不是";
				mes "足夠的錢。";
				mes "你何不先去";
				mes "該去多賺點現金的時候嗎？";
				close ;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[艾德加]";
		mes "我就是這個樣子。況且，你那壞壞的笑容讓我想起了我心愛的金髮兒子。他多年前離家，自己跑去當水手，有了自己的船。願他平安，不管他在哪裡。";
		close;
	}
	mes "[艾德加]";
	mes "我的城鎮依斯魯得，透過西邊的港口和艾爾貝塔相連。我們之間的往來實在太頻繁了，我差點都要變成艾爾貝塔人了。呵呵～";
	next;
	mes "[艾德加]";
	mes "我有個挺熟的人叫菲利克斯，住在艾爾貝塔。那傢伙真的超小氣……什麼都要收錢！";
	next;
	mes "[艾德加]";
	mes "不過他其實是個好人，也很樂於助人。他心地善良，只要你付得起他的價錢，他就會支持你。";
	next;
	mes "[艾德加]";
	mes "最近艾爾貝塔的人都說，他其實是真心想幫大家，他要的果凍只是幌子。嗯，你應該去看看他能提供什麼。";
	next;
	if (select("你能告訴我怎麼去艾爾貝塔嗎？:結束對話") == 1) {
		mes "[艾德加]";
		mes "咦？嗯，你可以用腳走過去。不過如果你有錢，我建議你搭船。";
		next;
		if (select("好，我知道了。:可是我走膩了，而且身無分文！") == 1) {
			mes "[艾德加]";
			mes "好啦，保重～";
			close;
		}
		mes "[艾德加]";
		mes "好吧……";
		mes "你既不想走路，身上又沒錢，卻還想去那裡？天啊……";
		next;
		mes "[艾德加]";
		mes "好好好。身為船長的我，可以用更便宜的價格載你過去。250 Zeny 如何？";
		next;
		if (select("好啦～！:呸，這也太坑人了吧！！") == 1) {
			set MISC_QUEST,MISC_QUEST | 16;
			if (Zeny < 250) {
				mes "[艾德加]";
				mes "嗯……";
				mes "這不是";
				mes "足夠的錢。";
				mes "去拿";
				mes "更多的錢吧。";
				close;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[艾德加]";
		mes "天啊，";
		mes "如果你覺得";
		mes "這樣很坑人……";
		close;
	}
	mes "[艾德加]";
	mes "好吧。";
	mes "待會見~";
	close;
}
