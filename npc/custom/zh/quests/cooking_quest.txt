S_SellSets:
	.@item_id = getarg(0);
	.@item_cost = getiteminfo(.@item_id, ITEMINFO_BUY);
	.@item_weight = getiteminfo(.@item_id, ITEMINFO_WEIGHT);
	mes "[瑪德琳·朱]";
	mes "要買幾 "+ (.@item_id == 12125 ? "戶外":"室內");
	mes "烹飪套件要";
	mes "買嗎？如果你";
	mes "想取消，請";
	mes "輸入數字0。";
	next;
	while(1) {
		input .@sell;
		if (.@sell == 0) {
			mes "[瑪德琳·朱]";
			mes "你改變主意了";
			mes "嗎？如果你需要";
			mes "購買 "+ getitemname(.@item_id);
			mes "套件，之後再來";
			mes "找我就好，好嗎？";
			close;
		}
		else if (.@sell > 100) {
			mes "[瑪德琳·朱]";
			mes "哦，很抱歉，但";
			mes "我一次不賣超過";
			mes "100 "+ getitemname(.@item_id);
			mes "的數量，只是為了安全。";
			next;
		}
		else break;
	}
	set .@total_cost,.@sell * .@item_cost;
	set .@total_weight,.@sell * .@item_weight;
	if (Zeny < .@total_cost) {
		mes "[瑪德琳·朱]";
		mes "哎呀，很抱歉，你";
		mes "你的錢買不起這麼多";
		mes getitemname(.@item_id);
		mes "請確認你的Zeny";
		mes "是否足夠購買我的套件~";
		close;
	}
	if (!checkweight(.@item_id,.@sell)) {
		mes "[瑪德琳·朱]";
		mes "很抱歉，您";
		mes "你的背包還有足夠空間";
		mes "放得下這麼多嗎？";
		mes getitemname(.@item_id) +"...";
		close;
	}
	set Zeny, Zeny-.@total_cost;
	getitem .@item_id,.@sell;
	mes "[瑪德琳·朱]";
	mes "給你~";
	mes "祝你";
	mes "烹飪修行順利！";
	return;
}
