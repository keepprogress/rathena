payon_in03,188,146,5	script	馬克斯·韓森#juice::Marx Hansen#juice	86,{
	if (checkweight(1201,1) == 0) {
		mes "- 請稍等！ -";
		mes "- 你目前攜帶的 -";
		mes "- 道具太多了。 -";
		mes "- 請稍後再來 -";
		mes "- 等你把部分道具存入卡普拉倉庫後 -";
		close;
	}
	if (MISC_QUEST&1) {
		mes "[商人馬克斯漢森]";
		mes "歡迎。";
		mes "你是來";
		mes "也會加工水果嗎？";
		next;
		switch(select("製作果汁。:交談並取得水果加工資訊。:取消。")) {
		case 1:
			mes "[商人馬克斯漢森]";
			mes "你想製作哪種果汁？";
			next;
			switch(select("蘋果汁:香蕉汁:紅蘿蔔汁:葡萄汁:取消")) {
			case 1:
				set .@fruit,512;
				set .@juice,531;
				break;
			case 2:
				set .@fruit,513;
				set .@juice,532;
				break;
			case 3:
				set .@fruit,515;
				set .@juice,534;
				break;
			case 4:
				set .@fruit,514;
				set .@juice,533;
				break;
			case 5:
				mes "[商人馬克斯漢森]";
				mes "那麼……";
				mes "下次見。";
				close;
			}
			if (countitem(.@fruit) == 0 || countitem(713) == 0 || Zeny < 3) {
				mes "[商人馬克斯漢森]";
				mes "糟糕……";
				mes "你的材料不齊。要製作 "+getitemname(.@juice)+"，我需要 1 個 "+getitemname(.@fruit)+" 和 1 個空瓶，還需要 3 Zeny 的費用。";
				next;
				mes "[商人馬克斯漢森]";
				mes "材料都準備好之後，我會把水果打成美味的果汁給你。";
				close;
			}

			mes "[商人馬克斯漢森]";
			mes "你要買多少？";
			next;
			switch(select("能做多少做多少。:我要指定數量。:取消。")) {
			case 1:
				set .@make,countitem(.@fruit);
				if (countitem(713) < .@make) set .@make, countitem(713);
				if (Zeny/3 < .@make) set .@make, Zeny/3;
				break;
			case 2:
				mes "[商人馬克斯漢森]";
				mes "請選擇小於 100 的數字。不想指定的話，請輸入「0」。最多可以製作 " + countitem(.@fruit) + " 瓶果汁。";
				next;
				while(1) {
					input .@input;
					if (.@input == 0) {
						mes "[商人馬克斯漢森]";
						mes "那麼……";
						mes "下次再來。";
						close;
					}
					else if (.@input > 100) {
						mes "[商人馬克斯漢森]";
						mes "超過 100 瓶是不可能的。請重新選擇數量。";
						next;
					}
					else {
						break;
					}
				}
				set .@make,.@input;
				break;
			case 3:
				mes "[商人馬克斯漢森]";
				mes "那麼……";
				mes "下次再來。";
				close;
			}

			set .@total_zeny,3 * .@make;

			if (countitem(.@fruit) < .@make || countitem(713) < .@make || Zeny < .@total_zeny) {
				mes "[商人馬克斯漢森]";
				mes "糟糕……";
				mes "你的材料不齊。我幫不了這種情況。你還是把需要的東西先收集齊吧。";
				close;
			}
			delitem .@fruit,.@make;
			delitem 713,.@make;
			set Zeny, Zeny-.@total_zeny;
			getitem .@juice,.@make;

			mes "[商人馬克斯漢森]";
			mes "給你！如約送上新鮮美味的果汁。應該非常清爽可口。";
			next;
			mes "[商人馬克斯漢森]";
			mes "那麼……";
			mes "下次再來。";
			close;
		case 2:
			mes "[商人馬克斯漢森]";
			mes "在人類能夠發展出如此龐大的社會之前，人們靠採集樹上的果實維生。水果或許是大自然的恩賜，讓我們得以在這個世界生存。";
			next;
			mes "[商人馬克斯漢森]";
			mes "生活變得富足之後，年輕一代似乎就不再吃水果了。所以我開始思考，該怎麼讓水果更容易入口。";
			next;
			mes "[商人馬克斯漢森]";
			mes "我發現把水果做成果汁，不但更方便食用，口感也好得多。";
			next;
			mes "果汁資訊";
			mes "^CC4E5C- 蘋果汁 -^000000";
			mes "蘋果 x 1 個，空瓶 x 1 個，3 Zeny。";
			mes "^E3CF57- 香蕉汁 -^000000";
			mes "香蕉 x 1 個，空瓶 x 1 個，3 Zeny。";
			mes "^ED9121- 紅蘿蔔汁 -^000000";
			mes "紅蘿蔔 x 1 個，空瓶 x 1 個，3 Zeny。";
			mes "^CC00FF- 葡萄汁 -^000000";
			mes "葡萄 x 1 個，空瓶 x 1 個，3 Zeny。";
			close;
		case 3:
			mes "[商人馬克斯漢森]";
			mes "嘿！";
			mes "如果你去拜訪";
			mes "別人，記得找他們聊聊！";
			close;
		}
	}
	else {
		mes "[商人馬克斯漢森]";
		mes "歡迎。";
		mes "有什麼可以為您效勞的嗎？";
		next;
		if (select("交談:取消") == 1) {
			mes "[商人馬克斯漢森]";
			mes "在人類能夠發展出如此龐大的社會之前，人們靠採集樹上的果實維生。水果或許是大自然的恩賜，讓我們得以在這個世界生存。";
			next;
			mes "[商人馬克斯漢森]";
			mes "生活變得富足之後，年輕一代似乎就不再吃水果了。所以我開始思考，該怎麼讓水果更容易入口。";
			next;
			mes "[商人馬克斯漢森]";
			mes "我發現把水果做成果汁，不但更方便食用，口感也好得多。";
			close;
		}
		mes "[商人馬克斯漢森]";
		mes "嘿！";
		mes "如果你去拜訪";
		mes "別人，記得找他們聊聊！";
		close;
	}
