		switch(select("當然，我帶來了道具。:活動資訊:取消")) {
		case 1:
			mes "[卡普拉職員]";
			mes "好的～";
			mes "讓我確認";
			mes "看看你有沒有帶來";
			mes "所有道具……";
			next;
			setarray .@Items[0],949,100,706,1,722,1,2213,1;
			for(set .@i,0; .@i<8; set .@i,.@i+2)
				if (countitem(.@Items[.@i]) < .@Items[.@i+1]) {
					mes "[卡普拉職員]";
					mes "哎呀，很抱歉";
					mes "但你還需要";
					mes "至少帶來";
					mes callfunc("F_InsertPlural",.@Items[.@i+1],getitemname(.@Items[.@i]))+".";

			mes "[卡普拉職員]";
			mes "太好了，我看到你";
			mes "已經收集到";
			mes "我需要的全部材料，";
			mes "足以製作兔耳髮圈。";
			mes "請稍等一下";
			mes "等我把它組裝起來……";
			next;
			for(set .@i,0; .@i<8; set .@i,.@i+2)
				if (countitem(.@Items[.@i]) < .@Items[.@i+1]) {
					mes "[卡普拉職員]";
					mes "嗯？抱歉，";
					mes "但我現在其實沒辦法";
					mes "馬上做好它。你";
					mes "還需要 "+callfunc("F_InsertPlural",.@Items[.@i+1],getitemname(.@Items[.@i]));
					mes "才能讓我把這個";
					mes "兔耳髮圈組合起來……";
					close;
				}
			delitem 949,100; //Feather
			delitem 706,1; //Four_Leaf_Clover
			delitem 722,1; //Scarlet_Jewel
			delitem 2213,1; //Cat_Hairband
			mes "[卡普拉職員]";
			mes "啊，給你～";
			mes "完美的兔耳髮圈！";
			mes "嗯，希望你會喜歡。";
			getitem 2214,1; //Bunny_Band
			set BUNYBND,0;
			next;
			mes "[卡普拉職員]";
			mes "感謝您";
			mes "參加這場";
			mes "特別活動，也感謝你的";
			mes "持續支持";
			mes "卡普拉服務～";
			close;
		case 2:
			goto L_Info;
		case 3:
			close;
		}
	}
	mes "[卡普拉職員]";
	mes "嗨～！你要不要";
	mes "參加我們的特別^529DFF兔耳";

	case 1:
		set BUNYBND,1;
		mes "[卡普拉職員]";
		mes "太好了！感謝你";
		mes "的參與！如果你";
		mes "還沒聽說過，";

	mes "這次的特別款，";
	mes "獨一無二的道具，";
	mes "請帶來 100 根羽毛，";
	mes "1 株四葉幸運草，";
	mes "1 個貓耳髮圈，以及";
	mes "1 顆珍珠。";

	mes "把道具交給我。";
	mes "（由";
	mes "艾爾貝塔商人公會贊助）。";
