niflheim,201,187,3	script	尼福海姆嚮導#01nif::Niflheim Guide#01nif	849,{
	mes "[尼福海姆嚮導]";
	mes "歡迎來到……^8B4513尼福海姆^000000……";
	mes "死者之境……";
	mes "需要我幫你在這片境界導航嗎……？";
	if (F_Navi() == 0) {
		next;
		mes "[尼福海姆嚮導]";
		mes "當你點擊..^B9062F[地點名稱]^000000……你就能享受最先進的^B9062F導航^000000服務……！";
	}
	while (1) {
		next;
		switch(select("[ 工具店 ]:[ 酒館 ]:[ 武器店 ]:[ 女巫城堡 ]:移除小地圖標記:取消")) {
		case 1:
			callsub L_Mark, F_Navi("Tool Shop","niflheim,218,196","^0000FF");
			viewpoint 1,218,196,0,0x0A82FF;
			break;
		case 2:
			callsub L_Mark, F_Navi("Pub","niflheim,189,208","^006400");
			viewpoint 1,189,208,1,0xAAFF00;
			break;
		case 3:
			callsub L_Mark, F_Navi("Weapon Shop","niflheim,218,170","^800080");
			viewpoint 1,218,170,2,0xDA70D6;
			break;
		case 4:
			callsub L_Mark, F_Navi("Witch's Castle","niflheim,255,195","^FF1493");
			viewpoint 1,255,195,3,0xFF1493;
			break;
		case 5:
			mes "[尼福海姆嚮導]";
			mes "好的，我會把你小地圖上的標記全部移除……";
			mes "還有什麼我可以為你做的嗎……？";
			viewpoint 2,1,1,0,0xFFFFFF;
			viewpoint 2,1,1,1,0xFFFFFF;
			viewpoint 2,1,1,2,0xFFFFFF;
			viewpoint 2,1,1,3,0xFFFFFF;
			break;
		case 6:
			mes "[尼福海姆嚮導]";
			mes "……掰掰~!..";
			close;
		}
	}
	end;
L_Mark:
	mes "[尼福海姆嚮導]";
	mes "我會在小地圖上標示";
	mes getarg(0)+"^000000...";
	mes "你的小地圖上標示它的位置。";
	mes "你想看看其他地點嗎？";
	return;
}
