ayothaya,203,169,4	script	Ayothaya Guide Noi#01ay	839,{
	if (strnpcinfo(0) == "Ayothaya Guide Noi#01ay")
		set .@n$, "[Noi]";
	else
		set .@n$, "[Noa]";
	mes .@n$;
	mes "你好～";
	mes "歡迎來到^8B4513阿尤塔亞^000000。";
	mes "這座城鎮透過野外森林與一條河流相連。";
	mes "這是一個非常寧靜的國家。";
	mes "慢慢看看吧。";
	F_Navi(.@n$);
	while (1) {
		next;
		switch(select("[ 武器店 ]:[ 工具商店 ]:[ 酒館 ]:[ 神社 ]:[ 釣魚點 ]:[ 阿伊巴克辛 ]:移除小地圖標記:取消")) {
		case 1:
			mes .@n$;
			mes "此外 "+F_Navi("武器店","ayothaya,163,90","^0000FF");
			mes "在阿尤塔亞擁有特殊武器。";
			mes "你在那裡能找到很棒的武器。";
			mes "還有什麼我能為你效勞的嗎？";
			viewpoint 1,163,90,0,0x0A82FF;
			break;
		case 2:
			mes .@n$;
			mes "我們的市民大多在 "+F_Navi("工具商店","ayothaya,131,86","^006400")+".";
			mes "記得在那裡買齊你需要的東西。";
			mes "還需要什麼嗎？";
			viewpoint 1,131,86,1,0xAAFF00;
			break;
		case 3:
			mes .@n$;
			mes "冒險的一部分就是收集情報。";
			mes "此外 "+F_Navi("酒館","ayothaya,229,72","^008080")+" 擠滿了許多旅人，在那裡可以獲得有用的情報。";
			mes "還有什麼我能為你效勞的嗎？";
			viewpoint 1,229,72,2,0x008080;
			break;
		case 4:
			mes .@n$;
			mes "想靜下心來，請前往 "+F_Navi("神社","ayothaya,208,280","^FF1493")+" 並且祈禱一下。";
			mes "還有什麼我能為你效勞的嗎？";
			viewpoint 1,208,280,3,0xFF1493;
			break;
		case 5:
			mes .@n$;
			mes "阿尤塔亞靠近海灘，所以大多數人都喜歡釣魚。";
			mes "他們真的很愛釣魚。";
			mes "去 "+F_Navi("釣魚","ayothaya,255,99","^8B4513")+" 試試看能不能釣到一條大魚。";
			mes "還有什麼我能為你效勞的嗎？";
			viewpoint 1,255,99,4,0x8B4513;
			break;
		case 6:
			mes .@n$;
			mes "如果你想回家，";
			mes "請找 "+F_Navi("阿伊巴克辛","ayothaya,151,68","^00BFFF");
			mes "並請他把你送回去。";
			mes "還有什麼我能為你效勞的嗎？";
			viewpoint 1,151,68,5,0x00BFFF;
			break;
		case 7:
			mes .@n$;
			mes "我會移除你小地圖上的所有標記。";
			mes "還需要什麼嗎？？";
			viewpoint 2,1,1,0,0xFFFFFF;
			viewpoint 2,1,1,1,0xFFFFFF;
			viewpoint 2,1,1,2,0xFFFFFF;
			viewpoint 2,1,1,3,0xFFFFFF;
			viewpoint 2,1,1,4,0xFFFFFF;
			viewpoint 2,1,1,5,0xFFFFFF;
			break;
		case 8:
			mes .@n$;
			mes "祝旅途愉快~！";
			close;
		}
	}
}
