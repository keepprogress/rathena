moc_para01,22,16,5	script	目錄魔法師#catal01::Catalog Magician#catal01	704,{
	if (!checkweight(1301,3)) {
		mes "- 你的道具太多了。請騰出空間。 -";
		close;
	}
	set .@ticket_cost, 200;
	mes "[目錄魔法師]";
	mes "你看……魔法學院";
	mes "在格芬的分院現在直接";
	mes "販售全新的";
	mes "^007777露天商店檢索卡(銀)^000000！";
	next;
	mes "他正盯著手上";
	mes "寫的字。";
	mes "他似乎太忙了，";
	mes "沒空看你。";
	next;
	mes "[目錄魔法師]";
	mes "一本全新的目錄";
	mes "剛剛推出了！";
	mes "你可以查看";
	mes "可以交易的道具";
	mes "以及有哪些道具在販售";
	mes "來自販售商人。";
	next;
	mes "[目錄魔法師]";
	mes "^007777露天商店檢索卡(銀)^000000";
	mes "只要 200z！";
	mes "一次最多可以買 50 個！";
	next;
	switch(select("購買 ^007777露天商店檢索卡(銀)^000000:我不需要！:你看起來不像商人！")) {
	case 1:
		mes "[目錄魔法師]";
		mes "你……要全部 50 個？";
		mes "還是要多少個？";
		next;
		while (1) {
			input .@input;
			mes "[目錄魔法師]";
			if (.@input == 0) {
				mes "交易已經中止了！";
				mes "我也不知道接下來該怎麼辦……";
				mes "我該怎麼辦？";
				emotion ET_PROFUSELY_SWEAT, playerattached();
				close;
			} else if (.@input < 1 || .@input > 50) {
				mes "應該要少於 50 個才對。";
				next;
			} else
				break;
		}
		set .@sell, .@ticket_cost * .@input;
		mes "你想購買的目錄總數為 " + .@input + " 個。";
		mes "需要 " + .@sell + " z。";
		if (Zeny < .@sell) {
			mes "你的錢似乎不夠。";
			mes "你可以把裝備";
			mes "賣掉來賺更多錢……如果你想要的話。";
			close;
		}
		mes "好，給你。";
		set Zeny, Zeny - .@sell;
		getitem 12580,.@input; //Vending_Search_Scroll
		close;
	case 2:
		mes "[目錄魔法師]";
		mes "唉……你真的很需要這";
		mes "個道具……要我";
		mes "再解釋一次嗎？我不知道";
		mes "接下來該怎麼做……";
		mes "我該怎麼辦？";
		emotion ET_PROFUSELY_SWEAT, playerattached();
		close;
	case 3:
		mes "[目錄魔法師]";
		mes "我真正的職業是魔法師。";
		mes "我本來想請商人";
		mes "公會來販售，但他們";
		mes "想把它們當作附屬品";
		mes "拿去推銷掃把。所以";
		mes "我才自己販售這些。";
		next;
		mes "[目錄魔法師]";
		mes "這裡有很多冒險者";
		mes "來來去去。";
		mes "我很確定很快就能全都賣給他們。";
		mes "你不這麼認為嗎？";
		mes "那麼……開始吧！多買一些吧，拜託！拜託……！";
		close;
	}
}
