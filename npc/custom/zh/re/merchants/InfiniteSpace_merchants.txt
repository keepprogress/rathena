cmd_fild07,375,167,1	script	遺跡黑商人#pa0829_01::Ruins Black Trader#pa0829_01	4_F_JOB_HUNTER,{
	mes "[遺跡黑商人]";
	mes "嘿，每次進出這裡不是很辛苦嗎？只要 ^0000ff20,000^000000 Zeny，就能拿到一張直接傳送到遺跡入口的卷軸，你覺得如何？";
Purchase:
	next;
	switch( select( "給我一張", "不用了" )) {
		case 1:
			if (!checkweight(22980,1) || (MaxWeight - Weight) < 1000) {
				mes "你的道具數量太多，無法繼續對話。";
				mes "請整理你的道具後再試一次。";
				close;
			}
			else if (Zeny < 20000) {
				mes "[遺跡黑商人]";
				mes "您的錢好像不夠了。卷軸要 20,000 Zeny。";
				close;
			}
			Zeny -= 20000;
			getitem 22980,1;
			mes "[遺跡黑商人]";
			mes "很划算吧。還需要嗎？";
			goto Purchase;
		case 2:
			mes "[遺跡黑商人]";
			mes "隨時再來";
			close;
	}
}

// Equipment Shop
cmd_fild07,57,275,5	script	神器鑑定師#pa0829_01::Artifact Appraiser#pa0829_01	1_F_02,{
	if (!checkweight(1201,1) || (MaxWeight - Weight) < 1000) {
		mes "你的道具數量太多，無法繼續對話。";
		mes "請整理你的道具後再試一次。";
		close;
	}
	.@stone_id = 6905;
	mes "[ 神器鑑定師 ]";
	mes "選擇您想購買的裝備種類。只要您持有足夠的 " + getitemname(.@stone_id) + ".";
	next;
	switch (select("取消:武器:盔甲")) {
		case 1:
			mes "[ 神器鑑定師 ]";
			mes "隨時歡迎再來～";
			close;

		case 2:
			setarray .@equip_id,1994,1938,13323,13126,28703,2024,16038,21014,28105,18128;
			.@price = 50;
			break;

		case 3:
			setarray .@equip_id,15141,22075,20779,19033;
			.@price = 50;
			break;
	}
	.@menu$ = "Cancel:";
	for (.@i = 0; .@i < getarraysize(.@equip_id); .@i++)
		.@menu$ += getitemname(.@equip_id[.@i]) + ":";
	.@s = select(.@menu$) - 1;
	switch (.@s) {
		case 0:
			mes "[ 神器鑑定師 ]";
			mes "隨時歡迎再來～";
			close;

		default:
			.@s--;
			mes "[ 神器鑑定師 ]";
			mes "您需要 ^0000FF" + .@price + "^000000 " + getitemname(.@stone_id) + " 才能購買 " + getitemname(.@equip_id[.@s]) + "~";
			next;
			if (select("取消:購買") == 1) {
				mes "[ 神器鑑定師 ]";
				mes "隨時歡迎再來～";
				close;
			}
			if (countitem(.@stone_id) < .@price) {
				mes "[ 神器鑑定師 ]";
				mes "您的 " + getitemname(.@stone_id) + " 不足，無法購買此物品。";
				close;
			}
			mes "[ 神器鑑定師 ]";
			mes "感謝您的信任。下次請再來。";
			delitem .@stone_id,.@price;
			getitem .@equip_id[.@s],1;
			close;
	}
}

// Equipment Enchanter
cmd_fild07,60,275,3	script	神器強化師#pa0829_01::Artifact Enhancer#pa0829_01	4_F_JOB_BLACKSMITH,{
	if (!checkweight(1201,1) || (MaxWeight - Weight) < 1000) {
		mes "你的道具數量太多，無法繼續對話。";
		mes "請整理你的道具後再試一次。";
		close;
	}
	disable_items;
	.@stone_id = 6905;
	function equip_check;
	mes "[ 神器強化師 ]";
	mes "如果您想強化在無限空間取得的裝備，那您來對地方了。";
	mes "您有任何 " + getitemname(.@stone_id) + "?";
	next;
	switch (select("我要怎麼附魔裝備？:附魔裝備。:重置裝備的附魔。")) {
		case 1:
			mes "[ 神器強化師 ]";
			mes "您將可以取得 " + getitemname(.@stone_id) + "，只要探索帕洛斯燈塔下方的空間即可。";
			next;
			mes "[ 神器強化師 ]";
			mes "有了那份材料，我就能強化您從無限空間取得的裝備。";
			next;
			mes "[ 神器強化師 ]";
			mes "我只能附魔第 3 和第 4 個插槽。";
			next;
			mes "[ 神器強化師 ]";
			mes "附魔過程中裝備不會毀壞，但重置附魔時很有可能會毀壞。";
			next;
			mes "[ 神器強化師 ]";
			mes "想附魔裝備的時候再來找我吧。";
			break;

		case 2:
			.@stone_id = 6905;
			.@fee = 20;
			mes "[ 神器強化師 ]";
			mes "請選擇您想附魔的裝備。";
			next;
			switch (select("取消:武器:盔甲:長靴:披風:金屬頭盔")) {
				case 1:
					mes "[ 神器強化師 ]";
					mes "想附魔裝備的時候再來找我吧。";
					close;

				case 2:
					.@part = EQI_HAND_R;
					break;

				case 3:
					.@part = EQI_ARMOR;
					break;

				case 4:
					.@part = EQI_SHOES;
					break;

				case 5:
					.@part = EQI_GARMENT;
					break;

				case 6:
					.@part = EQI_HEAD_TOP;
					break;
			}
			.@equip_id = getequipid(.@part);
			.@refine = getequiprefinerycnt(.@part);
			equip_check(.@equip_id);
			for (.@i = 0;.@i < 4;.@i++) {
				.@card[.@i] = getequipcardid(.@part,.@i);
				.@check[.@i] = getequipcardid(.@part,.@i);
			}
			if (.@card[2] > 0) {
				mes "[ 神器強化師 ]";
				mes "我只能附魔到第三個插槽。您的裝備無法再附魔了。";
				close;
			}
			switch (.@part) {
				case EQI_HAND_R:
					setarray .@enchant$,
					"4700,4701:4710,4711:4720,4721",
					"4811,4810,4809,4808,4820,4821,4822,4823:4815,4814,4813,4812,4826,4827,4828,4829:4832,4833,4834,4835,4836,4837,4838,4839";
					break;

				case EQI_ARMOR:
				case EQI_SHOES:
					setarray .@enchant$,
					"4700,4701,4702,4703:4710,4711,4712,4713:4720,4721,4722,4723",
					"4795,4796,4797:4870,4871,4800:4870,4871,4800";
					break;

				case EQI_GARMENT:
				case EQI_HEAD_TOP:
					setarray .@enchant$,
					"4700,4701,4702,4703:4710,4711,4712,4713:4720,4721,4722,4723",
					"4861,4862,4867,4868,4900:4861,4862,4867,4868,4900:4861,4862,4867,4868,4900";
					break;
			}
			.@slot = (.@card[3] > 0 ? 2 : 3);
			.@index = (.@slot == 3 ? 0 : 1);
			mes "[ 神器強化師 ]";
			mes "您可以選擇 3 種附魔，附魔費用是 " + .@fee + " " + getitemname(.@stone_id) + ".";
			mes "我保證會順利附魔的。";
			next;
			.@type = select("離開:物理:魔法:遠程") - 2;
			mes "[ 神器強化師 ]";
			if (.@slot == 3)
				mes "好，我們開始第一次附魔。";
			else
				mes "好，我們進行第二次附魔。";
			next;
			if (select("我等會再來。:請繼續。") == 1) {
				mes "[ 神器強化師 ]";
				mes "如果您改變主意，隨時回來找我。";
				close;
			}
			if (countitem(.@stone_id) < .@fee) {
				mes "[ 神器強化師 ]";
				mes "嗯。順帶一提，您好像沒在聽，您的 " + getitemname(.@stone_id) + ".";
				close;
			}
			explode(.@T$,.@enchant$[.@index],":");
			explode(.@TT$,.@T$[.@type],",");
			.@enchant = atoi(.@TT$[rand(getarraysize(.@TT$))]);
			.@card[.@slot] = .@enchant;
			delitem .@stone_id,.@fee;
			if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipRefineHack", .@part, .@refine) || callfunc("F_IsEquipCardHack", .@part, .@check[0], .@check[1], .@check[2], .@check[3]))
				close;
			delequip .@part;
			getitem2 .@equip_id,1,1,.@refine,0,.@card[0],.@card[1],.@card[2],.@card[3];
			specialeffect2 EF_REPAIRWEAPON;
			mes "[ 神器強化師 ]";
			mes "嗯。做得很好。請立即確認您的裝備。";
			break;
		
		case 3:
			.@stone_id = 6905;
			.@fee = 30;
			.@break_chance = 30;
			mes "[ 神器強化師 ]";
			mes "請選擇您想附魔的裝備。";
			next;
			switch (select("取消:武器:盔甲:長靴:披風:金屬頭盔")) {
				case 1:
					mes "[ 神器強化師 ]";
					mes "想附魔裝備的時候再來找我吧。";
					close;

				case 2:
					.@part = EQI_HAND_R;
					break;

				case 3:
					.@part = EQI_ARMOR;
					break;

				case 4:
					.@part = EQI_SHOES;
					break;

				case 5:
					.@part = EQI_GARMENT;
					break;

				case 6:
					.@part = EQI_HEAD_TOP;
					break;
			}
			.@equip_id = getequipid(.@part);
			.@refine = getequiprefinerycnt(.@part);
			equip_check(.@equip_id);
			for (.@i = 0; .@i < 4; .@i++) {
				.@card[.@i] = getequipcardid(.@part,.@i);
				.@check[.@i] = .@card[.@i];
			}
			if (.@card[3] == 0) {
				mes "[ 神器強化師 ]";
				mes "您的裝備上沒有任何附魔。";
				close;
			}
			mes "[ 神器強化師 ]";
			mes "重置附魔的過程中，您的無限空間裝備有可能會毀壞。您仍要繼續嗎？";
			next;
			if (select("離開:繼續") == 1) {
				mes "[ 神器強化師 ]";
				mes "如果您改變主意，隨時回來找我。";
				close;
			}
			if (countitem(.@stone_id) < .@fee) {
				mes "[ 神器強化師 ]";
				mes "嗯。順帶一提，您好像沒在聽，您的 " + getitemname(.@stone_id) + ".";
				close;
			}
			delitem .@stone_id,.@fee;
			if (callfunc("F_IsEquipIDHack", .@part, .@equip_id) || callfunc("F_IsEquipRefineHack", .@part, .@refine) || callfunc("F_IsEquipCardHack", .@part, .@check[0], .@check[1], .@check[2], .@check[3]))
				close;
			delequip .@part;
			if (rand(1,100) > .@chance) {
				getitem2 .@equip_id,1,1,.@refine,0,.@card[0],.@card[1],0,0;
				specialeffect2 EF_REPAIRWEAPON;
				mes "[ 神器強化師 ]";
				mes "嗯。做得很好。請立即確認您的裝備。";
			} else {
				specialeffect2 EF_REFINEFAIL;
				mes "[ 神器強化師 ]";
				mes "嗯，我可是提醒過您了。您的運氣不太好啊。";
			}
			break;
	}
	close;

	function	equip_check	{
		setarray .@equip_id,1994,1938,13323,13126,28703,2024,16038,21014,28105,18128,15141,22075,20779,19033;
		if (!getarg(0)) {
			mes "[ 神器強化師 ]";
			mes "您有把裝備卸下來嗎？";
			close;
		}
		if (inarray(.@equip_id,getarg(0)) == -1) {
			mes "[ 神器強化師 ]";
			mes "這件裝備不適合附魔。別忘了，只有無限系列的裝備才能附魔。";
			close;
		}
		return;
	}
