prontera,82,108,5	script	心之商人#life01::Heart Merchant#life01	4_M_NFDEADMAN2,5,5,{
	.@old_item = 12209;
	.@new_item = 6413;
	.@old_item_name$ = getitemname(.@old_item);
	.@new_item_name$ = getitemname(.@new_item);

	if (!checkweight(1201,1) || (MaxWeight - Weight) < 500) {
		mes "你的道具數量太多，無法繼續對話。";
		mes "請整理你的道具後再試一次。";
		close;
	}
	mes "[心之商人]";
	mes "你好！我看你是一位把心交給旅途、搭船橫渡大陸、享受冒險刺激的冒險者！";
	next;
	if(!countitem(.@old_item)){
		mes "[心之商人]";
		mes "您有任何 " + mesitemlink( .@old_item ) + " 在你的倉庫裡嗎？";
		next;
		mes "[心之商人]";
		mes "如果你帶來一個 " + .@old_item_name$ + "，我會把它換成 " + mesitemlink( .@new_item ) + "。你覺得如何？你知道這個道具是什麼嗎？";
	} else {
		mes "[心之商人]";
		mes "你有沒有想過把你的 " + mesitemlink( .@old_item ) + " 換成更好的東西呢？例如…… " + mesitemlink( .@new_item ) + "。你覺得如何？";
	}
	next;
	switch(select("那是什麼？",(countitem(.@old_item)?"請幫我兌換！":""),"我沒興趣")) {
	case 1:
		mes "[心之商人]";
		mes mesitemlink( .@old_item ) + " 是一種道具，使用後 30 分鐘內可防止一次死亡時損失經驗值。";
		next;
		mes "[心之商人]";
		mes "不過很多人都為時間限制感到困擾吧？所以我研究了一下，準備了 " + mesitemlink( .@new_item ) + ".";
		next;
		mes "[心之商人]";
		mes mesitemlink( .@new_item ) + " 是一種道具，只要持有就能防止一次死亡時的經驗值損失，^EE0000無論持續時間多久^000000。當然，死亡後 "+.@new_item_name$+" 會被消耗。";
		next;
		select("你為什麼要兌換它？");
		mes "[心之商人]";
		mes "你很好奇嗎？我是個商人。買賣東西是我的宿命，不論那是關乎生死與否。你以為我會錯過這個賺錢的機會嗎？我賺錢，而活著的人也能從死亡的痛苦中獲得喘息。這不是很划算嗎？";
		next;
		mes "[心之商人]";
		mes "兌換比率是 5 個 " + mesitemlink( .@old_item ) + " 換 1 個 " + mesitemlink( .@new_item ) + "。你有什麼問題嗎？";
		next;
		mes "[心之商人]";
		mes "嗯～你覺得如何？這不是很公平的交易嗎？你難道想一直為時間擔憂嗎？對於享受驚險冒險的冒險者來說，這是必備的道具。";
		close;
		
	case 2:	
		mes "[心之商人]";
		mes "對對，很好的決定。換成更好的東西不是比較好嗎？";
		next;
		mes "[心之商人]";
		mes "我能把你的 ^0000cd5 "+.@old_item_name$+" 兌換成 1 "+.@new_item_name$+".";
		next;
		mes "[心之商人]";
		mes "不過，^EE0000[非賣品] " + .@old_item_name$ + " 不符合兌換資格^000000，所以拿來這裡要求兌換是沒用的。";
		next;
		mes "[心之商人]";
		mes "現在要兌換嗎？";
		next;
		if (select("請幫我兌換！:我不想兌換。") == 2)
			break;
		mes "[心之商人]";
		mes "你想兌換幾個？^0000cd最多可以兌換 10 個 "+.@new_item_name$+" 一次^000000。";
		next;
		input .@amount;
		if (.@amount == 0) {
			mes "[心之商人]";
			mes "嗯？你沒興趣嗎？這是很棒的商品，很適合像你這樣的人。不過如果你不想保護你的生命，我也不會強迫你。";
			close;
		}
		if (.@amount > 10) {
			mes "[心之商人]";
			mes "我說過了，^0000cd一次最多只能兌換 10 個^000000。再確認一次數量吧。";
			close;
		}
		if (countitem(.@old_item) < (.@amount*5)) {
			mes "[心之商人]";
			mes "呃，不好意思……您沒有足夠的 " +.@old_item_name$+ "，對吧？需要 5 個 " + mesitemlink( .@old_item ) + " 換 1 個 " + mesitemlink( .@new_item ) + "。請再確認一下數量。";
			close;
		}
		mes "[心之商人]";
		mes "現在，我已經把你的 ^0000cd" + .@old_item_name$ + "^000000 換成了 ^EE0000" + .@new_item_name$ + "^000000！！這筆交易很划算吧？下次如果你拿到 "+.@old_item_name$+"，再來找我吧！";
		delitem .@old_item,.@amount*5;
		getitem .@new_item,.@amount;
		close;
	case 3:
		break;
	}
	mes "[心之商人]";
	mes "這筆交易很棒！冒險者，這是像你這樣會踏入危險之地的人必備的道具！";
	next;
	mes "[心之商人]";
	mes "如果你對它感興趣，記得再回來兌換喔，好嗎？";
	close;
	
OnTouch:
	if(countitem(12209))
		npctalk "你有很好的東西呢。想把你的 " + getitemname(12209) + " 換成更好的嗎？", "", bc_self;
	end;
}
