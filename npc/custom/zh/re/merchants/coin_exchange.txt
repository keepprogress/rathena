malangdo,220,167,5	script	硬幣兌換機CX-1::Coin Exchanger CX-1	564,{
	if (checkweight(1201,1) == 0) {
		mes "你的道具種類太多了。請減少一些後再繼續吧。";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "你攜帶的物品太重，無法繼續。請減輕重量後再試一次吧。";
		close;
	}
	mes "[硬幣兌換機CX-1]";
	mes "歡迎使用 CX-1。如果你有硬幣，可以將它們兌換成高等級或低等級的硬幣。";
	next;
	setarray .@coins[1],
		6417, //Silvervine
		6418, //Agrade_Coin
		6419, //Bgrade_Coin
		6420, //Cgrade_Coin
		6421, //Dgrade_Coin
		6422; //Egrade_Coin
	.@menu$ = "Stop:";
	for(.@i = 1; .@i<=6; .@i++) {
		if (countitem(.@coins[.@i]))
			.@menu$ = .@menu$ + getitemname(.@coins[.@i]) + ":";
		else
			.@menu$ = .@menu$ + "^aaaaaa"+getitemname(.@coins[.@i])+" (None)^000000:";
	}
	.@i = select(.@menu$);
	switch(.@i) {
	case 1:
		mes "[硬幣兌換機CX-1]";
		mes "感謝光臨。";
		close;
	default:
		.@coin = .@coins[.@i-1];
		.@coin_select = .@i-1;
		break;
	}
	if (countitem(.@coin) == 0) {
		mes "[硬幣兌換機CX-1]";
		mes "你選擇了一種你沒有的硬幣。";
		mes "感謝光臨。";
		close;
	}
	mes "[硬幣兌換機CX-1]";
	mes "----兌換比率----";
	mes "高級兌換 - 30 換 10";
	mes "低級兌換 - 10 換 30";
	mes "------------------------";
	mes "^ff3333兌換時，我們會依當前市價收取手續費。^000000";
	next;
	setarray .@exchange_name$[0],"higher","lower";
	setarray .@exchange_rate[0],30,10;
	setarray .@exchange_loss[0],rand(1,4),rand(1,2); // Amount deducted per exchange.
	setarray .@exchange_id[0],.@coins[.@coin_select-1],.@coins[.@coin_select+1];
	if (.@exchange_id[0] == .@coins[1]) .@exchange_id[0] = 0; // Cannot exchange for Silvervine.
	.@menu$ = "Stop:";
	for(.@i = 0; .@i<2; .@i++) {
		if (.@exchange_id[.@i] == 0)
			.@menu$ = .@menu$ + "^ff3333Unavailable exchange to "+.@exchange_name$[.@i]+" coin^000000:";
		else if (countitem(.@coin) < .@exchange_rate[.@i])
			.@menu$ = .@menu$ + "^aaaaaaExchange to "+.@exchange_name$[.@i]+" coin (null)^000000:";
		else
			.@menu$ = .@menu$ + "Exchange to "+.@exchange_name$[.@i]+" coin - "+getitemname(.@coin)+" ("+.@exchange_rate[.@i]+" needed):";
	}
	.@i = select(.@menu$)-2;
	if (.@i == -1) {
		mes "[硬幣兌換機CX-1]";
		mes "感謝光臨。";
		close;
	}
	if (.@exchange_id[.@i] == 0) {
		mes "[硬幣兌換機CX-1]";
		mes "這種硬幣無法兌換成 "+.@exchange_name$[.@i]+" 硬幣。";
		mes "感謝光臨。";
		close;
	}
	if (countitem(.@coin) < .@exchange_rate[.@i]) {
		mes "[硬幣兌換機CX-1]";
		mes "你的硬幣不足以兌換成 "+.@exchange_name$[.@i]+" 硬幣。";
		mes "感謝光臨。";
		close;
	}
	.@exchange_total = .@exchange_rate[(!.@i)] - .@exchange_loss[(!.@i)];
	delitem .@coin, .@exchange_rate[.@i];
	getitem .@exchange_id[.@i], .@exchange_total;
	mes "[硬幣兌換機CX-1]";
	mes "^003399"+.@exchange_rate[.@i]+"^000000 個 ^003399 "+getitemname(.@coin)+"^000000 已取走。";
	mes "我已將它們兌換成 ^003399"+.@exchange_name$[.@i]+" 個硬幣^000000，扣除 ^003399"+.@exchange_loss[(!.@i)]+"^000000 個後，^003399"+.@exchange_total+"^000000 個剩餘。";
	close;
}
