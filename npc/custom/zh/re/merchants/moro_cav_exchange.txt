moro_cav,32,73,5	script	英雄勳章兌換員#1::Hero Token Exchanger#1	4_F_SHADOWCHASER,{
	disable_items;
	mes "[英雄勳章兌換員]";
	mes "若你曾與摩洛克交戰";
	mes "並收集了";
	mes "英雄勳章，";
	mes "我可以用特製戒指為你兌換。";
	mes "想看看我的戒指嗎？";
	while(true) {
		next;
		if (checkweight(2163,2) == 0) {
			mes "[英雄勳章兌換員]";
			mes "你的背包裡道具很多，";
			mes "整理完再來找我吧！";
			close;
		}
		switch( select( "戒指說明", "用戒指兌換", "結束對話。" ) ) {
		case 1:
			mes "[英雄勳章兌換員]";
			mes "這枚英雄戒指";
			mes "或許不";
			mes "看起來";
			mes "不那麼出眾，";
			next;
			mes "[英雄勳章兌換員]";
			mes "但你可以把它拿去";
			mes "附魔專家拜恩斯，";
			mes "他能為它附魔，";
			mes "將其中四個孔位全部";
			mes "打通，並激發出";
			mes "其潛藏的能力。";
			next;
			mes "[英雄勳章兌換員]";
			mes "英雄戒指";
			mes "能助你";
			mes "討伐";
			mes "摩洛克。";
			mes "還有其他問題嗎？";
			continue;
		case 2:
			mes "- 英雄戒指 -";
			mes "^006400為戰士打造的戒指^000000";
			mes "^006400為了擊敗摩洛克的使命……^000000";
			mes "^006400附魔之後，^000000";
			mes "^006400它將賦予主人^000000";
			mes "^006400對抗摩洛克的力量。^000000";
			mes "^006400職業: 飾品/防禦: 0^000000";
			mes "^006400重量: 20/需求等級: 160^000000";
			mes "^006400卡槽: 0/裝備職業: 全職業^000000";
			next;
			mes "[英雄勳章兌換員]";
			mes "英雄戒指需要";
			mes "30 個英雄勳章";
			mes "來兌換。";
			mes "你要交換嗎？";
			next;
			if (select( "上一個選單", "兌換 30 個英雄勳章" ) == 1) {
				mes "[英雄勳章兌換員]";
				mes "我再考慮";
				mes "一下。";
				continue;
			}
			if (countitem(6684) < 30) {
				mes "[英雄勳章兌換員]";
				mes "^006400英雄勳章^000000";
				mes "是頒給";
				mes "勇敢對抗摩洛克的人。";
				mes "你沒有";
				mes "30 個^006400英雄勳章^000000，";
				mes "所以我無法幫你。";
				continue;
			}
			mes "[英雄勳章兌換員]";
			mes "我會把";
			mes "30 個^006400英雄勳章^000000";
			mes "兌換成 1 個^006400英雄戒指^000000。";
			mes "還有其他事嗎？";
			delitem 6684,30;	// TokenOfHero
			getitem 2981,1;		// RingOfHero
			continue;
		case 3:
			mes "[英雄勳章兌換員]";
			mes "總有一天";
			mes "你會發現自己";
			mes "需要英雄戒指。";
			close;
		}
	}
}

moro_cav,34,65,5	script	附魔專家拜恩斯::Expert Enchanter Byrnes	1_M_03,{
	disable_items;
	mes "[拜恩斯]";
	mes "你聽說過英雄戒指嗎？只有真正的英雄才配擁有那枚華麗的戒指，而我一直在等待這樣的人。";
	mes "有什麼我能為你做的嗎？";
	next;
	switch( select( "為英雄戒指附魔", "重置英雄戒指的附魔", "取消。" ) ) {
	case 1:
		mes "[拜恩斯]";
		mes "我只能為穿戴中的戒指附魔。你想為哪一枚穿戴中的戒指附魔？";
		next;
		setarray .@eqi_slot[0], EQI_ACC_L, EQI_ACC_R;
		.@s = select( "左邊", "右邊" ) - 1;
		switch( getequipid(.@eqi_slot[.@s]) ) {
		case -1:
			mes "[拜恩斯]";
			mes "請穿戴你想附魔的英雄戒指。";
			close;
		case 2981:
			break;
		default:
			mes "[拜恩斯]";
			mes "請穿戴你的英雄戒指，其他東西我無法附魔。";
			close;
		}
		// custom at this point
		setarray .@card[0],
			getequipcardid(.@eqi_slot[.@s],0),
			getequipcardid(.@eqi_slot[.@s],1),
			getequipcardid(.@eqi_slot[.@s],2),
			getequipcardid(.@eqi_slot[.@s],3);

		if (.@card[0] > 0) {
			mes "[拜恩斯]";
			mes "已完全附魔的英雄戒指無法再附魔。";
			close;
		}
		if (.@card[3] == 0) {
			setarray .@enchant_list$[0], "DEF", "MDEF";
			.@slot = 4;
		}
		else if (.@card[2] == 0) {
			setarray .@enchant_list$[0], "Max HP", "Max SP";
			.@slot = 3;
		}
		else if (.@card[1] == 0) {
			setarray .@enchant_list$[0], "Atk", "MAtk";
			.@slot = 2;
		}
		else {
			setarray .@enchant_list$[0], "STR", "INT", "AGI", "VIT", "DEX", "LUK";
			.@slot = 1;
		}
		mes "[拜恩斯]";
		mes "你想在第 " + .@slot + "?";
		next;
		.@enchant_list_size = getarraysize(.@enchant_list$);
		for ( .@i = 0; .@i < .@enchant_list_size; ++.@i )
			.@menu$ += .@enchant_list$[.@i] + ":";
		.@type = select(.@menu$) - 1;

		switch(.@slot) {
		case 4:
			switch(.@type) {// enchant list index
			case 0:	// DEF
				setarray .@bonus[0], 4791, 4792, 4793, 4794, 4893;
				break;
			default:	// MDEF
				setarray .@bonus[0], 4890, 4786, 4891, 4787, 4892;
				break;
			}
			break;
		case 3:
			switch(.@type) {
			case 0:	// Max HP
				setarray .@bonus[0], 4861, 4862, 4867, 4868, 4900;
				break;
			default:	// Max SP
				setarray .@bonus[0], 4800, 4870, 4871, 4801, 4802;
				break;
			}
			break;
		case 2:
			switch(.@type) {
			case 0:	// Atk
				setarray .@bonus[0], 4882, 4766, 4767, 4894, 4895;
				break;
			default:	// Matk
				setarray .@bonus[0], 4883, 4896, 4897, 4898, 4899;
				break;
			}
			break;
		case 1:
			switch(.@type) {
			case 0:	// STR
				setarray .@bonus[0], 4700, 4701, 4702, 4703, 4704;
				break;
			case 1:	// INT
				setarray .@bonus[0], 4710, 4711, 4712, 4713, 4714; 
				break;
			case 2:	// AGI
				setarray .@bonus[0], 4730, 4731, 4732, 4733, 4734;
				break;
			case 3:	// VIT
				setarray .@bonus[0], 4740, 4741, 4742, 4743, 4744;
				break;
			case 4:	// DEX
				setarray .@bonus[0], 4720, 4721, 4722, 4723, 4724;
				break;
			default:	// LUK
				setarray .@bonus[0], 4750, 4751, 4752, 4753, 4754;
				break;
			}
			break;
		}

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@eqi_slot[.@s], 2981) || callfunc("F_IsEquipCardHack", .@eqi_slot[.@s], .@card[0], .@card[1], .@card[2], .@card[3]))
			close;

		.@bonus_size = getarraysize(.@bonus);
		.@card[.@slot-1] = .@bonus[ rand(.@bonus_size) ];	// unknown rates
		specialeffect2 EF_ANGEL2;
		mes "[拜恩斯]";
		mes "附魔完成！";
		delequip .@eqi_slot[.@s];
		getitem2 2981,1,1,0,0,.@card[0],.@card[1],.@card[2],.@card[3];
		close;
	case 2:
		mes "[拜恩斯]";
		mes "要重置你英雄戒指的附魔，我需要 3 個英雄勳章。";
		next;
		if (countitem(6684) < 3) {
			mes "[拜恩斯]";
			mes "您沒有足夠的英雄勳章，我無法重置你的英雄戒指。";
			close;
		}
		mes "[拜恩斯]";
		mes "我只能重置穿戴中的戒指。你想重置哪一枚穿戴中的戒指？";
		next;
		setarray .@eqi_slot[0], EQI_ACC_L, EQI_ACC_R;
		.@s = select( "左邊", "右邊" ) - 1;
		switch( getequipid(.@eqi_slot[.@s]) ) {
		case -1:
			mes "[拜恩斯]";
			mes "你必須先穿戴該道具，才能重置它的附魔狀態。";
			close;
		case 2981:
			break;
		default:
			mes "[拜恩斯]";
			mes "請穿戴你的英雄戒指，其他東西我無法重置。";
			close;
		}
		// custom at this point
		if (countitem(6684) < 3) {
			mes "[拜恩斯]";
			mes "您沒有足夠的英雄勳章，我無法重置你的英雄戒指。";
			close;
		}
		setarray .@card[0],
			getequipcardid(.@eqi_slot[.@s],0),
			getequipcardid(.@eqi_slot[.@s],1),
			getequipcardid(.@eqi_slot[.@s],2),
			getequipcardid(.@eqi_slot[.@s],3);

		if (.@card[3] == 0) {
			mes "[拜恩斯]";
			mes "你的英雄戒指目前沒有附魔。";
			close;
		}

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@eqi_slot[.@s], 2981) || callfunc("F_IsEquipCardHack", .@eqi_slot[.@s], .@card[0], .@card[1], .@card[2], .@card[3]))
			close;

		specialeffect2 EF_ANGEL2;
		mes "[拜恩斯]";
		mes "初始化完成。";
		delitem 6684,3;
		delequip .@eqi_slot[.@s];
		getitem 2981,1;
		close;
	case 3:
		close;
	}
}
