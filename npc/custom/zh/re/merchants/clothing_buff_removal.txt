//===== rAthena Script =======================================
//= Clothing Effect Removal Service
//===== Description: =========================================
//= Service to remove clothing effects like Xmas, Wedding, ...
//===== Changelog: ===========================================
//= 1.0 Initial version. [Balferian, Everade, Lemongrass]
//============================================================

prontera,80,106,5	script	服裝效果解除::Clothing Effect Removal Service	4_EP16_COOK2,{
	mes "[德瑞克射擊]";
	mes "你是不是因為服裝增益而無法參與戰鬥？真是讓人困擾吧。";
	next;
	mes "[德瑞克射擊]";
	mes "雖然穿在你身上很好看，但也真的很麻煩。我只要一個手勢就能幫你解除！！我會把它整理得妥妥當當。準備好了的話，請選擇你的選項！";
	next;
	switch( select( "保留它", "立刻解除" ) ){
		case 1:
			mes "[德瑞克射擊]";
			mes "如果你改變主意，再回來找我。";
			close;
		case 2:
			sleep2 500;
			specialeffect2 EF_SPHERE;
			sleep2 500;
			specialeffect2 EF_SPHERE;
			sleep2 500;
			specialeffect2 EF_SPHERE;
			mes "[德瑞克射擊]";
			mes "解除完成了。請確認你現在可以自由行動。";
			atcommand "@changedress";
			close;
	}
}
