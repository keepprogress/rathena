		getequipcardid(EQI_SHOES,1),

		.@amount_coag = .@en_amount_coag[0];
		.@amount_poll = .@en_amount_polluted[0];

		mes "[黑暗魔法大師]";
		mes "想要附魔^0000ff" + .@equip_name$ + " (插槽)^000000？第一次附魔需要 ^0000ff" + .@amount_poll + " " + getitemname(.@polluted_id) + ", " + .@amount_coag + " " + getitemname(.@coagulated_id) + "，以及 " + callfunc("F_InsertComma",.@cost_zeny) + " Zeny^000000。";
		mes "有一定的失敗風險……";
		next;
		.@s = select( "離開", "鬥志", "弓術", "魔力", "活力", "攻擊速度", "幸運" ) - 2;

			mes "[黑暗魔法大師]";
			mes "好的，準備好再來吧。";
			close;
		}
		.@card[3] = .@enchant_1[.@s];
		.@string$ = "enchant number ^6300001^000000.";
		.@num = 0;
	}
	else {
		for ( .@num = 1; .@num < 5; .@num++ ) {
			for ( .@type = 0; .@type < 6 && .@card[3] != getd( ".@enchant_" + .@num + "[" + .@type + "]" ); .@type++ )
				continue;
			if (.@type < 6)
				break;
		}
		if (.@num == 5) {
			mes "[黑暗魔法大師]";
			mes "出了點問題。";
			close;
		}
		.@amount_coag = .@en_amount_coag[.@num];
		.@amount_poll = .@en_amount_polluted[.@num];

		mes "[黑暗魔法大師]";
		if (.@num == 4) {
			.@card[2] = callfunc("F_Rand",4875,4876,4877,4878,4879,4880);// Bear's_Power, Runaway_Magic, Speed_Of_Light, Muscle_Fool, Hawkeye, Lucky_Day
			.@string$ = "^990000 Bonus effect ^000000 upgrade.";

		}
		else {
			.@level = .@num + 1;
			.@card[3] = getd( ".@enchant_" + (.@num+1) + "[" + .@type + "]" );
			.@string$ = "enchant number ^990000" + .@level + "^000000.";
			mes "附魔 ^0000ff" + .@equip_name$ + " (插槽)^000000的第 4 格插槽為 ^0000ff" + .@level + "^000000等級效果，需要 ^0000ff" + .@amount_poll + " " + getitemname(.@polluted_id) + ", " + .@amount_coag + " " + getitemname(.@coagulated_id) + "，以及 " + callfunc("F_InsertComma",.@cost_zeny) + " z^000000。有一定的失敗風險……";
		}
		next;
		if (select( "離開", "效果升級！" ) == 1) {
			mes "[黑暗魔法大師]";
			mes "好的，準備好再來吧。";
			close;
		}
	}
	if (countitem(.@coagulated_id) < .@amount_coag) {
		mes "[黑暗魔法大師]";

	if (countitem(.@polluted_id) < .@amount_poll) {
		mes "[黑暗魔法大師]";

	}
	if (Zeny < .@cost_zeny) {
		mes "[黑暗魔法大師]";
		mes "你必須帶來 ^0000ff" + callfunc("F_InsertComma",.@cost_zeny) + " zeny^000000。天下沒有白吃的午餐喔……";
		close;
	}
	delitem .@coagulated_id, .@amount_coag;
	delitem .@polluted_id, .@amount_poll;
	Zeny -= .@cost_zeny;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", EQI_SHOES, .@equip_id) || callfunc("F_IsEquipCardHack", EQI_SHOES, .@equip_card[0], .@equip_card[1], .@equip_card[2], .@equip_card[3]) || callfunc("F_IsEquipRefineHack", EQI_SHOES, .@equip_refine))
		close;

	delequip EQI_SHOES;
	if (.@chances[.@num] < rand(1,100)) {
		specialeffect2 EF_LORD;
		mes "[黑暗魔法大師]";
		mes "啊啊，失敗了。下次好運吧。";
		close;
	}
	specialeffect2 EF_REPAIRWEAPON;
	mes "[胡金的魔法師]";
	mes "正在嘗試 " + .@string$;
	getitem2 .@equip_id,1,1,.@equip_refine,0,0,0,.@card[2],.@card[3];
	close;
}
