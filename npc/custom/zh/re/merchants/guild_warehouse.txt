-	script	GuildWarehouse	-1,{
	mes "[倉庫管理員]";
	mes "你好嗎？我們專門經營公會倉庫。這是我們的^0000cd故事^000000:";
	mes "為什麼公會成員不能共用一個倉庫？我們就是從這個簡單的問題開始的。";
	next;
	mes "[倉庫管理員]";
	mes "你可以支付 1000 Zeny 開啟倉庫，或查看公會的使用紀錄。";
	mes "有什麼我能幫你的嗎？";
	next;
	if( select( "開啟公會倉庫:查看倉庫使用紀錄" ) == 1 ){
		.@guildid = getcharid( 2 );

		if( .@guildid == 0 ){
			mes "[倉庫管理員]";
			mes "只有公會成員才能使用公會倉庫。";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "[倉庫管理員]";
			mes "看來你的公會目前還無法使用公會倉庫，或是你沒有倉庫的存取權限。";
			mes "請在確認公會技能與倉庫存取權限後再來。";
			close;
		}

		if( Zeny < 1000 ){
			mes "[倉庫管理員]";
			mes "很抱歉，你的金錢不足以支付手續費。";
			mes "手續費為 1000 Zeny。";
			close;
		}

		mes "[倉庫管理員]";
		mes "那我就為你開啟公會倉庫。祝你有段難忘的時光！";
		close2;

		if( Zeny < 1000 ){
			// Cheat prevention
			end;
		}

		if( guildopenstorage() == GSTORAGE_OPEN ){
			Zeny -= 1000;
			end;
		}else{
			mes "[倉庫管理員]";
			mes "很抱歉，現在有其他人正在使用倉庫。";
			mes "請稍後再試一次。";
			close;
		}
	}else{
		.@guildid = getcharid( 2 );

		if( .@guildid == 0 ){
			mes "[倉庫管理員]";
			mes "只有公會成員才能使用公會倉庫。";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "[倉庫管理員]";
			mes "看來你的公會目前還無法使用公會倉庫，或是你沒有倉庫的存取權限。";
			mes "請在確認公會技能與倉庫存取權限後再來。";
			close;
		}

		mes "[倉庫管理員]";
		mes "我來為你顯示倉庫的使用紀錄。使用紀錄最多保留 3 個月。";
		mes "祝你玩得愉快！";
		close2;
		guildopenstorage_log();
		end;
	}
}
