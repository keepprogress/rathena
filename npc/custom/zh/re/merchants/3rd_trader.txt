job3_guil01,91,93,3	script	稀有草藥收集者::Rare Herb Collector	1_M_03,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
		mes "- 等一下！！ -";
		mes "- 你目前攜帶的 -";
		mes "- 道具太多了。 -";
		mes "- 請再試一次 -";
		mes "- 等你減輕一些負重後。 -";
		close;
	}
	if (Class == Job_Guillotine_Cross || Class == Job_Guillotine_Cross_T || Class == Job_Baby_Guillotine_Cross) {
		mes "[稀有毒草收集者]";
		mes "我走遍世界收集稀有的毒草。最近我開始經手一種名叫伊茲多爾的草藥。如果你有興趣，可以向我購買。";
		next;
		.@i = select("要怎麼買？:用動物之血交換:用苦味草藥交換:用致命毒草交換:用冰凍玫瑰交換:用門特交換:用希娜爾交換")-2;
		if (.@i == -1) {
			mes "[稀有毒草收集者]";
			mes "只要 10,000 Zeny 再加上其中一種稀有道具，就能買到：動物之血、苦味草藥、致命毒草、冰凍玫瑰、門特或希娜爾。";
			next;
			mes "[稀有毒草收集者]";
			mes "你問我為什麼要收錢？那只是對像我這樣的收集者表示一點敬意罷了。哈哈哈……";
			close;
		} else {
			setarray .@exchange[0],702,621,631,749,605,703;
			.@item = .@exchange[.@i];
			.@price = 10000;

			mes "[稀有毒草收集者]";
			mes "你要多少？";
			mes "最多只能購買 '2000' 個。";
			mes "如果要取消，請輸入 '0'。";
			next;
			input .@amount;
			if (.@amount == 0) {
				mes "[稀有毒草收集者]";
				mes "你已取消交易。";
				close;
			}
			if (.@amount > 2000) {
				mes "[稀有毒草收集者]";
				mes "數量必須少於 2,000！";
				close;
			}
			.@total = .@amount * .@price;
			if (countitem(.@item) < .@amount || Zeny < .@total) {
				mes "[稀有毒草收集者]";
				mes "喂，等你備齊交換所需的所有條件再回來吧。";
				close;
			}
			mes "[稀有毒草收集者]";
			mes "很好。我已經收到錢和特殊道具了。";
			delitem .@item, .@amount;
			Zeny = Zeny - .@total;
			getitem 709, .@amount; //Izidor
			close;
		}
	}
	mes "[稀有毒草收集者]";
	mes "我走遍世界收集稀有毒草。不過我可不想把草藥賣給你這種人……哈哈哈……";
	close;
}
