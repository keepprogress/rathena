	mes "歡迎。";
	mes "我來這裡販售";
	mes "^0000FF基因學者專用製作書^000000。";
	next;
	mes "[製作書商人]";
	mes "基因學者擁有一些技能";
	mes "需要製作書才能啟動。";
	next;
	mes "[製作書商人]";
	mes "你擁有的製作書越多，";
	mes "你能使用的技能就越多。";
	next;
	mes "[製作書商人]";
	mes "請理解，";
	mes "我專門販售這些特殊書籍，而且";
	mes "我 ^FF0000不販售其他道具^000000。";
	next;
	mes "[製作書商人]";
	mes "要看看我的製作書嗎？";
	next;
	if(select("好啊。:不要。") == 2) {
		mes "[製作書商人]";
		mes "感謝光臨我的商店。";
		mes "旅途請小心平安。";
		close;
	}
	if (checkweight(1201,1) == 0) {
		mes "[製作書商人]";
		mes "很抱歉，你的背包快滿了。";
		mes "請先清空你的背包。";
		close;
	}
	if (MaxWeight - Weight < 2500) {
		mes "[製作書商人]";
		mes "請先清空你的背包。";
		close;
	}
	mes "[製作書商人]";
	mes "好的，這是可購買的製作書清單。";
	next;
	switch(select("[蘋果炸彈製作書] 100,000 Zeny:[鳳梨炸彈製作書] 100,000 Zeny:[椰子炸彈製作書] 100,000 Zeny:[哈密瓜炸彈製作書] 100,000 Zeny:[香蕉炸彈製作書] 100,000 Zeny:[植物基因培育法] 100,000 Zeny:[高級藥水製作手冊] 100,000 Zeny:[混合料理書] 100,000 Zeny:[體力提升研究書] 100,000 Zeny:[活力飲料配方] 100,000 Zeny:關閉")){
	case 1: set .@item,6279; break;
	case 2: set .@item,6280; break;
	case 3: set .@item,6281; break;
	case 4: set .@item,6282; break;
	case 5: set .@item,6283; break;
	case 6: set .@item,6284; break;
	case 7: set .@item,6285; break;
	case 8: set .@item,11022; break;
	case 9: set .@item,11023; break;
	case 10: set .@item,11024; break;
	case 11:
		mes "[製作書商人]";
		mes "感謝您的惠顧。";
		mes "歡迎再次光臨。";
		close;
	}
	mes "[製作書商人]";
	mes "^0000FF["+getitemname(.@item)+"] 的價格為";
	mes "100,000 Zeny。";
	mes "你想購買多少個？";
	next;
	input .@amount;
	if (.@amount == 0) {
		mes "[製作書商人]";
		mes "要看看其他的書嗎？";
		close;
	}
	if (.@amount < 1 || .@amount > 99 ){
		mes "[製作書商人]";
		mes "一次最多只能購買 100 個。";
		close;
	}
	mes "[製作書商人]";
	mes "你輸入了 "+.@amount+"個 ^0000FF["+getitemname(.@item)+"]^000000 要購買。";
	mes "您要繼續嗎？";
	next;
	if (select("是:否") == 1) {
		set .@total, .@amount * 100000;
		if (Zeny < .@total) {
			mes "[製作書商人]";
			mes "很抱歉，你的錢不夠。";
			close;
		}
		if (checkweight(.@item,.@amount) == 0){
			mes "[製作書商人]";
			mes "看來你無法全部攜帶。";
			mes "請檢查你背包的空間。";
			close;
		}
		mes "[製作書商人]";
		mes "感謝您的惠顧。";
		set Zeny,Zeny - .@total;
		getitem .@item,.@amount;
		close;
	}
	mes "[製作書商人]";
	mes "請慢慢考慮，";
	mes "再做決定。";
	close;
}
