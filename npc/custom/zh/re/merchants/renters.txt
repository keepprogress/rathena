job3_rune01,88,62,5	script	飛龍飼育員::Dragon Breeder	105,{
	mes "[飛龍飼育員]";
	if ((eaclass()&EAJ_THIRDMASK)==EAJ_RUNE_KNIGHT) {
		mes "歡迎光臨。要租用飛龍嗎？";
		next;
		switch(select("是:否")) {
		case 1:
			//if (!getskilllv("RK_DRAGONTRAINING")) {
			if (!getskilllv("KN_RIDING")) {
				mes "[飛龍飼育員]";
				mes "請先學會騎乘飛龍。";
				close;
			}
			else if (checkdragon()) {
				mes "[飛龍飼育員]";
				mes "你已經有飛龍了。";
				close;
			}
			else if (ismounting()) {
				mes "[飛龍飼育員]";
				mes "請先卸下你的點券坐騎。";
				close;
			}
			setdragon;
			close;
		case 2:
			mes "[飛龍飼育員]";
			mes "我知道了。那祝你有愉快的一天。";
			close;
		}
	}
	mes "你在這裡做什麼？";
	mes "只有符文騎士能租用飛龍。";
	close;
}

// Mado Gear Renter :: madogear
//============================================================
-	script	::mgm	-1,{
	mes "[魔導機甲裝配師]";
	if ((eaclass()&EAJ_THIRDMASK)==EAJ_MECHANIC) {
		mes "歡迎光臨，機械師。";
		mes "要租用手推車還是";
		mes "騎乘魔導機甲？";
		next;
		switch(select("租用手推車:騎乘魔導機甲:購買緊急魔導機甲:升級冷卻器:取消")) {
		case 1:
			if (checkcart()) {
				mes "[魔導機甲裝配師]";
				mes "很抱歉，你已經";
				mes "有手推車了。";
				close;
			}
			setcart;
			mes "[魔導機甲裝配師]";
			mes "給你！";
			close;
		case 2:
			if (checkmadogear()) {
				mes "[魔導機甲裝配師]";
				mes "很抱歉，你已經";
				mes "正騎著魔導機甲。";
				close;
			}
			else if (!getskilllv("NC_MADOLICENCE")) {
				mes "[魔導機甲裝配師]";
				mes "請先學會取得魔導機甲執照的技能。";
				close;
			}
			else if(ismounting()) {
				mes "[魔導機甲裝配師]";
				mes "請先卸下你的點券坐騎。";
				close;
			}
			setmadogear;
			mes "[魔導機甲裝配師]";
			mes "玩得愉快，歡迎再次光臨！";
			close;
		case 3:
			mes "[魔導機甲裝配師]";
			mes "緊急魔導機甲在緊急情況下非常實用，售價為 1,000,000 Zeny。";
			next;
			if (select("購買:取消") == 2) {
				mes "[魔導機甲裝配師]";
				mes "我知道了。有需要時請隨時來找我";
				mes "請再回來找我。";
				close;
			}
			if (countitem(23277) > 0) {
				mes "[魔導機甲裝配師]";
				mes "很抱歉，你已經有緊急魔導機甲了。";
				close;
			}
			if (Zeny < 1000000) {
				mes "[魔導機甲裝配師]";
				mes "很抱歉，你的 Zeny 不足以購買緊急魔導機甲。";
				close;
			}
			Zeny -= 1000000;
			getitem 23277,1; //Mado_Box
			mes "[魔導機甲裝配師]";
			mes "給你！";
			close;
		case 4:
			mes "[魔導機甲裝配師]";
			mes "你想升級哪個裝置？";
			next;
			if (select("冷卻器:高級冷卻器") == 1) {
				mes "[魔導機甲裝配師]";
				mes "將冷卻器升級為高級冷卻器需要 1 個冷卻器和 2,000,000 Zeny。";
				next;
				.@itemid = 2804;
				.@cost = 2000000;
			} else {
				mes "[魔導機甲裝配師]";
				mes "將高級冷卻器升級為特殊冷卻器需要 1 個高級冷卻器和 4,000,000 Zeny。";
				next;
				.@itemid = 2809;
				.@cost = 4000000;
			}
			if (select("升級:取消") == 2) {
				mes "[魔導機甲裝配師]";
				mes "我知道了。有需要時請隨時來找我";
				mes "請再回來找我。";
				close;
			}
			if (!countitem(.@itemid)) {
				mes "[魔導機甲裝配師]";
				mes "很抱歉，你沒有 " + getitemname(.@itemid) + ".";
				close;
			}
			if (Zeny < .@cost) {
				mes "[魔導機甲裝配師]";
				mes "很抱歉，你的 Zeny 不足以升級這個裝置。";
				close;
			}
			Zeny -= .@cost;
			delitem .@itemid,1;
			getitem (.@itemid == 2804 ? 2809 : 2810),1; //High_Quality_Cooler,Special_Cooler
			mes "[魔導機甲裝配師]";
			mes "給你！你專屬的 " + getitemname(.@itemid) + ".";
			close;
		case 5:
			close;
		}
	}
	mes "有什麼可以為您效勞的嗎？";
	mes "魔導機甲只提供給機械師。";
	close;
}
