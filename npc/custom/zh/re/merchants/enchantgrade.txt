grademk,34,184,4	script	斯拉托斯#sratos::Sratos#sratos	4_JP_GARM_H,{
	mes "[斯拉托斯]";
	mes "歡迎光臨，親愛的顧客，今天就來期待奇蹟吧。";
	mes "有什麼我能幫你的嗎？";
	next;
	switch( select( "強化裝備的等階。", "兌換乙太道具", "什麼是等階強化？" ) ){
		case 1:
			mes "[斯拉托斯]";
			mes "我們的顧客。你想要強化裝備的等階。";
			mes "要轉移寶石的魔力可不容易。";
			next;
			mes "[斯拉托斯]";
			mes "魔力可能會當場爆炸。";
			mes "那麼要繼承魔力的武器也會一起爆炸！";
			next;
			switch( select( "我還是要做！", "我再考慮一下。" ) ){
				case 1:
					mes "[斯拉托斯]";
					mes "祝勇敢的顧客們好運！";
					close2;
					enchantgradeui();
					end;
				case 2:
					mes "[斯拉托斯]";
					mes "等你有勇氣挑戰時，再回來吧。";
					close;
			}
		case 2:
			mes "[斯拉托斯]";
			mes "這些是用來提升等階的寶石。";
			mes "乙太星塵是製作的基底，只要你付一點小錢，我就能把它兌換成乙太寶石。";
			close2;
			callshop "EnchantGradeExchange";
			end;

		case 3:
			mes "[斯拉托斯]";
			mes "偶爾會發現蘊含大量純粹魔力、但很不穩定的寶石。";
			mes "我們把這些叫做乙太寶石，對吧？";
			next;
			mes "[斯拉托斯]";
			mes "你可以把乙太寶石的魔力轉移到其他武器上，但當然有一定風險。";
			mes "它很脆弱，不小心的話，砰！就會爆炸。";
			next;
			mes "[斯拉托斯]";
			mes "總之，使用這顆乙太寶石，就能強化你正在用的武器。";
			mes "效能會提升。我們把這稱為等階強化。";
			next;
			mes "[斯拉托斯]";
			mes "你也一樣吧？如果你心愛的武器能再進一步成長，不是很開心嗎？";
			mes "我們是負責協助的人。雖然有點危險，哈哈哈。";
			close;
	}

OnInit:
	setunittitle(getnpcid(0), "<Grade Enhancer>");
	end;
}
