//===== rAthena Script =======================================
//= HD Refiners
//===== Description: =========================================
//= [Official Conversion]
//= Refiners that use HD ores to refine equipment. Upon
//= failure, the equipment is not destroyed; rather, its
//= refine level decreases by 1. The success rate is identical
//= to that for Enriched ores.
//= - "Blacksmith Mighty Hammer" only refines from +7~9.
//= - "Basta" only refines from +10 and up.
//===== Changelog: ===========================================
//= 1.0 First version. [Euphy]
//= 1.1 Removed re-roll behavior. [Secret]
//============================================================

// Blacksmith Mighty Hammer (+7~9) :: cash_smelting79
//============================================================
-	script	::MightyHammer	-1,{
	disable_items;
	mes "[鐵匠大鐵槌]";
	mes "和其他人不同，我是只精煉極少數物品的鐵匠。";
	mes "我只精煉 ^CC0000+7 至 +9^000000 的物品。";
	next;
	mes "[鐵匠大鐵槌]";
	mes "我的特色是，就算精煉失敗，也只會降低 1 級精煉等級，裝備不會消失。很棒吧？";
	next;
	mes "[鐵匠大鐵槌]";
	mes "那就全力以赴吧，怎麼樣？你想精煉什麼物品？";
	next;
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for ( .@i = 1; .@i <= 10; ++.@i )
		.@menu$ = .@menu$ + ( getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) + "-[Not equipped]" ) + ":";
	.@part = .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[鐵匠大鐵槌]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "我是鐵匠，不是髮型師。";
			break;
		case EQI_ARMOR:
			mes "用我的鐵鎚，我會讓你成為天空中最閃耀的星星。";
			break;
		case EQI_HAND_L:
		case EQI_HAND_R:
			mes "製作義手不是我的專長。";
			break;
		case EQI_GARMENT:
			mes "把物品拿出來，讓我精煉它！";
			break;
		case EQI_SHOES:
			mes "這股腳臭是哪來的？";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "飾品在哪裡？";
			break;
		case EQI_HEAD_MID:
			mes "你想讓我精煉什麼？";
			break;
		case EQI_HEAD_LOW:
			mes "嗯？你想讓我做什麼？";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[鐵匠大鐵槌]";
		mes "這個物品無法精煉。";
		close;
	}
	switch( getequiprefinerycnt(.@part) ) {
	case 7:
		.@blacksmith_blessing_count = 1;
		break;
	case 8:
		.@blacksmith_blessing_count = 2;
		break;
	case 9:
		.@blacksmith_blessing_count = 4;
		break;
	default:
		mes "[鐵匠大鐵槌]";
		mes "我只處理精煉等級 +7 至 +9 的物品。";
		close;
	}

	.@refineitemid = getequipid(.@part); // save id of the item
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_MATERIAL_ID);

	mes "[鐵匠大鐵槌]";
	mes "要精煉您選的裝備，您需要 ^ff9999" + getitemname(.@material) + "^000000 以及 20,000 Zeny 作為手續費。";
	mes "東西都準備好了嗎？";
	next;
	if (select("是:否") == 2) {
		mes "[鐵匠大鐵槌]";
		mes "你準備好之前我會等的。";
		close;
	}
	if (getequippercentrefinery(.@part) < 100) {
		mes "[鐵匠大鐵槌]";
		mes "看來這個物品很可能會精煉失敗。";
		mes "嗯，就算失敗，也只會降低 1 級精煉等級。";
		mes "要繼續精煉嗎？";
		next;
		if (countitem(6635) < .@blacksmith_blessing_count)
			setarray .@menu$[0], "", "Yes", "Not yet";
		else {
			mes "[鐵匠大鐵槌]";
			mes "啊！是 ^0000ff鐵匠的祝福^000000 嗎？";
			mes "有了鐵匠的祝福，精煉失敗時裝備就不會消失！";
			next;
			mes "[鐵匠大鐵槌]";
			if ( .@itemtype != IT_WEAPON )
				mes "如果是 +" + getequiprefinerycnt(.@part) + " 的裝備，請用^316AC5 " + .@blacksmith_blessing_count + " 個鐵匠的祝福^000000 可以防止裝備消失。要用它來精煉嗎？";
			else
				mes "如果是 +" + getequiprefinerycnt(.@part) + " 的武器，請用^316AC5 " + .@blacksmith_blessing_count + " 個鐵匠的祝福^000000 可以防止裝備消失。要用它來精煉嗎？";
			next;
			setarray .@menu$[0], "Use it to refine", "Refine directly without it", "Don't refine yet";
		}
		switch( select(.@menu$[0], .@menu$[1], .@menu$[2]) ) {
		case 1:
			.@bless_who = 1;
			break;
		case 2:
			break;
		case 3:
			mes "[鐵匠大鐵槌]";
			mes "只有克服了對失敗的恐懼，才能得到傑作。";
			close;
		}
	}
	if ((.@bless_who && countitem(6635) < .@blacksmith_blessing_count) || countitem(.@material) == 0 || Zeny < .@price) {
		mes "[鐵匠大鐵槌]";
		mes "你不是剛說都準備好了嗎？";
		close;
	}
	if (.@bless_who)
		delitem 6635, .@blacksmith_blessing_count;
	delitem .@material,1;
	Zeny = Zeny - .@price;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
		mes "[鐵匠大鐵槌]";
		emotion ET_FRET;
		mes "等一下……";
		mes "你當我是笨蛋嗎？！";
		mes "你趁我沒注意的時候換了道具！給我滾出去！";
		close;
	}

	mes "[鐵匠大鐵槌]";
	mes "叮！叮！叮！";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		next;
		emotion ET_BEST;
		mes "[鐵匠大鐵槌]";
		mes "每次聽到這個聲音，都讓我精神一振。";
		mes "拿去吧。精煉完美成功！";
		close;
	}
	if (.@bless_who == 1) {
		specialeffect EF_HOLYHIT;
		next;
		emotion ET_HUK;
		mes "[鐵匠大鐵槌]";
		mes "什麼？！！";
		next;
		mes "[鐵匠大鐵槌]";
		mes "哎呀！我現在可丟臉了。哼哼。";
		close;
	}
	downrefitem .@part;
	next;
	emotion ET_HUK;
	mes "[鐵匠大鐵槌]";
	mes "糟糕！！";
	next;
	mes "[鐵匠大鐵槌]";
	mes "我相信像您這樣的人，不會因為精煉等級降了 1 就怪我吧。嗯。";
	close;
}

// Basta (+10 and up) :: cash_smelting
//============================================================
-	script	::Basta	-1,{
	disable_items;
	mes "[巴斯塔]";
	mes "我是全世界最棒的鐵匠，巴斯塔。";
	mes "不過我不提供一般的精煉服務。";
	mes "我只精煉 ^CC0000+10 以上^000000 的裝備。";
	next;
	mes "[巴斯塔]";
	mes "你想精煉哪件裝備？";
	next;
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for ( .@i = 1; .@i <= 10; ++.@i )
		.@menu$ = .@menu$ + ( getequipisequiped(.@indices[.@i]) ? getequipname(.@indices[.@i]) : F_getpositionname(.@indices[.@i]) + "-[Unequipped]" ) + ":";
	.@part = .@indices[ select(.@menu$) ];
	if (!getequipisequiped(.@part)) {
		mes "[巴斯塔]";
		switch(.@part) {
		case EQI_HEAD_TOP:
			mes "你的頭是裝備嗎？";
			break;
		case EQI_ARMOR:
			mes "你想讓我做什麼？";
			break;
		case EQI_HAND_L:
		case EQI_HAND_R:
			mes "製作義手不是我的專長。";
			break;
		case EQI_GARMENT:
			mes "你知道什麼是長袍嗎？";
			break;
		case EQI_SHOES:
			mes "如果你想精煉你的腳，別來找我，去跑馬拉松吧。";
			break;
		case EQI_ACC_L:
		case EQI_ACC_R:
			mes "飾品在哪裡？";
			break;
		case EQI_HEAD_MID:
			mes "嗯……我沒看到值得精煉的裝備。";
			break;
		case EQI_HEAD_LOW:
			mes "我沒辦法讓你變聰明，那得去找學校老師。";
			break;
		}
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "[巴斯塔]";
		mes "連我也無法精煉這個物品，沒辦法。";
		close;
	}
	.@refine_count = getequiprefinerycnt(.@part);
	if (.@refine_count < 10) {
		mes "[巴斯塔]";
		mes "我不是跟你說過嗎？我只精煉 +10 以上的裝備。";
		close;
	}
	else if (.@refine_count == 10)
		.@blacksmith_blessing_count = 7;
	else if (.@refine_count == 11)
		.@blacksmith_blessing_count = 11;
	else if (.@refine_count == 20) {
		mes "[巴斯塔]";
		mes "這把武器已經很完美了，不需要再精煉囉～";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_HD, REFINE_MATERIAL_ID);

	if ( .@itemtype != IT_WEAPON )
		.@type$ = "armor";
	else
		.@type$ = "weapon";
	mes "[巴斯塔]";
	mes "嗯……這是你想精煉的那件嗎？";
	mes "要精煉這件裝備，我需要 1 個 ^ff9999" + getitemname(.@material) + "^000000 和 " + callfunc("F_InsertComma",.@price) + " Zeny 作為手續費。";
	mes "你真的要精煉這件嗎？";
	next;
	if (select("是:否") == 2) {
		mes "[巴斯塔]";
		mes "好吧。如果你堅持的話……";
		close;
	}
	if (getequippercentrefinery(.@part, true) < 100) {
		mes "[巴斯塔]";
		mes "這件 " + .@type$ + " 已經精煉得很高了。";
		mes "如果再精煉，精煉等級可能會下降。";
		next;
		mes "[巴斯塔]";
		mes "我和其他地方的鐵匠不一樣。";
		mes "精煉等級不可能一下降 3 或 4 級……聽起來好可怕。";
		mes "在這裡最多只會降 1 級。";
		next;
		if (.@blacksmith_blessing_count == 0 || countitem(6635) < .@blacksmith_blessing_count) {
			mes "[巴斯塔]";
			mes "和其他鐵匠相比，風險比較小。";
			mes "我已經做好所有防範措施了。要試試看嗎？";
			next;
			setarray .@menu$[0], "", "Yes", "No";
		}
		else {
			mes "[巴斯塔]";
			mes "哇～是 ^316AC5鐵匠的祝福^000000 嗎？很難得到的～不過你居然有！";
			next;
			mes "[巴斯塔]";
			mes "如果是 +" + .@refine_count + " " + .@type$ + " 需要 ^316AC5" + .@blacksmith_blessing_count + " 個鐵匠的祝福^000000 可以防止裝備消失。要用鐵匠的祝福來精煉嗎？";
			next;
			setarray .@menu$[0], "Refine with Blacksmith Blessing", "Refine without Blacksmith Blessing", "Don't refine yet";
		}
		switch( select(.@menu$[0], .@menu$[1], .@menu$[2]) ) {
		case 1:
			.@bless_who = 1;
			break;
		case 2:
			break;
		case 3:
			mes "[巴斯塔]";
			mes "嗯～";
			mes "完全不挑戰，也是人生的一種智慧。";
			close;
		}
	}
	if ((.@bless_who && countitem(6635) < .@blacksmith_blessing_count) || countitem(.@material) == 0 || Zeny < .@price) {
		mes "[巴斯塔]";
		mes "嗯……你沒有帶齊所需的材料。";
		mes "材料都備齊了再來吧。";
		close;
	}
	if (.@bless_who)
		delitem 6635, .@blacksmith_blessing_count;
	delitem .@material,1;
	Zeny = Zeny - .@price;

	// anti-hack
	if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
		mes "[巴斯塔]";
		emotion ET_FRET;
		mes "等一下……";
		mes "你當我是笨蛋嗎？！";
		mes "你趁我沒注意的時候換了道具！給我滾出去！";
		close;
	}

	mes "咚！咚！咚！咚！";
	if (getequippercentrefinery(.@part, true) > rand(100)) {
		successrefitem .@part;
		next;
		emotion ET_BEST;
		mes "[巴斯塔]";
		mes "太棒了！做得好！！";
		mes "我真的是全世界最棒的鐵匠！";
		close;
	}
	if (.@bless_who == 1) {
		specialeffect EF_HOLYHIT;
		next;
		emotion (!rand(5))?ET_MONEY:ET_HUK;
		mes "[巴斯塔]";
		mes "啊啊啊啊啊！！！";
		next;
		mes "[巴斯塔]";
		mes "精煉失敗了！";
		mes "像我這樣全世界最棒的鐵匠……";
		mes "並不保證百分之百成功～";
	}
	else {
		downrefitem .@part;
		next;
		emotion (!rand(5))?ET_MONEY:ET_HUK;
		mes "[巴斯塔]";
		mes "啊啊啊啊啊！！！";
		next;
		mes "[巴斯塔]";
		mes "該死！";
		mes "精煉失敗，精煉等級也降低了！";
		mes "就算是全世界最棒的鐵匠，也不保證百分之百成功！";
	}
	mes "真可惜。";
	next;
	mes "[巴斯塔]";
	mes "下次我會做得更好！別擔心！";
	close;
}
