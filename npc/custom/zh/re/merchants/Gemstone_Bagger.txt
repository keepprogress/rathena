-	script	#GemstoneBagger	HIDDEN_NPC,{
	mes "[寶石包裝員]";
	mes "我們會用方便尋找的方式，把寶石裝進袋子裡。";
	mes "你想做什麼？";
	next;
	switch(select("交換寶石:取消")) {
	case 1:
		mes "[寶石包裝員]";
		mes "我來幫你準備一個寶石袋。";
		close2;
		callshop "Gemstone_Bagger";
		break;
	case 2:
		mes "[寶石包裝員]";
		mes "需要幫忙的話再回來吧。";
		close;
	}
}
