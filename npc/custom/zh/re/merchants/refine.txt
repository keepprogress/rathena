//============================================================
// +11 and above Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//=
//= On official servers, if an item is unsuccessfully refined
//= it will break at a 20% rate and downgrade at an 80% rate.
//============================================================
function	script	refinenew	{
	.@npc_name$ = getarg(0);
	disable_items;
	mes "["+ .@npc_name$ +"]";
	mes "我是史上最棒的武器匠！";
	mes "我不精煉普通又無聊的道具。";
	mes "我只精煉 10 級以上的道具。";
	next;
	mes "["+ .@npc_name$ +"]";
	mes "總之，道具等級在 10 以上才能使用我的服務。";
	mes "你想讓我精煉什麼？";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<=10; ++.@i) {
		if (getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "我想我無法精煉你擁有的任何道具……";
		close;
	}
	.@part = .@indices[ select(.@menu$) ];

	if (!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "你身上沒有穿戴";
		mes "任何";
		mes "我能精煉的東西。";
		emotion ET_FRET;
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "我想我沒辦法";
		mes "精煉這個道具了……";
		close;
	}
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	if (.@refinerycnt < 10) {
		mes "["+ .@npc_name$ +"]";
		mes "我說過了，我不處理 10 級以下的道具。";
		close;
	}
	if (.@refinerycnt >= 20) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "這個我沒辦法再";
		mes "精煉了。這已經";
		mes "是精煉的極限了！";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@safe = 10;
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	
	if( .@itemtype == IT_WEAPON ){
		.@article$ = "a";
		.@type$ = "weapon";
	}else if( .@itemtype == IT_ARMOR ){
		.@article$ = "an";
		.@type$ = "armor";
	}else{
		// TODO:
		close;
	}

	mes "["+ .@npc_name$ +"]";
	mes "嗯 " + .@article$ + " " + .@type$ + "，可以嗎？";
	mes "如果你想精煉這個 " + .@type$ + ",";
	mes "我需要 1 個 ^003366" + getitemname(.@material) + "^000000 和 " + callfunc("F_InsertComma",.@price) + " Zeny。";
	mes "你確定要繼續嗎？";
	next;
	if (select("是:否") == 2) {
		mes "["+ .@npc_name$ +"]";
		mes "嗯……如果你介意……那就算了……";
		close;
	}
	if (getarg(1) != 1) {
		if (getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "這件 "+.@type$+" 已經精煉過好幾次了。";
			mes "如果我再試一次，它可能會被摧毀。";
			mes "雖然不一定會壞，但還是有小機率。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "你可能會讓 ^FF0000精煉等級下降^000000 的 "+.@type$+",";
			mes "如果壞掉，你將失去上面的 ^FF0000所有卡片^000000 或特殊屬性。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "你還是要我精煉嗎？";
			mes "我想我已經警告你夠多次了。";
			next;
			if(select("是。:否。") == 2) {
				mes "["+ .@npc_name$ +"]";
				mes "好吧，不挑戰也是一種選擇……";
				mes "沒有風險……這也許是明智的選擇。";
				close;
			}
		}
		if (countitem(.@material) < 1 || Zeny < .@price) {
			mes "["+ .@npc_name$ +"]";
			mes "嗯。你似乎錢不夠或 "+getitemname(.@material)+".";
			mes "等你備齊了再來吧。";
			close;
		}
		Zeny = Zeny - .@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "等一下……";
			mes "你當我是笨蛋嗎？！";
			mes "你趁我沒注意的時候換了道具！給我滾出去！";
			close;
		}

		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "鏘！鏘！鏘！鏘！";
			successrefitem .@part;
			next;
			emotion ET_BEST;
			mes "["+ .@npc_name$ +"]";
			mes "好！成功了！！！";
			mes "我是最棒的武器匠。";
			close;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "鏘！鏘！鏘！鏘！";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "啊啊啊！！！";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "天啊！";
				mes "精煉等級下降了……";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "鏘！鏘！鏘！";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "嗯嗯嗯！";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "哎呀！我精煉失敗了……";
				mes "我不是故意的！";
			}
			mes "就算我是史上最棒的武器匠，也可能會失手。";
			mes "這就是命吧。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "下次我會做得更好！別擔心！";
			close;
		}
	}
// New +11 and above Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "我可以把它精煉到安全上限，或是精煉到你想要的次數。由你決定。";
		next;
		.@menu2 = select("請精煉到安全上限。","我自己決定次數。","我改變主意了……");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "你想讓我把道具精煉幾次？";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 20) {
			mes "["+ .@npc_name$ +"]";
			mes "這個道具我沒辦法精煉那麼多次。";
			close;
		}
		if (.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
			mes "這將嘗試把裝備精煉超過安全上限 " + .@refinecheck + " 次。你的裝備可能會被摧毀……這樣可以嗎？";
			next;
			if(select("是的……","不……") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "你都這麼說了……那就這樣吧。";
				close;
			}
		}
		break;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "你都這麼說了……那就這樣吧。";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "這將會花費你 " + .@refinecnt + " " + getitemname(.@material) + " 與 " + .@fullprice + " Zeny。這樣可以嗎？";
	next;
	if(select("是:否……") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "你都這麼說了……那就這樣吧。";
		close;
	}
	if (countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
		mes "嗯。你似乎錢不夠或 "+getitemname(.@material)+".";
		mes "等你備齊了再來吧。";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "喂……你身上根本沒有穿戴任何道具……";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt))
			close;

		if (getequipid(.@part) != .@refineitemid || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "鏘……不，你竟然以為我有這麼笨嗎？！";
			mes "你改了……";
			mes "在我用錘子把你打暈之前，給我滾出去！！";
			close;
		}
		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "鏘！鏘！鏘！鏘！";
			successrefitem .@part;
			.@refinecnt = .@refinecnt - 1;
			next;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "鏘！鏘！鏘！鏘！";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "啊啊啊！！！";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "天啊！";
				mes "精煉等級下降了……";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "鏘！鏘！鏘！";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "嗯嗯嗯！";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "哎呀！我精煉失敗了……";
				mes "我不是故意的！";
			}
			mes "就算我是史上最棒的武器匠，也可能會失手。";
			mes "這就是命吧。";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "下次我會做得更好！別擔心！";
			close;
		}
		.@refinerycnt = getequiprefinerycnt(.@part);
	}
	emotion ET_BEST;
	mes "["+ .@npc_name$ +"]";
	mes "好！成功了！！！";
	mes "我是最棒的鐵匠。";
	close;
}

	mes "你需要 ^ff9999"+getitemname(.@material)+"^000000 和 ^ff9999"+.@price+"^000000 Zeny。你有嗎？";
	next;
	if(select("是，我要!!:算了!!") == 2) {

		mes "我就知道！！";

		close;

	if (getequippercentrefinery(.@part) < 100) {
		mes "[克林克]";
		mes "哇！！";
		mes "這件 "+.@type$+" 已經精煉很多次了，對吧？";
