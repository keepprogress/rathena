dali02,51,130,4	script	次元旅人::Dimension Traveler	4_F_SHABBY,{
	if (checkweight(28100,1) == 0) {// Thanos Axe
		mes "- 您攜帶的物品太多，無法繼續。 -";
		close;
	}
	.@item_name_req$ = getitemname(6672);// Gray_Shard
	mes "[次元旅人]";
	mes "冒險者朋友， " + mesitemlink( 6672 ) + " 正是我需要的。要不要跟我換點東西？";
	next;
	.@type = select( "給我看看你的武器。", "給我看看你的盔甲。", "有什麼特別的嗎？", "我不需要。" ) - 1;
	mes "[次元旅人]";
	switch(.@type) {
	case 0:
		mes "喔！您對武器有興趣嗎？幽靈宮能拿到的武器，我們這裡都有。挑一件看看吧？";
		break;
	case 1:
		mes "盔甲嗎？幽靈宮能拿到的盔甲，我們這裡都有。挑一件看看吧？";
		break;
	case 2:
		mes "呼！看來很多人都聽說我這裡有最好的東西了。請隨意看看我的收藏吧。";
		break;
	case 3:
		mes "再見，朋友～歡迎再來！";
		close;
	}
	next;
	mes "[次元旅人]";
	mes "啊！寫在旁邊的數字 " + mesitemlink( 6672 ) + " 將會需要。";
	mes "請記住這點。";

	switch(.@type) {
	case 0:
		setarray .@items[0],
			13441, 200,		// Thanos Sword
			21009, 200, 	// Thanos Great Sword
			1438,  200,		// Thanos Spear
			1496,  200,		// Thanos Long Spear
			1669,  200,		// Thanos Staff
			2023,  200,		// Thanos Rod
			18119, 200,		// Thanos Bow
			13093, 200,		// Thanos Dagger
			28000, 200,		// Thanos Katar
			1836,  200,		// Thanos Knuckle
			16028, 200,		// Thanos Hammer
			28100, 200,		// Thanos Axe
			1933,  200,		// Thanos Violin
			1988,  200;		// Thanos Whip
		break;
	case 1:
		setarray .@items[0],
			2187,  100,		// Shield Of Gray
			15090, 100, 	// Armor Of Gray
			15091, 100,		// Gray Robe
			20721, 100,		// Cloak Of Gray
			22033, 100,		// Boots Of Gray
			18820, 100;		// Gray Helmet
		break;
	case 2:
		setarray .@items[0],
			27184, 10000,		// Knight Sakray Card
			23676, 50, 			// Gray Charcoal Melee (Melee)
			23677, 50,			// Gray Charcoal Magic (Magic)
			23678, 50;			// Gray Charcoal Range (Range)
		break;
	}
	.@size = getarraysize(.@items);
	for ( .@i = 0; .@i < .@size; .@i += 2 )
		.@menu$ += getitemname(.@items[.@i]) + " - " + .@items[.@i+1] + ":";

	while(true) {
		next;
		.@s = (select(.@menu$) - 1) * 2;
		mes "[次元旅人]";
		if (.@type != 1)	// armor type doesn't display item info
			mes "您是指 " + mesitemlink( .@items[.@s] ) + "?";
		mes "" + .@items[.@s+1] + " ^006400" + .@item_name_req$ + "^000000 為兌換所需。";
		next;
		switch( select( "來兌換吧。", "讓我看看其他東西。", "我晚點再來。" ) ) {
		case 1:
			break;
		case 2:
			mes "[次元旅人]";
			switch(.@type) {
			case 0:
				mes "我的武器已經夠多了。";
				break;
			case 1:
				mes "我們有很多盔甲可供您選擇。挑一件吧？";
				break;
			case 2:
				mes "如果您帶來 " + .@item_name_req$ + "，我就能用您想要的任何東西跟您交換。";
				break;
			}
			continue;
		case 3:
			mes "[次元旅人]";
			mes "保重，朋友～下次見！";
			close;
		}
		if (countitem(6672) < .@items[.@s+1]) {
			mes "[次元旅人]";
			mes "朋友，你讓我陷入了為難的處境。";
			mes "我需要 " + .@item_name_req$ + "。我會用它換取 ^006400" + .@items[.@s+1] + " " + .@item_name_req$ + "^000000.";
			close;
		}
		mes "[次元旅人]";
		mes "謝謝。如果還需要什麼，再來找我吧。";
		delitem 6672, .@items[.@s+1];
		getitem .@items[.@s], 1;
		close;
	}
}
