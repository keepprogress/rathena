-	script	RareDiamondMerchant	58,{

	set .@npc$, "[Rare Diamond Merchant]";

	mes .@npc$;
	mes "各位女士先生！！！";
	mes "你聽過傳聞，卻從未親眼見過";
	mes "世界上最稀有的鑽石！";
	mes "^FF82FF'17 克拉鑽石'^000000 現在特價出售！！";
	mes "如果你對珍貴的鑽石感興趣，就仔細聽好！";
	next;
	
	mes .@npc$;
	mes "你可以把鑽石送給你的戀人當作禮物。";
	mes "收到^FF82FF'17 克拉鑽石'^000000的人會珍惜它，";
	mes "並把你珍藏在心中。";
	emotion ET_BLABLA;
	next;
	
	mes .@npc$;
	mes "什麼？你沒有戀人嗎？";
	mes "那你還是可以自己留著。";
	mes "^FF82FF'17 克拉鑽石'^000000 迷人的光輝，很可能";
	mes "為你的生活帶來愛情。";
	next;
	
	mes .@npc$;
	mes "被譽為鑽石中最迷人的鑽石……";
	mes "它就是^FF82FF'17 克拉鑽石'^000000！";
	next;

	set .@menu$, "^FF82FF'17 Carat Diamond'^000000?";
	if (countitem(6024))
		set .@menu$, .@menu$ + ":Exchange my 17 Carat Diamond for zeny...";

	switch(select(.@menu$)) {
	case 1:
		mes .@npc$;
		mes "好！";
		mes "呵呵，你現在看到的^FF82FF'17 克拉鑽石'^000000";
		mes "是最高品質的 17 克拉鑽石之一。";
		mes "跟你可能見過的任何豌豆大小、無色的石頭相比";
		mes "從夢羅克挖出來的那些，根本無法比擬！";
		emotion ET_HUK;
		next;
		
		mes .@npc$;
		mes "有人說這顆鑽石，";
		mes "就是在克魔島被卡丘亞看見的那一顆！";
		next;
		
		if (select("結束對話:多少錢？") == 1) {
			mes .@npc$;
			mes "相當便宜喔！";
			mes "有興趣的話，隨時來找我！";
			close;
		}
		
		mes .@npc$;
		mes "當然！最重要的就是價格！";
		mes "正如我先前所說，^FF82FF'17 克拉鑽石'^000000 已經";
		mes "被鑑定為 17 克拉的鑽石。";
		mes "世界上這樣的鑽石不多。";
		next;
		
		mes .@npc$;
		mes "價格只要 5 億 Zeny！";
		mes "另外還有 1,000,000z 的手續費。";
		mes "與這顆鑽石的品質相比，";
		mes "這點代價算不了什麼。";
		next;
		
		if (select("^828282太貴了。^000000:我買。") == 1) {
			mes .@npc$;
			mes "說一顆像";
			mes "^FF82FF'17 克拉鑽石'^000000 太貴了， ";
			mes "這話讓我無言以對。";
			close;
		}
		
		mes .@npc$;
		mes "喔喔！我一見到你就知道，你是這顆美麗鑽石的正統";
		mes "主人。";
		mes "很高興我沒有看錯！";
		mes "我相信這顆鑽石很適合你。";
		mes "我只需要你完成這份契約。";
		next;
		
		while(1) {
			if (select("^B9062F我要簽署契約。^000000:^828282算了，我改變主意了。^000000") == 2) {
				mes .@npc$;
				mes "這樣啊，那真可惜……";
				close;
			}
			
			mes .@npc$;
			mes "你叫什麼名字？";
			next;
			
			mes "[" + strcharinfo(0) + "]";
			mes "我的名字是 " + strcharinfo(0) + ".";
			next;

			mes .@npc$;
			mes "好吧，請在這裡簽名。";
			next;
			
			input .@charname$;
			if (.@charname$ == strcharinfo(0)) {
				mes .@npc$;
				mes "好，很好。";
				mes "剩下的就只有付款了。";
				next;
				
				if (Zeny >= 501000000) {
					mes .@npc$;
					mes ". . . . . .";
					next;
					
					mes .@npc$;
					mes "正在計算價格和手續費……";
					mes "501,000,000 Zeny。";
					mes "金額已確認。";
					emotion ET_BLABLA;
					set Zeny, Zeny - 501000000;
					getitem 6024, 1;
					next;

					mes .@npc$;
					mes "你隨時都可以把鑽石換回 Zeny。";
					mes "不過會收取 1,000,000z 的手續費。";
					mes "感謝您的惠顧。";
					emotion ET_THANKS;
					close;
				}
				else {
					mes .@npc$;
					mes "嗯？";
					mes "很抱歉，你的餘額不足。";
					mes "包含手續費，總共需要 501,000,000z。";
					mes "請確認你的 Zeny 餘額後再試一次。";
					emotion ET_SCRATCH;
					close;
				}
			}
			else {
				mes .@npc$;
				mes "這真的是你的簽名嗎？";
				mes "要再簽一次確認嗎？";
				next;
			}
		}
	case 2:
		mes .@npc$;
		mes "啊。你想把你的^FF82FF'17 克拉鑽石'^000000換成 Zeny 嗎？";
		emotion ET_SURPRISE;
		next;
		if (select("是:否") == 2) {
			mes .@npc$;
			mes "那麼，你想要什麼呢……";
			mes ". . . . . ....";
			emotion ET_THINK;
			close;
		}

		mes .@npc$;
		mes "我明白了。收到你的簽名後， ";
		mes "你必須交出你的^FF82FF'17 克拉鑽石'^000000。";
		mes "也別忘了還有 1,000,000z 的手續費！";
		next;
		
		mes .@npc$;
		mes "你叫什麼名字？";
		next;
		
		mes "[" + strcharinfo(0) + "]";
		mes "我的名字是 " + strcharinfo(0) + ".";
		next;
		
		mes .@npc$;
		mes "好吧，請在這裡簽名。";
		next;
		
		while (1) {
			if (select("簽名:不簽名") == 2) {
				mes .@npc$;
				mes "沒有你的簽名，交易無法完成。";
				close;
			}
			
			input .@charname$;
			if (.@charname$ == strcharinfo(0)) {
				mes .@npc$;
				mes "已收到你的簽名。";
				mes "現在我將收下你的^FF82FF'17 克拉鑽石'^000000……";
				mes "之後你將收到^0000FF499,000,000z^000000。";
				next;
				
				mes .@npc$;
				mes "感謝您的惠顧。";
				mes "當你需要^FF82FF'17 克拉鑽石'^000000時，";
				mes "隨時帶著 5 億 Zeny 回來就行。";
				emotion ET_THANKS;
				delitem 6024, 1;
				set Zeny, Zeny + 499000000;
				close;
			}
			else {
				mes .@npc$;
				mes "這真的是你的簽名嗎？";
				mes "要再簽一次確認嗎？";
				next;
			}
		}
	}
}
