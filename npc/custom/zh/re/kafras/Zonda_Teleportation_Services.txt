-	script	ZondaTeleportationServices	HIDDEN_NPC,{
	cutin "zonda_do01", 2;
	mes "[Zonda 職員]";
	mes "歡迎。";
	mes "我們的團隊一直致力於超越您對服務品質的期待。";
	mes "請選擇您的目的地。";
	next;
	switch( select( "北部彩虹橋", "艾克拉格", "斯普蘭迪德", "米德加德遠征營地", "馬努克", "卡米達爾山郊外", "取消。") ) {
	case 1:
		callsub( S_Zone, "North Bifrost", 17000,850, 7213 ); // Teleport value not confirmed
		warp "bif_fild02",160,230; // Unverified Coordinates
		end;
	case 2:
		callsub( S_Zone, "Eclage", 17000,850, 11311 );
		warp "eclage",111,39;
		end;
	case 3:
		callsub( S_Zone, "Splendide", 17000,850, 10078 );
		warp "splendide",171,115;
		end;
	case 4:
		callsub( S_Zone, "Midgard Expedition Camp", 19000,950, 10078 );
		warp "mid_camp",210,288;
		end;
	case 5:
		callsub( S_Zone, "Manuk", 22000,1100, 10078 );
		warp "manuk",263,177;
		end;
	case 6:
		callsub( S_Zone, "Outskirts of Kamidal Mountain", 24000,1200, 7184 );
		warp "dic_fild01",188,187;
		end;
	case 7:
		mes "[Zonda 職員]";
		mes "我們一直努力為顧客提供最好的服務。";
		mes "感謝您使用我們的服務。";
		close3;
	}

S_Zone:
	.@zeny_req = getarg(1);
	.@item_req = getarg(2);
	.@quest_req = getarg(3);
	
	if (isbegin_quest(.@quest_req) == 0) {
		mes "[Zonda 職員]";
		//mes getarg(0);
		mes "您必須先完成通行任務，才能使用傳送服務前往 " + getarg(0) + " 。";
		close3;
	}
	mes "[Zonda 職員]";
	mes getarg(0) + " 可用於 ^ff0000" + .@zeny_req + " Zeny^000000 或 ^ff0000" + .@item_req + " Malang Sp Can^000000。";
	next;
	switch( select( .@zeny_req + " Zeny", .@item_req + " Malang Sp Can", "取消" ) ) {
	case 1:
		if (Zeny < .@zeny_req)
			break;
		Zeny -= .@zeny_req;
		return;
	case 2:
		if (countitem(12636) < .@item_req)
			break;
		delitem 12636,.@item_req; // Malang_Sp_Can
		return;
	case 3:
		mes "[Zonda 職員]";
		mes "能告訴我其他服務地區的資訊嗎？";
		close3;
	}
	mes "[Zonda 職員]";
	mes "先生！您沒有足夠我們所要求的價值。";
	mes "請稍後再來看看。";
	close3;
}
