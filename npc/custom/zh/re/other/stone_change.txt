geffen_in,99,174,5	script	礦石鑑定員#sorty::Ore Tester#sorty	8_F_GIRL,{
	mes "[礦石鑑定員索提]";
	mes "你的背包裡有^3131FF任何同類的礦石^000000，佔了很多格子嗎？";
	mes "我可以立刻為你分辨並整理礦石~！";
	next;
	switch(select("整理我的礦石:取消")) {
	case 1:
		mes "[礦石鑑定員索提]";
		mes "目前我只能分辨並整理^FF0000鋼鐵^000000、^FF0000鐵^000000、^FF0000鐵礦石^000000和^FF0000神之金屬原石^000000。";
		mes "請注意，^3131FF其他道具無法處理^000000！";
		next;
		mes "[礦石鑑定員索提]";
		mes "那麼，要我幫你整理礦石嗎？";
		next;
		.@i = select("請整理鋼鐵:請整理鐵:請整理鐵礦石:請整理神之金屬原石:不用了，謝謝");
		if (.@i == 5) {
			mes "[礦石鑑定員索提]";
			mes "哼，我最討厭邋遢的人了！";
			close;
		}
		setarray .@ores[0],999,998,1002,756;
		.@item = .@ores[.@i-1];
		if (countitem(.@item) == 0) {
			mes "[礦石鑑定員索提]";
			mes "你沒有任何 "+getitemname(.@item)+"...";
			close;
		}
		// while(1) {
			// if (countitem(.@item)) {
				// set .@count, countitem(.@item);
				// set .@total, .@total + .@count;
				// delitem .@item, .@count;
			// } else {
				// getitem .@item, .@total;
				// break;
			// }
		// }
		.@count = countitem(.@item);
		delitem .@item, .@count;
		getitem .@item, .@count;
		mes "[礦石鑑定員索提]";
		mes "整理完成了。看起來很棒吧！你不覺得嗎？";
		close;
	case 2:
		mes "[礦石鑑定員索提]";
		mes "你的背包看起來好乾淨呢？呼呼！你大概是整理大師吧！";
		close;
	}
}
