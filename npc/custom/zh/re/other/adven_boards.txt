function	script	F_AvenBoard	{
	.@mo_board = select( "依樓層查詢怪物",
					"怪物情報查詢系統",
					( countitem(909) == 30000 ? "請求" : "" ),
					"取消" );
	if (.@mo_board == 1)
		return 1;
	else if (.@mo_board == 2) {
		while(1) {
			mes "[怪物情報查詢系統]";
			mes "這裡是怪物情報查詢系統。可以查詢此地城出現的怪物情報。搜尋時請輸入怪物名稱，包括空格。";
			next;
			input .@inputstr$;
			setarray .@element$[0],"Neutral","Water","Earth","Fire","Wind","Poison","Holy","Shadow","Ghost","Undead";
			setarray .@size$[0],"Medium","Small","Big";
			setarray .@race$[0],"Formless","Undead","Animal","Plant","Insect","Fish","Demon","Demihuman","Angel","Dragon";
			.@size = getargcount();
			.@i = 0;
			while( .@i < .@size && getarg(.@i+1,"") != .@inputstr$ ) .@i += 3;
			mes "[怪物情報查詢系統]";
			mes "搜尋中";
			next;
			mes "[怪物情報查詢系統]";
			mes "搜尋完成。";
			mes "正在顯示搜尋結果。";
			next;
			mes "[怪物情報查詢系統]";
			if (.@i >= .@size) {
				mes "該怪物沒有可用的情報。";
				mes "請確認怪物名稱。";
			}
			else {
				.@mob_id = getarg(.@i);
				mes "- 名稱 : "+ getarg(.@i+1);
				mes "- 屬性 : "+ .@element$[ getmonsterinfo(.@mob_id,MOB_ELEMENT) ];
				mes "- 體型 : "+ .@size$[ getmonsterinfo(.@mob_id,MOB_SIZE) ];
				mes "- 種族 : "+ .@race$[ getmonsterinfo(.@mob_id,MOB_RACE) ];
				mes "--------------------------------^FD8D02";
				mes getarg(.@i+2);
			}
			next;
			mes "[怪物情報查詢系統]";
			mes "要查詢其他怪物情報嗎？";
			next;
			if (select( "搜尋","取消" ) == 1)
				continue;
			mes "[怪物情報查詢系統]";
			mes "怪物搜尋系統即將關閉。";
			next;
			return 2;
		}
	}
	else if (.@mo_board == 3) {
		mes "將顯示請求的情報 - 出現就沒問題";
		next;
		return 3;
	}
	else if (.@mo_board == 4)
		close;
	else {
		mes "操作錯誤。";
		close;
	}
}

	next;
	return;
}
