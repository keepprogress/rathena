//===== rAthena Script =======================================
//= Dynamic NPC: Kachua's Secret Box
//===== Description: =========================================
//- [Official conversion]
//= Kachua's Secret Box is a Gachapon NPC.
//= It lets the player exchange a Kachua's Secret Key for a random item.
//= It also gives a Kachua's Mileage Coupon for each pull.
//===== Changelogs: ==========================================
//= 1.0 First version. [secretdataz]
//============================================================

sec_in02,126,178,3	script	卡丘雅寶箱#bm::Kachua's Secret Box#bm	4_TREASURE_BOX,{
	.@key$ = "K_Secret_Key";
	.@keyname$ = getitemname( .@key$ );
	.@box$ = "Main_Lucky_Box";
	.@boxname$ = getitemname( .@box$ );
	mes "一個沒人知道裡面裝什麼的神秘寶箱。";
	mes "^4d4dff開啟此寶箱需要消耗 " + mesitemlink( .@key$ ) + ".^000000";
	next;
	switch(select("^4d4dff開啟 1 次（需要 1 " + .@keyname$ + ")^000000","^4d4dff開啟 10 次（需要 10 " + .@keyname$ + "）^000000")) {
		case 1:
			mes "^FF0000[告示]^000000";
			mes "^FF0000以隨機道具交換一個 " + .@keyname$ + ".^000000";
			mes "^FF0000上述交換的道具無法取回，也無法再換成 " + .@keyname$ + ".^000000";
			next;
			if(select("繼續。","結束對話。") == 2) {
				mes "你決定不開啟寶箱。";
				close;
			} else if (countitem(.@key$) < 1) {
				mes "數量不足 " + mesitemlink( .@key$ ) + ".";
				close;
			} else {
				if (checkweight(1201,1) == 0 || ((MaxWeight - Weight) * 100 / MaxWeight) < 10) {
					mes "^4d4dff請確認背包有足夠的空間。^000000";
					close;
				}
				delitem(.@key$, 1);
				consumeitem(.@box$);
				mes .@boxname$ + " 已被開啟！";
				mes "運氣如何？";
				specialeffect2 EF_VALLENTINE;
				close;
			}
		case 2:
			mes "^FF0000[告示]^000000";
			mes "^FF0000以隨機道具交換 10 個 " + .@keyname$ + "。^000000";
			mes "^FF0000上述交換的道具無法取回，也無法再換成 " + .@keyname$ + ".^000000";
			next;
			if(select("繼續。","結束對話。") == 2) {
				mes "你決定不開啟寶箱。";
				close;
			} else if (countitem(.@key$) < 10) {
				mes "你沒有足夠的 " + mesitemlink( .@key$ ) + ".";
				close;
			} else {
				for (.@i = 1; .@i <= 10; ++.@i) {
					progressbar "4d4dff",2;
					if (checkweight(1201,1) == 0 || ((MaxWeight - Weight) * 100 / MaxWeight) < 10) {
						mes "^4d4dff請確認背包有足夠的空間。^000000";
						close;
					}
					if (countitem(.@key$) < 1) { // Custom check, just in case
						close;
					}
					delitem(.@key$, 1);
					consumeitem(.@box$);
					dispbottom .@boxname$ + " 已被開啟 " + .@i + " 次。還有一個正在開啟中。",0xFFFFFF;
					specialeffect2 EF_VALLENTINE;
				}
				mes .@boxname$ + " 已被開啟 10 次！";
				mes "運氣如何？";
				close;
			}
	}
}
