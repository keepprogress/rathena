function	script	VIP_Third_Class	{
	if (!vip_status(VIP_STATUS_ACTIVE)) {
		mes "[不明聲音]";
		mes "只有持有一定數量重置石的人才能使用高級內容。";
		next;
		if (countitem(6320) < 1) {
			mes "[不明聲音]";
			mes "要轉職為三轉，你必須攜帶 1 個^0000CC重置石^000000，或擁有高級帳號。";
			//mes "Please visit our iShop to purchase the ^0000CCReset Stone^000000.";
			close;
		} else {
			mes "[不明聲音]";
			mes "你要用 1 個^0000CC重置石^000000參加這個任務嗎？";
			next;
			switch(select("是:否")) {
			case 1:
				mes "[不明聲音]";
				mes "祝你好運，冒險者。";
				delitem 6320,1; //Premium_Reset_Stone
				next;
				break;
			case 2:
				mes "[不明聲音]";
				mes "好的。";
				close;
			}
		}
	}
	return;
}
