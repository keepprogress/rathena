-	script	Goods Cabinet#00::pggc	-1,{
	mes "這是一個物品保管櫃。";
	mes "紙上寫著一段訊息。";
	next;
	mes "+ 保管櫃專供 +";
	mes "+ 伊甸園集團使用 +";
	mes "若你想使用這個";
	mes "保管櫃，請先確認";
	mes "以下事項:";
	next;
	mes "1. 你是否為";
	mes "   伊甸園成員嗎？";
	mes "2. 你是否已學會";
	mes "   足夠的基礎技能？";
	mes "3. 保管櫃費用為";
	mes "  ^4d4dff每次 500 Zeny^000000！";
	next;
	if (callfunc("F_HasEdenGroupMark")) {
		mes "你需要投入 Zeny 才能使用保管櫃。";
		mes "費用 : 500 Zeny ";
		mes "你想要使用它嗎？";
		next;
		switch(select("使用保管櫃。:取消。")) {
			case 1:
				if (Zeny > 499) {
					if(!callfunc("F_CanOpenStorage")){
						mes "保管櫃對我沒反應。";
						mes "也許我還沒有資格使用保管櫃。";
						close;
					}
					set Zeny, Zeny - 500;
					close2;
					openstorage;
					end;
				}
				mes "我的 Zeny 不夠。";
				mes "使用保管櫃至少需要 500 Zeny。";
				close;
			case 2:
				mes "下次再用吧。";
				close;
		}
	}
	mes "我需要伊甸園徽章才能使用這個保管櫃。";
	close;
}
