S_Quest:
	if (isbegin_quest(getarg(0)))
		erasequest getarg(0);
	setquest getarg(1);
	if (getarg(2,0))
		getexp getarg(2),getarg(3);
	return;
S_Erasequest:
	.@total = getargcount();
	for ( .@i = 0; .@i < .@total; .@i += 3 ) {
		.@quest_id = getarg(.@i);
		switch( checkquest(.@quest_id,HUNTING) ) {
		case -1:
			if (checkquest(.@quest_id+1) > -1)
				erasequest (.@quest_id+1);
			continue;
		case 0:
		case 1:
			mes "[拉基]";
			switch( getarg(.@i+1) ) {
			case 1:
				mes "其餘的 "+ getarg(.@i+2) +" 的故事沒關係，我去問別人吧。";
				break;
			case 2:
			case 3:
				mes "我會去問其他冒險者關於 "+ getarg(.@i+2) +".";
				break;
			}
			break;
		case 2:
			mes "[拉基]";
			switch( getarg(.@i+1) ) {
			case 1:
				mes "你聽說過 "+ getarg(.@i+2) +"的故事嗎？謝謝你。我保證會寫出一首好歌。";
				getexp 1100000,300000;
				break;
			case 2:
				mes "夢羅克的故事夠不夠寫成歌？你做得很好。";
				getexp 1100000,200000;
				break;
			case 3:
				mes "你的冒險怎麼樣？好玩嗎？改天也讓我聽聽你的冒險故事吧。";
				getexp 1100000,200000;
				break;
			}
		}
		erasequest .@quest_id;
		next;
	}
	return;
