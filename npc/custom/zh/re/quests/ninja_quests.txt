que_ng,21,72,6	script	防具大師::Master Armor Craftsman	4_M_RASWORD,{
	mes "[吉奧]";
	mes "你會如何定義勝利？";
	mes "擊倒敵人……";
	mes "不過，最重要的是……";
	mes "絕不能讓自己倒下！";
	next;
	mes "[吉奧]";
	mes "為此，你必須更好地保護自己。";
	mes "若妨礙自己的防護，你就無法保護「他人」。";
	next;
	if(select("合成護甲。:結束對話。") == 2) {
		mes "[吉奧]";
		mes "謹慎在戰鬥中也是重要的美德。";
		close;
	}
	switch(select("狼護臂[1]:新月護臂[1]:忍者鱗片鎧甲[1]:霧中暗影[1]:取消。")) {
	case 1:
		mes "[吉奧]";
		mes "狼護臂……一件野性十足的護甲。";
		next;
		callsub L_Select,2172;
		if (countitem(2171) > 0 && countitem(6252) > 9) {  //custom translation
			delitem 2171,1; //Fox_Armguard
			delitem 6252,10; //Wolf_Blood
			getitem 2172,1; //Wolf_Armguard
			mes "[吉奧]";
			mes "你感受到狼的本性了嗎？";
			mes "既狂野又迷人……";
			close;
		}
		break;
	case 2:
		mes "[吉奧]";
		mes "新月護臂……既美麗又堅強……兩者皆不遜色。";
		next;
		callsub L_Select,2173;
		if (countitem(2171) > 0 && countitem(7321) > 99) {  //custom translation
			delitem 2171,1; //Fox_Armguard
			delitem 7321,100; //Fragment_Of_Crystal
			getitem 2173,1; //Crescent_Armguard
			mes "[吉奧]";
			mes "如此美麗……";
			mes "彷彿全身都溢滿了魔力。";
			close;
		}
		break;
	case 3:
		mes "[吉奧]";
		mes "忍者鱗片鎧甲……據說傳說中的忍者曾經穿過它。";
		next;
		callsub L_Select,15054;
		if (countitem(7451) > 29 && countitem(7562) > 29 && countitem(6091) > 29) {  //custom translation
			delitem 7451,30; //Scale_Of_Red_Dragon
			delitem 7562,30; //Ice_Scale
			delitem 6091,30; //Dark_Red_Scale
			getitem 15054,1; //Ninja_Scale_Armor
			mes "[吉奧]";
			mes "這真是一件傳說中的護甲……";
			mes "即使我不相信傳說……";
			close;
		}
		break;
	case 4:
		mes "[吉奧]";
		mes "暗影王的護甲……它是為了保護他人而打造的。";
		next;
		callsub L_Select,15055;
		if (countitem(15056) > 0 && countitem(6089) > 9) {  //custom translation
			delitem 15056,1; //Special_Ninja_Suit_
			delitem 6089,10; //Piece_Of_Darkness

		close;
	}
	mes "[吉奧]";
	mes "您的材料不足。";
	mes "再看看……";
	close;

L_Select:
	.@string1$ = "Check Ingredients.";
	.@string2$ = "Check Stats.";
	while(1) {
		switch(select(.@string1$,.@string2$,"合成。","離開。")) {
		case 1:
			.@string1$ = "";
			callsub L_Ingredients, getarg(0);
			break;
		case 2:
			.@string2$ = "";
			callsub L_Stats, getarg(0);
			break;
		case 3:
			callsub L_Combine;
			return;
		case 4:
			mes "[吉奧]";
			mes "看來你還需要多一點時間考慮。";
			close;
		}
	}

L_Ingredients:
	mes "[吉奧]";
	switch (getarg(0)) {
	case 2172:	//Wolf_Armguard
		mes "要製作^ff0000狼護臂，你需要1個狐狸護臂和10個狼之血^000000。";
		mes "你感覺到了嗎？狼血的悸動……";
		break;
	case 2173:	//Crescent_Armguard
		mes "要製作^ff0000新月護臂，你需要1個狐狸護臂和100個水晶碎片^000000。";
		mes "一道如水晶般美麗透明的新月將會被刻在上面。";
		break;
	case 15054:	//Ninja_Scale_Armor
		mes "要製作^ff0000忍者鱗片鎧甲，你需要30個火龍鱗片、30個冰鱗片和30個暗紅鱗片碎片^000000。";
		mes "你能感受到它傳說中的力量。";
		break;
	case 15055:	//Tenebris_Latitantes
		mes "要製作^ff0000暗影王的護甲，你需要一件帶有卡槽的特殊忍服血櫻和10個暗黑碎片^000000。";
		mes "與黑暗一同，抹消存在本身……";
		break;
	}
	next;
	return;

L_Stats:
	mes "[" + getitemname(getarg(0)) + "]";
	switch (getarg(0)) {
	case 2172:	//Wolf_Armguard
		mes "近戰攻擊時有機率獲得5秒 ATK+100、FLEE -50。";
		mes "盾牌類型，防禦力45，孔數1";
		mes "需求等級100，僅限忍者";
		break;
	case 2173:	//Crescent_Armguard
		mes "每精煉一次，技能後延遲減少2%。";
		mes "盾牌類型，防禦力70，孔數1";
		mes "需求等級100，僅限忍者";
		break;
	case 15054:	//Ninja_Scale_Armor
		mes "最大HP+15%，最大SP-30%";
		mes "防具類型，防禦力90，孔數1";
		mes "需求等級100，僅限忍者";
		break;
	case 15055:	//Tenebris_Latitantes
		mes "被近戰攻擊時有機率發動等級1的幻影 - 暗影。";
		mes "防具類型，防禦力60，孔數1";
		mes "需求等級100，僅限忍者";
		break;
	}
	next;
	return;

L_Combine:
	mes "[吉奧]";
	mes "請確認^ff0000你手邊已備有正確的材料與裝備^000000。";
	mes "事後再後悔也沒用了。";
	next;
	switch(select("合成:離開")) {
	case 1:
		if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
			mes "- 等一下！！ -";
			mes "- 無法接收道具 -";
			mes "- 因為你攜帶的物品太多。 -";
			mes "- 請再試一次 -";
			mes "- 請減輕負重後再來。 -";
			close;
		}
		return;
	case 2:
		mes "[吉奧]";
		mes "看來你還需要多一點時間考慮。";
		close;
	}
	end;
}
