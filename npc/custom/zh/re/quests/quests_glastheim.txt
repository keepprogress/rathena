glast_01,195,273,3	script	胡金的追隨者#1::Hugin's follower#1	2_F_MAGICMASTER,{
	mes "[胡金的追隨者]";
	mes "我不能像胡金大師那樣穿梭時間與次元，但我可以假裝能。";
	next;
	mes "[胡金的追隨者]";
	mes "是什麼讓這座克雷斯特漢姆古城被如此強大的黑暗力量所籠罩？";
	next;
	if( select( "不在乎:很有趣，請多告訴我" ) == 2 ){
		mes "[胡金的追隨者]";
		mes "多虧胡金，我找到了進入這座被強大黑暗力量籠罩的舊克雷斯特漢姆古城的方法。";
		next;
		mes "[胡金的追隨者]";
		mes "你覺得如何？想去那裡探索嗎？";
		next;
		if( select( "我不去:我真的很想去探索！" ) == 2 ){
			mes "[胡金的追隨者]";
			mes "當然可以，但這可不是免費服務！我得拿點回報。呵呵……";
			next;
			mes "[胡金的追隨者]";
			mes "我可以送你去兩個地方。你想去哪裡？";
			next;
			switch( select( "取消:修道院墓地 (等級100~130) - 5000z:二樓走廊 (等級140以上) - 10000z" ) ){
				case 1:
					mes "[胡金的追隨者]";
					mes "好吧，改變主意隨時再來找我。我想我們會有很多話可以聊的……";
					close;
				case 2:
					.@map$ = "gl_chyard_";
					.@x = 148;
					.@y = 144;
					.@zeny = 5000;
					break;
				case 3:
					.@map$ = "gl_cas02_";
					.@x = 104;
					.@y = 31;
					.@zeny = 10000;
					break;
			}
			if( Zeny < .@zeny ){
				mes "[胡金的追隨者]";
				mes "什麼？你以為這項服務是免費的嗎？";
				close;
			}
			Zeny -= .@zeny;
			warp .@map$, .@x, .@y;
			end;
		}
	}
	mes "[胡金的追隨者]";
	mes "嗯，如果你沒興趣，就算了。";
	close;
}
