c_tower1,232,222,5	script	貝爾賈明·巴頓#ct::Belljamin Button#ct	968,{
	mes "[貝爾賈明·巴頓]";
	mes "這裡的怪物能夠操控時間。";
	if( BaseLevel < 150 ){
		mes "等你達到^0000FF150等^000000再回來吧。";
		close;
	}
	mes "時間、記憶、環境……這裡的一切都可能扭曲。";
	next;
	mes "[貝爾賈明·巴頓]";
	mes "我求你別再追問了。這可能會讓你丟掉性命。";
	next;
	.@menu = select("別問了:前往扭曲之塔二樓:前往扭曲之塔三樓");
	switch(.@menu){
		case 1:
			break;
		case 2:
		case 3:
			if( .@menu == 2 ){
				.@map$ = "c_tower2_";
				.@x = 268;
				.@y = 26;
			}else{
				.@map$ = "c_tower3_";
				.@x = 65;
				.@y = 147;
			}
			mes "[貝爾賈明·巴頓]";
			mes "選擇時請謹慎。";
			next;
			mes "[貝爾賈明·巴頓]";
			mes "讓你進去需要^0000FF10,000z^000000。那裡相當危險。";
			next;
			switch( select( "算了:支付10,000z:為什麼要錢？" ) ){
				case 1:
					break;
				case 2:
					if( Zeny < 10000 ){
						mes "[貝爾賈明·巴頓]";
						mes "真的嗎？你沒有錢，還要我帶你去那裡？";
						close;
					}
					Zeny -= 10000;
					mes "[貝爾賈明·巴頓]";
					mes "我不確定你在裡面要怎麼掌握時間，但還是請盡量小心。";
					close2;
					warp .@map$, .@x, .@y;
					end;
				case 3:
					mes "[貝爾賈明·巴頓]";
					mes "我整天待在這裡，就是為了冒險者的安全。我總是提醒他們危險。所以我需要一些報酬。";
					emotion ET_SWEAT;
					close;
			}
			break;
	}
	mes "[貝爾賈明·巴頓]";
	mes "呼……你做了個明智的選擇。這裡非常殘酷，你最好回到安全的地方。";
	close;
}
