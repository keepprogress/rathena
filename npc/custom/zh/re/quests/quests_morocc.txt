		close;

	next;
	mes "[可疑的貓]";

	mes "什麼~？別用那種眼神看我！你知道貓在Midgard要活下去有多難嗎！";

	mes "嗯，這不關我的事。不過我還是警告過你了！";

		if (Zeny < 5000) {
			mes "[可疑的貓]";
			mes "你想去哪！你窮成這樣！";
			close;
		}
		set Zeny, Zeny-5000;
		mes "[可疑的貓]";
		mes "嘖，這什麼舊錢……你沒有比較閃亮的東西嗎？！";

		end;
	case 2:
		mes "[可疑的貓]";
		mes "天啊，你連5000 Zeny都沒有吧？";
		close;
	}
}
