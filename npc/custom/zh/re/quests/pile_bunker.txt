yuno,179,174,4	script	戈梅爾::Gomer	851,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 1000) {
		mes "[戈梅爾]";
		mes "你帶了太多東西了。先把一些行李放下再回來找我吧。";
		close;
	}
	if (countitem(1549)) {
		mes "[戈梅爾]";
		mes "你的打樁機用得還順手嗎？";
		close;
	}
	if (Class != Job_Mechanic && Class != Job_Mechanic_T && Class != Job_Baby_Mechanic) {
		mes "[戈梅爾]";
		mes "我已經退休了，應該沒有什麼事需要跟你談。";
		close;
	}
	mes "[戈梅爾]";
	mes "怎麼樣？騎上魔導機甲是什麼感覺？";
	mes "那是沒能找到自己道路的人無法擁有的東西。";
	next;
	mes "[戈梅爾]";
	mes "我嗎？我很久以前就退休了。因為出了意外……受了重傷，再也騎不了了。我很難過。";
	next;
	mes "[戈梅爾]";
	mes "所以我決定留在這裡，為魔導機甲鍛造零件。";
	mes "咦？我可沒說對這份工作有什麼不滿喔。";
	next;
	mes "[戈梅爾]";
	mes "被迫去做不喜歡的事，沒有比這更悲慘的了。";
	next;
	mes "[戈梅爾]";
	mes "有些人只要當個支撐的後盾就很開心，你知道的，就像是讓別人靠的牆一樣。";
	next;
	mes "[戈梅爾]";
	mes "聽你這麼說我很高興。我正準備做新的零件，想請你幫忙。你覺得如何？";
	next;
	switch(select("沒興趣:你要做什麼?:材料從哪來?:我帶來材料了")) {
	case 1:
		mes "[戈梅爾]";
		mes "我也無能為力";
		mes "那就沒辦法了。";
		mes "下次見！";
		close;
	case 2:
		mes "[戈梅爾]";
		mes "嗯，是一根巨大的金屬棒，能一擊把對手打碎！";
		next;
		mes "[戈梅爾]";
		mes "雖然看起來不起眼，但我設計了一個彈簧裝置，能以很大的力道攻擊。";
		next;
		mes "[戈梅爾]";
		mes "它是隱藏式的，所以只要找到機會，一擊就能讓對手陷入瀕死狀態！";
		mes "真等不及看它實際運作了！";
		next;
		mes "[戈梅爾]";
		mes "如果你能想像這種武器有多好，我就會把它做出來。";
		next;
		mes "[戈梅爾]";
		mes "帶200個鋼鐵、30根軟管和1把魔力之矛來給我。有了這些材料，我就能做出剛才說的東西。";
		close;
	case 3:
		mes "[戈梅爾]";
		mes "你應該能從狗頭人等許多怪物身上取得鋼鐵。";
		next;
		mes "[戈梅爾]";
		mes "軟管比較稀有，會從艾音布羅克和里希塔樂附近的金屬蟲身上掉落。";
		next;
		mes "[戈梅爾]";
		mes "最後，魔力之矛很難找，但你應該有辦法。哈哈，去拿來吧！";
		close;
	case 4:
		if (countitem(999) < 200 || countitem(7325) < 30 || countitem(1415) < 1) {
			mes "[戈梅爾]";
			mes "你還沒把我要的材料全部帶來。";
			next;
			mes "[戈梅爾]";
			mes "我有個壞習慣，一有靈感就會放下手上的東西，去做新的東西。";
			mes "你最好快一點！";
			close;
		} else {
			mes "[戈梅爾]";
			mes "喔，我一直在等。你沒準時來，我還以為你忘了。請在那邊稍等一下。";
			specialeffect EF_REPAIRWEAPON;
			next;
			mes "[戈梅爾]";
			mes "做好了。你覺得如何？不錯吧？雖然是我自己做的，但我得說，做得相當不錯。";
			next;
			mes "[戈梅爾]";
			mes "重量、體積，還有色澤與光澤的搭配……讓我心動！";
			next;
			mes "[戈梅爾]";
			mes "...........";
			next;
			mes "[戈梅爾]";
			mes "嗯，稍加改良就能量產出貨了。我確信能再做出更多，所以這個就給你吧。";
			next;
			mes "[戈梅爾]";
			mes "喔，對了，我其實用不了那麼多鋼鐵。所以這些，大部分還你吧。";
			delitem 999,50; //Steel
			delitem 7325,30; //Tube
			delitem 1415,1; //Skewer
			getitem 1549,1; //Pilebuncker
			getitem 1360,1; //Two_Handed_Axe
			close;
		}
	}
}
