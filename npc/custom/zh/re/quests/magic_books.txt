function	script	Magic_Book_Essence	{
	if (MaxWeight - Weight < 1000) {
		mes "請減輕背包的重量。";
		close;
	}
	.@item_req = getarg(1);
	.@req_amount = getarg(2);
	if (checkquest(getarg(0)) == 1 && countitem(.@item_req) >= .@req_amount) {
		specialeffect2 EF_BLIND;
		specialeffect2 EF_BEGINSPELL;
		progressbar "ffff00",4;
		delitem .@item_req,.@req_amount;
		if (rand(2)) {
			mes "您使用了 "+ callfunc("F_InsertPlural",.@req_amount,getitemname(.@item_req)) +" 來製作^999900"+ getarg(3) +" 精華^000000。";
			next;
			mes "^008800此 "+getarg(3)+" 精華將永久保存，除非發生問題。^000000";
			completequest getarg(0);
			specialeffect2 EF_LORD;
			close;
		}
		mes "^008888精煉精華失敗，使用了 "+ callfunc("F_InsertPlural",.@req_amount,getitemname(.@item_req)) +".^000000";
		close;
	}
	if (checkquest(getarg(0)) == 2) {
		mes "您已在此區域製作了 "+getarg(3)+" 精華。";
		close;
	}
	mes "已經有人使用過這個區域了。";
	close;
}
