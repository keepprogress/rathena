			close2;
			delitem 6081,1; //6081
			percentheal 100,100;

			end;
		case 2:
			if (Zeny < 5000) {
				mes "[櫃檯人員]";
				mes "看來你的資源不足以繼續。可以麻煩你再確認一下帳戶嗎？";

			mes "請隨意休息～";

			percentheal 100,100;

			end;

			close;
		}
