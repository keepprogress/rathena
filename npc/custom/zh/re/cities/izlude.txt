// Channel Warper :: channel
//============================================================
function	script	F_IzludeChannel	{
	mes "[頻道傳送員]";
	mes "依斯魯得真是個熱鬧的地方，不是嗎？";
	mes "我在這裡，就是為了讓你在依斯魯得過得更順暢！";
	mes "我可以把你送到另一個依斯魯得分流。";
	mes "你想前往其他分流嗎？";
	next;
	set .@i, select(getarg(0)+":不用了");
	if (.@i < 6) {
		setarray .@maps$[1],"izlude","izlude_a","izlude_b","izlude_c","izlude_d";
		savepoint .@maps$[.@i],128,98,1,1;
		warp .@maps$[.@i],128,98;
		end;
	}
	close;
}
