	mes "[德瓦塔水手]";
	mes "你想前往^8B4513德瓦塔島^000000嗎？";
	mes "這裡波光粼粼、風景迷人，是個很棒的放鬆之地。";
	if (VIP_SCRIPT) {
		set .@cost, vip_status(VIP_STATUS_ACTIVE)?1000:10000;
		mes "交通費是 10,000 Zeny，VIP 則為 1,000 Zeny。";
	} else {
		set .@cost,10000;
		mes "交通費是 10,000 Zeny。";
	}
	next;
	switch(select("是!:否。")) {
	case 1:
		if (Zeny >= .@cost) {
			mes "[德瓦塔水手]";
			mes "^8B4513德瓦塔^000000是一個美麗而寧靜的島國。";
			mes "旅途愉快～";
			close2;
			set Zeny, Zeny - .@cost;
			warp "dewata",232,53;
			end;

			mes "[德瓦塔水手]";
			mes "小子，你沒有錢啊。";
			mes "等你錢包再鼓一點再來吧。";
			close;

		mes "[德瓦塔水手]";
		mes "準備好出發的時候，";
		mes "跟我說一聲～";
		close;
	}
}
