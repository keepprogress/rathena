ma_fild02,312,317,5	script	衛兵#buwaya_cave::Guard#buwaya_cave	570,{
	if (BaseLevel < 130) {
		mes "[衛兵]";
		mes "等級低於 ^ff0000130^000000 的人";
		mes "無法進入此地。";
		mes "這裡很危險。請回去吧。";
		close;
	}

	set .@party_id,getcharid(1);
	set .@md_name$,"Buwaya Cave";

	if (!.@party_id) {
		mes "[衛兵]";
		mes "寶箱巨鱷很危險。";
		mes "請先組成";
		mes "多人隊伍後再來。";
		close;
	}
	if (is_party_leader() == true) {
		mes "[衛兵]";
		mes "這裡很危險。請回去吧。";
		mes "如果你在這裡沒有要事，請回去吧。";
		next;
		while(1) {
			switch(select("為什麼這麼危險？:我是來獵殺寶箱巨鱷的。:結束對話。")) {
			case 1:
				mes "[衛兵]";
				mes "最近，發生了一樁案件，";
				mes "村民接連失蹤。";
				mes "調查了足跡之後， ";
				mes "我們查出巨大怪物寶箱巨鱷";
				mes "是罪魁禍首。";
				next;
				mes "[衛兵]";
				mes "已經派出士兵與傭兵去 ";
				mes "獵殺寶箱巨鱷，但從未";
				mes "完全成功過。";
				next;
				mes "[衛兵]";
				mes "寶箱巨鱷至今仍在綁架";
				mes "村民。";
				mes "請集結一些人手，除掉寶箱巨鱷。";
				next;
				break;
			case 2:
				set .@playtime, checkquest(4229,PLAYTIME);
				if (.@playtime == -1) {
					// fall through
				} else if (.@playtime == 0 || .@playtime == 1) {
					mes "[衛兵]";
					mes "寶箱巨鱷還躲著。";
					mes "就算現在進去，寶箱巨鱷也不會出來。 ";
					mes "請稍後再來。";
					close;
				} else {
					erasequest 4229;
					// fall through
				}
				if (instance_create(.@md_name$) < 0) {
					mes "[衛兵]";
					mes "隊伍名稱是…… "+getpartyname(.@party_id)+".";
					mes "隊長是…… "+strcharinfo(0);
					mes "^0000ff"+.@md_name$+"^000000 現在無法開啟。";
					mes "請稍後再試。";
					close;
				}
				mes "[衛兵]";
				mes "我會把通道打開到 ^0000ff"+.@md_name$+"^000000.";
				mes "祝你好運。";
				close;
			case 3:
				mes "[衛兵]";
				mes "這裡很危險，你不能進入。";
				close;
			}
		}
	} else {
		mes "[衛兵]";
		mes "讓我跟隊長談談。";
		mes "請把隊長叫來。";
		close;
	}
}

OnTouch:
	if (BaseLevel < 130) {
		mes "[衛兵]";
		mes "等級低於 ^ff0000130^000000 的人";
		mes "無法進入此地。";
		mes "這裡很危險。請回去吧。";
		close;
	}
	set .@playtime, checkquest(4229,PLAYTIME);
	if (.@playtime == -1) {
		// fall through
	} else if (.@playtime == 0 || .@playtime == 1) {
		mes "[衛兵]";
		mes "寶箱巨鱷還躲著。";
		mes "就算現在進去，寶箱巨鱷也不會出來。";
		mes "請稍後再來。";
		close;
	} else {
		erasequest 4229;
		// fall through
	}
	switch(select("進入:折返。")) {
	case 1:
		switch(instance_enter("Buwaya Cave")) {
		case IE_OTHER:
			mes "[衛兵]";
			mes "喔，現在不是個好時機。";
			mes "請稍後再試一次。";
			close;
		case IE_NOINSTANCE:
		case IE_NOMEMBER:
			mes "[衛兵]";
			mes "這裡很危險，請不要進入。";
			close;
		case IE_OK:
			mapannounce "ma_fild02",getpartyname(getcharid(1))+" 隊伍的 "+strcharinfo(0)+" 成員開始在寶箱巨鱷洞穴獵殺寶箱巨鱷。",bc_map,"0x00ff99"; //FW_NORMAL 12 0 0
			setquest 4229;
			//warp "1@ma_c",35,57;
			end;
		default:
			mes "- 因不明力量而無法進入。 -";
			close;
		}
	case 2:
		mes "[衛兵]";
		mes "你必須珍惜自己的生命。";
		mes "行動前最好仔細想清楚。";
		close;
	}
}
