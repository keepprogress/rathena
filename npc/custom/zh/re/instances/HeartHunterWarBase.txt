ein_fild04,275,342,5	script	突擊部隊指揮官#::Striker Unit Commander#	4_M_REBELLION3,{
	if (terra_gloria_main < 19) {
		mes "[突擊部隊指揮官]";
		mes "嘿，誰在那裡？";
		next;
		select("我正與反抗軍部隊一同執行任務。");
		mes "[突擊部隊指揮官]";
		mes "這裡很危險，你必須趕快離開。我們已經接近敵軍基地了。";
		close;
	}
	switch( checkquest(12375,PLAYTIME) ) {
	case -1:
		break;
	case 0:
	case 1:
		mes "[突擊部隊指揮官]";
		mes "我們正在整頓基地內部的情況。現在還不能讓你進去。";
		close;
	case 2:
		mes "^0000ff基地內部似乎已經整頓好了。 ^000000";
		erasequest 12375;// Base maintenance
		break;
	}
	if (isbegin_quest(12374) == 0) {
		mes "[突擊部隊指揮官]";
		mes "等行動準備就緒，你就會和我一起進去。";
		next;
		mes "[突擊部隊指揮官]";
		mes "對了，你有從朱利安將軍那裡領取任務簡報嗎？領取後再回來。";
		close;
	}
	mes "[突擊部隊指揮官]";
	mes "等行動準備就緒，你就會和我一起進去。";
	next;
	if (terra_gloria_main == 19)
		.@md_name$ = "Heart Hunter War Base 1";
	else
		.@md_name$ = "Heart Hunter War Base 2";
	switch( instance_enter(.@md_name$) ) {
	case IE_NOMEMBER:
		mes "^0000ff看來沒有隊伍的冒險者無法進入。^000000";
		close;
	case IE_NOINSTANCE:
		mes "[突擊部隊指揮官]";
		mes "作戰區域為 " + .@md_name$ + " 請確認作戰是否已正確申請。";
		next;
		mes "^ff0000未清除邪心獵人戰爭基地就前往其外圍邊界是被禁止的。此外，已清除的邪心獵人戰爭基地也無法再進入。請確認你的作戰據點^000000";
		next;
		mes "^ff0000這是指你的進度與隊長不同的情況。若是如此，請找其他能與你前往相同地點的隊員。^000000";
		next;
		mes "^ff0000若紀念副本尚未建立為作戰場地，也無法進入。^000000";
		close;
	case IE_OTHER:
		mes "發生了未知的錯誤。";
		close;
	case IE_OK:
		mapannounce "ein_fild04", "" + strcharinfo(0) + " 隊伍中的 " + getpartyname( getcharid(1) ) + "，正在進入 " + .@md_name$ + ".", bc_map, 0xFF99;
		setquest 12375;// Base maintenance
		// if (terra_gloria_main == 20)
		// 	warp "1@swat",35,65;
		// else
		// 	warp "1@swat",73,118;
		end;
	}
}

			mes "[朱利安]";
			mes "聽起來很帥。你就像外籍兵團的人，作戰時待在我身邊。";
		}
		else {
			mes "[朱利安]";
			mes "嗯，你還沒接受簡報嗎？時間不夠，我簡單跟你說明。";
			next;
			mes "[朱利安]";
			mes "我們的兵力不足以攻打基地。";
			next;
			mes "[朱利安]";
			mes "為了機動性，我們放棄了重型火器。由於首要任務是在不大幅破壞的前提下佔領基地，大部分爆炸性武器也都被排除在外。";
			next;
			mes "[朱利安]";
			mes "必要時可以利用戰場上的地形來加強火力……但必須降到最低。";
			next;
			mes "[朱利安]";
			mes "無論如何，作戰主要應以殺傷人員為主，並盡可能維持基地設施。";
			next;
			mes "[朱利安]";
			mes "先遣隊已控制南部基地的邊界面。我們將走那條路線前往中央區域。各突擊部隊則會各自走不同路線。";
			next;
			mes "[朱利安]";
			mes "你就像外籍兵團的人，作戰時待在我身邊。還有問題嗎？當然，就算你有，我也不會回答。";
		}
		next;
		setquest 12374;// Heart Hunter Base occupation
		mes "[朱利安]";
		mes "之後聽從你們隊長的指示。基地大門一開，就去找那邊的突擊隊指揮官開始行動。";
		if (is_party_leader() == true) {
			next;
			mes "[朱利安]";
			mes "準備好了的話，我會通知你攻擊時間。";
			next;
			if (select( "我準備好了。", "我還需要一點時間。" ) == 2) {

				mes "我知道了。既然時間緊迫，請盡快完成準備並通知我。";
				close;
			}
			break;
		}
		close;
	case 1:
		if (is_party_leader() == false) {
			mes "[朱利安]";
			mes "之後聽從你們隊長的指示。基地大門一開，就去找那邊的突擊隊指揮官開始行動。";
			close;

		mes "你去過廁所了嗎？我正在找你。";

			mes "之後聽從你們隊長的指示。基地大門一開，就去找那邊的突擊隊指揮官開始行動。";
			close;
		}

		mes "在我們佔領的基地外圍有一大片空地。我們在那裡蓋了訓練場，但設施還不完善。我需要能夠檢驗的精英人員。";
		next;
		mes "[朱利安]";
		mes "喔，這不是我第一次跟你提這件事嗎？如果你能幫忙檢驗設施，我會很高興。你覺得如何？";
		next;
		if (select( "讓我來做吧。", "下次再說吧……" ) == 2) {
			mes "[朱利安]";
			mes "我明白了。是我要求太多了。";
			close;
		}
		break;
	}

		instance_create("Heart Hunter War Base 1");
	else
		instance_create("Heart Hunter War Base 2");
	mes "[朱利安]";
	mes "很快就會開了。基地大門一開，就去找那邊的突擊隊指揮官開始行動。";
	close;
}
