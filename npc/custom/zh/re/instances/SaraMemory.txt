dali,139,118,4	script	次元裝置#sara::Dimensional Device#sara	10007,{
	if (BaseLevel < 99) {
		mes "[冒險者里昂]";
		mes "喂！你還太年輕，不能玩那東西！等你至少達到^0000FF等級99^000000再回來！";
		close;
	}
	if ((saramemory < 1) || (saramemory == 2) || (saramemory == 3) || (saramemory == 5)) {
		mes "[冒險者里昂]";
		mes "在你亂玩那東西之前，何不先過來跟我聊聊？";
		close;
	}

	// 20 hour cooldown
	set .@ins_sara_time,checkquest(15002,PLAYTIME);
	if (.@ins_sara_time == -1) {

		set .@party_id,getcharid(1);
		set .@p_name$,getpartyname(.@party_id);
		set .@md_name$,"Sara's Memories";

		if (!.@party_id) {
			mes "[冒險者里昂]";
			mes "進入之前你得先組隊！你不知道怎麼做嗎？請輸入 /organize PARTYNAMEHERE。";
			close;
		}

		if (is_party_leader() == true)
			set .@menu$, "Boot up the dimensional device.:Use the dimensional device.:Cancel.";
		else
			set .@menu$, ":Use the dimensional device.:Cancel.";

		switch(select(.@menu$)) {
		case 1:
			if (instance_create(.@md_name$) < 0) {
				mes "隊伍名稱: "+.@p_name$;
				mes "隊長: "+strcharinfo(0);
				mes "^0000ff"+.@md_name$+" ^000000- 預約失敗！";
				close;
			}
			mes "^FF0000次元裝置已順利啟動。請使用該裝置進入時空裂縫。^000000";
			close;
		case 2:
			switch(instance_enter(.@md_name$)) {
			case IE_OTHER:
				mes "發生了未知的錯誤。";
				close;
			case IE_NOINSTANCE:
				mes "紀念地城 "+.@md_name$+" 不存在。";
				mes "隊長尚未生成副本。";
				close;
			case IE_NOMEMBER:
				mes "只有登記的成員才能進入副本 "+.@md_name$+".";
				close;
			case IE_OK:
				mapannounce "dali",strcharinfo(0)+"，隊伍成員 "+.@p_name$+" 已進入副本 "+.@md_name$+".",bc_map,"0x00ff99";
				setquest 15002;
				end;
			}
		case 3:
			mes "^FF0000隊長沒有啟動次元裝置。^000000";
			close;
		}
	} else if (.@ins_sara_time == 0 || .@ins_sara_time == 1) {
		mes "[冒險者里昂]";
		mes "我想在你上次進入裂縫後大約二十小時，它就會穩定到足以讓你再次穿越。";
		close;
	}
	else {
		erasequest 15002;
		mes "^0000FF[ 莎拉的記憶 ] 冷卻時間已清除，現在可以再次使用次元裝置了。^000000";
		close;
	}
}
