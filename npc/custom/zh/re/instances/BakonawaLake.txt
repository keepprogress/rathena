ma_scene01,174,179,4	script	塔霍::Taho	541,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "你攜帶的道具太多了。請先減少一些再來。";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "你攜帶的重量太重了。請先減輕負重再來。";
		close;
	}
	if (BaseLevel < 140) {
		mes "[塔霍]";
		mes "喂。這裡對你來說太危險了，你沒辦法保護自己。快離開這裡，立刻去村子！";
		close;
	}
	if (malaya_bakona1 < 7 && malaya_bakona2 < 15) {
		mes "[塔霍]";
		mes "我們都會被食月暴龍殺掉的。啊，我們必須討伐牠，但很遺憾我們還沒準備好。";
		close;
	}

	set .@party_id,getcharid(1);
	set .@md_name$, "Bakonawa Lake";

	set .@baku_time, checkquest(12278,PLAYTIME);
	if (.@baku_time == -1) {
		if (checkquest(12279) == -1) setquest 12279;
	}
	if (!.@party_id) {
		mes "[塔霍]";
		mes "沒有能跟你同行的夥伴嗎？就算你想單獨行動，也請先組好隊伍再來。";
		close;
	}
	if (.@baku_time == -1) {
		if (is_party_leader() == true) {
			mes "[塔霍]";
			mes "你是食月暴龍討伐隊的隊長嗎？路況不好，我們得用繩子下懸崖，可以嗎？";
			next;
			set .@i, select("請編繩索:現在就下去:取消");
		} else {
			mes "[塔霍]";
			mes "你是來討伐食月暴龍的吧。隊長必須幫我編繩索，才能下去。";
			next;
			set .@i, select(":現在就下去:取消");
		}
		switch(.@i) {
		case 1:
			if (instance_create(.@md_name$) < 0) {
				mes "[塔霍]";
				mes "哎呀，繩子掉下去了。我得重新做一條。";
				close;
			}
			mes "[塔霍]";
			mes "我正在編，編好了你就可以下去了。";
			close;
		case 2:
			switch(instance_enter(.@md_name$)) {
			case IE_OTHER:
				mes "[塔霍]";
				mes "發生未知的錯誤。";
				close;
			case IE_NOINSTANCE:
				mes "[塔霍]";
				mes "繩索還沒編好，請稍等一下。";
				close;
			case IE_NOMEMBER:
				mes "[塔霍]";
				mes "平民不得進入。";
				close;
			case IE_OK:
				mapannounce "ma_scene01","隊伍成員 "+strcharinfo(0)+" 所屬隊伍 "+getpartyname(.@party_id)+" 正在進入地城， "+.@md_name$+".",bc_map,"0x00ff99"; //FW_NORMAL 12 0 0
				setquest 12278;
				//warp "1@ma_b",64,51;
				close;
			}
		case 3:
			close;
		}
	} else if (.@baku_time == 0 || .@baku_time == 1) {
		mes "[塔霍]";
		mes "通往湖泊的繩索已經斷了。我們正在清理周圍，請過一段時間再來。";
		close;
	} else if (.@baku_time == 2) {
		mes "[塔霍]";
		mes "這附近已經清理完畢了。現在取得繩索後就能再次下去了。";
		erasequest 12278;
		close;
	}
}
