prt_q,243,75,3	script	要塞入口傳送門#1::Fortress Entry Warp Portal#1	PORTAL,{
	if (BaseLevel < 145) {
		mes "- 你的等級必須達到145或以上";
		mes "才能前進。 -";
		close;
	}
	if (checkquest(9419,PLAYTIME) == 2 || (checkquest(9419,PLAYTIME) == -1 && isbegin_quest(9418) != 1)) {
		setarray .@menu$[0], "Don't step into the warp.", "Step into the warp.";
		.@type = 1;
	}
	else {
		mes "- 傳送門似乎無法運作。";
		mes "你似乎無法進入…… -";
		next;
		setarray .@menu$[0], "End the dialogue.", "", "Put the - Dungeon Pass - near to the warp.";
	}
	switch( select( .@menu$[0], .@menu$[1], .@menu$[2] ) ) {
	case 1:
		mes "- 傳送門正散發出耀眼的光芒。 -";
		close;
	case 2:
		break;
	case 3:
		if (rentalcountitem(14505) < 1 && rentalcountitem(14506) < 1) {// Sky Fortress Ticket
			mes "- 你沒有 -";
			mes "- 地城通行證。 -";
			close;
		}
		break;
	}
	.@md_name$ = "Sky Fortress Invasion";
	switch( instance_enter(.@md_name$) ) {
	case IE_OTHER:
		mes "發生未知的錯誤。";
		close;
	case IE_NOINSTANCE:
		mes "紀念地城 " + .@md_name$ + " 不存在。";
		mes "隊長尚未建立紀念地城。";
		close;
	case IE_NOMEMBER:
		mes "只有隊伍成員可以進入紀念地城。";
		close;
	case IE_OK:
		mapannounce "prt_q", "隊員 " + strcharinfo(0) + " 所屬的隊伍 " + getpartyname( getcharid(1) ) + " 進入了 " + .@md_name$ + "", bc_map,0xFF99;
		if (.@type == 1) {
			if (isbegin_quest(9419) > 0)
				erasequest 9419;
			if (isbegin_quest(9418) > 0)
				erasequest 9418;
			setquest 9418;// Attack Sky Fortress Invading Prontera
		}
		fortress_entrance_loc = 0;	// warp back to prt_q on exit
		// warp "1@sthb",54,67;
		end;
	}
	end;

OnInit:
	while(true) {
		sleep 10000;
		specialeffect EF_ENHANCE;
	}
	end;
}
