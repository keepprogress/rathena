		case -1:
		case 2:

		mes "你持有的物品種類太多了，請清理背包。";
		close;

		mes "你攜帶的物品太重，無法繼續。請清理背包以減輕總重量。";
		close;

		mes "[禮品包裝員]";
		mes "需要包裝服務時隨時再來！";
		close;
	}
	Zeny -= .@zeny;
	delitem .@item,5;
	getitem .@box,1;
	end;
