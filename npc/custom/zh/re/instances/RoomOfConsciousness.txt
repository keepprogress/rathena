1@mir,103,80,3	script	Fenrir#boss1a	4_F_FENRIR,2,2,{
	end;
OnEvent:
	'fenrir_name$ = instance_npcname( strnpcinfo(0) );
	'iris_name$ = instance_npcname( "Iris#" + strnpcinfo(2) );
	enablenpc 'fenrir_name$;
	enablenpc 'iris_name$;
	end;

OnTouchNPC:
	if ('touch_mob == 0) {
		npctalk "潘利爾: 幹得好！接下來就靠你了，菖蒲！", 'fenrir_name$;
		'touch_mob = 1;
		initnpctimer;
	}
	end;
OnTimer2000:
	npctalk "菖蒲: 好，交給我！我菖蒲想要的是……！", 'iris_name$;
	specialeffect EF_BEGINSPELL,AREA, 'iris_name$;
	end;
OnTimer4000:
	npctalk "菖蒲: 魔比返魂符！去吧！！", 'iris_name$;
	if ('target_event == 1)
		unittalk 'boss_id, "啊啊啊——！！";
	else
		unittalk 'boss_id, "畢尤: 怎麼回事……！！";
	end;
OnTimer5000:
	npctalk "菖蒲: 成功了！", 'iris_name$;
	end;
OnTimer7000:
	npctalk "潘利爾: 那就輪到我了。~振奮的吶喊~", 'fenrir_name$;
	specialeffect EF_BEGINSPELL,AREA, 'fenrir_name$;
	specialeffect EF_TETRACASTING,AREA, 'fenrir_name$;
	progressbar_npc "000000",3;
	end;
OnTimer10000:
	npctalk "潘利爾: 接招吧！！", 'fenrir_name$;
	specialeffect EF_SUI_EXPLOSION,AREA, 'fenrir_name$;
	specialeffect EF_LORD,AREA, 'fenrir_name$;
	specialeffect EF_FLAMELAUNCHER,AREA, 'fenrir_name$;
	if ('target_event == 1)
		unittalk 'boss_id, "~尖叫~";
	else
		unittalk 'boss_id, "畢尤: ~呻吟~ 比我想的還痛……！";
	getunitdata 'boss_id, .@data;
	if (.@data[UMOB_HP] > 0) {
		.@damage = rand(700,1300) * 1000;
		if (.@damage >= .@data[UMOB_HP])
			.@mob_hp = 0;
		else
			.@mob_hp = .@data[UMOB_HP] - .@damage;
		setunitdata 'boss_id, UMOB_HP, .@mob_hp;
	}
	end;
OnTimer13000:
	npctalk "潘利爾: 雖然不完美，但我想我們已造成相當可觀的傷害！", 'fenrir_name$;
	end;
OnTimer14500:
	npctalk "潘利爾: 我消耗太多魔力了，得稍微休息一下。那就靠你們了！", 'fenrir_name$;
	end;
OnTimer16500:
	donpcevent instance_npcname("eq#mir3") + "::OnStart";
OnStop:
	stopnpctimer;
	disablenpc();
	disablenpc instance_npcname( "Iris#" + strnpcinfo(2) );
	'touch_mob = 0;
	'random_letter$ = "";
	end;
}
