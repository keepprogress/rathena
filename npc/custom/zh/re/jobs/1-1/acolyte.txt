				mes "[馬里烏斯神父]";
				mes "啊，我感覺你已經承受過";
				mes "前世的經歷。你在進入英靈殿前一定學到了很多事情。";

				mes "從英靈殿歸來，";
				mes "祝你";

				next;
				skill "NV_TRICKDEAD",0,SKILL_PERM;
				jobchange Job_Acolyte_High;
				skill "AL_HOLYLIGHT",1,SKILL_PERM;
				mes "[馬里烏斯神父]";
				mes "現在，出發吧，去尋找需要你幫助的人。願神照亮你的道路。";
				close;
			}else{
				mes "[馬里烏斯神父]";
				mes "現在，出發去尋找需要你幫助的人吧。願神指引你的道路。";
				close;
			}
		}else{
			mes "[馬里烏斯神父]";
			mes "我感覺你已經經歷過前世。你在進入英靈殿之前一定學到了很多東西。";
			next;
			mes "[馬里烏斯神父]";
			mes "不過，我看得出來你並不適合成為修道士。請記住你前世的身分，找到屬於你的道路。";
			close;
		}
	}
	mes "[馬里烏斯神父]";
	mes "你來這裡所尋求的是什麼？";
	next;
	switch(select("轉職為修道士。:詢問修道士的轉職條件。:算了。")) {
	case 1:
		mes "[馬里烏斯神父]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Acolyte)
				mes "你今天還好嗎？從你的服裝看來，你已經是修道士了。你不是在開玩笑吧？";
			else
				mes "很抱歉，你似乎已經有自己的職業了，不是嗎？";
			close;
		}else{
			mes "你真心想成為神的僕人嗎？";
			mes "讓我看看你是否準備好了……嗯……";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[馬里烏斯神父]";
				mes "哎呀？！你還沒完成基礎修練？！路還很遠呢！等你提升職業等級後再來吧！";
				close;
			}
			mes "[馬里烏斯神父]";
			mes "嗯……你的職業等級已經足夠了……";
			mes "很好。現在我將授予你成為修道士的資格。";
			next;
			mes "[馬里烏斯神父]";
			mes "請永遠記得感謝時時照顧我們的神。在混亂與困難的時刻，請以堅定的信仰面對你的磨難。";
			callfunc "Job_Change",Job_Acolyte;
			getitem 1545,1;	// N_Mace
			next;
			mes "[馬里烏斯神父]";
			mes "最後，我要誠摯地恭喜你堅持完成了你的苦修試煉。";
			close;
		}
	case 2:
		mes "[馬里烏斯神父]";
		mes "你想成為修道士嗎？";
		mes "那麼，你必須把以下的條件當作神所賜予的修練來完成。";
		next;
		mes "[馬里烏斯神父]";
		mes "首先，你必須達到初心者職業等級 9，並學會所有基本技能。";
		mes "這是最基本的要求，所以你必須把它當作自我修練的方式。";
		next;
		mes "[馬里烏斯神父]";
		mes "當你認為已經完成這項要求時，就回來找我。屆時你將擁有一個能夠傳播神的旨意的神聖職業。";
		close;
	case 3:
		close;
	}
}
