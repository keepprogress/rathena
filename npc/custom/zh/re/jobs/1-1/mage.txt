			if (Class == Job_Novice_High) {
				mes "[魔法師公會]";
				mes "哇，好久不見！不過你不是應該已經死了嗎？";
				next;
				mes "[魔法師公會]";
				mes "啊，你一定是轉生了。嗯，很高興你回來了。";
				next;
				if (!callfunc("F_CanChangeJob")) {
					mes "[魔法師公會]";
					mes "很抱歉，我想你還沒準備好學習魔法。何不先去學完基本技能呢？";
					next;
					mes "[魔法師公會]";
					mes "慢慢來。你學得越多，就越能準備好再次學習魔法。";
					close;
				}
				mes "[魔法師公會]";
				mes "嗯，既然你曾通過過魔法師的測驗，我就不會質疑你的資格。你想立刻恢復你的魔法技能，對吧？";
				next;
				skill "NV_TRICKDEAD",0,SKILL_PERM;
				jobchange Job_Mage_High;
				skill "MG_ENERGYCOAT",1,SKILL_PERM;
				mes "[魔法師公會]";
				mes "哇，不知為何，你看起來比之前好多了。總之，我相信你當魔法師也會做得更好。";
				close;
			}else{
				mes "[魔法師公會]";
				mes "還有什麼我能幫你的嗎？如果沒有，何不去測試一下你的技能呢？世界正在等著你~！";
				close;
			}
		}else{
			mes "[魔法師公會]";
			mes "什麼，你對魔法師公會有興趣嗎？我本不想告訴你，但你並不屬於這裡。";
			next;
			mes "[魔法師公會]";
			mes "不太明白你為什麼還站在我面前，不過我看得出來你不適合當魔法師。";
			close;
		}
	}
	mes "[魔法師公會]";
	if(BaseJob != Job_Novice) {
		if(BaseJob == Job_Mage) {
			mes "嘿，你還沒發現嗎？你早就是魔法師了，傻瓜！";
			mes "總有一天你會意識到自己體內的力量，到時候你就能用意念製造火焰！";
		} else
			mes "嘿～別鬧了，別再玩遊戲。你不能當魔法師，因為你已經有其他職業了。";
		close;
	}else{
		mes "嘿？";
		next;
		switch(select("我想成為魔法師:成為魔法師需要什麼條件？:沒什麼，謝謝。")) {
		case 1:
			mes "[魔法師公會]";
			mes "想當魔法師？唔……";
			next;
			mes "[魔法師公會]";
			if (Sex == SEX_MALE) {
				mes "嘿，看看你！你還挺可愛的～！不過不是我的菜……";
			}else{
				mes "喔喔，你真是個大美女～！";
				mes "我喜歡像你這樣的女孩～";
			}
			mes "對了，你說你想當魔法師？";
			next;
			if(select("我想成為魔法師。:沒什麼，謝謝。") == 2) {
				mes "[魔法師公會]";
				mes "什麼～？！你剛說想當魔法師，馬上就改變主意了？！做事乾脆一點！";
				close;
			}
			mes "[魔法師公會]";
			if (!callfunc("F_CanChangeJob")) {
				mes "喔，老兄，你的基本技能等級還不夠當魔法師。";
				mes "回去把你的基本技能練到更高等級吧。";
				close;
			}
			mes "好吧……在申請表上簽名吧……哇～你的字寫得真漂亮！啊，好了。夠了。";
			mes "你是…… "+strcharinfo(0)+".";
			next;
			mes "[魔法師公會]";
			mes "嗯，我看得出你用自己的方式很努力。雖然看起來有點笨拙，不過我想還行！";
			mes "很好！紮實練好基本功才是最棒的！我馬上幫你轉職。";
			next;
			mes "[魔法師公會]";
			mes "哈啊……！你現在是魔法師了，是我們的同伴之一！";
			mes "歡迎你加入魔法師公會，我們的新朋友！";
			next;
			mes "[魔法師公會]";
			mes "「歡迎來到魔法師公會～」";
			mes "恭喜你成為魔法師公會的一員！加油！";
			callfunc "Job_Change",Job_Mage;
			getitem 1639,1;	// N_Rod
			close;
		case 2:
			mes "[魔法師公會]";
			mes "想當魔法師，是吧？";
			mes "像你這樣的美女，我很樂意說明條件！";
			next;
			mes "[魔法師公會]";
			mes "首先，你必須將新手職業等級練到10，並學會所有基本技能。";
			next;
			mes "[魔法師公會]";
			mes "以前有一道複雜的藥水製作測驗。因為這樣，我們的申請者一直在流失。";
			mes "所以，我們決定接受所有符合基本條件的申請者。";
			next;
			mes "[魔法師公會]";
			mes "別猶豫，去當魔法師吧！";
			close;
		case 3:
			mes "[魔法師公會]";
			mes "沒什麼……？";
			close;
		}
	}
}
