moc_prydb1,39,129,2	script	盜賊嚮導#thief::Thief Guide#thief	69,{
	if (Upper == 1) {
		if (ADVJOB == Job_Assassin_Cross || ADVJOB == Job_Stalker) {
			if (Class == Job_Novice_High) {
				mes "[盜賊嚮導]";
				mes "咦？我認識你嗎？你看起來好眼熟，真讓人毛骨悚然。你該不會有個雙胞胎吧？";
				next;
				if (!callfunc("F_CanChangeJob")) {
					mes "[盜賊嚮導]";
					mes "什麼，你想當盜賊？抱歉，你看起來還需要多加訓練。";
					next;
					mes "[盜賊嚮導]";
					mes "慢慢來，把所有基本技能都學會吧，好嗎？那就待會見囉～！";
					close;
				}
				mes "[盜賊嚮導]";
				mes "嗯，我感覺你好像經歷過一輩子的戰鬥，所以我現在就幫你轉職成盜賊。看來我最好給你們這些硬漢想要的……";
				next;
				skill "NV_TRICKDEAD",0,SKILL_PERM;
				jobchange Job_Thief_High;
				skill "TF_SPRINKLESAND",1,SKILL_PERM;
				skill "TF_BACKSLIDING",1,SKILL_PERM;
				skill "TF_PICKSTONE",1,SKILL_PERM;
				skill "TF_THROWSTONE",1,SKILL_PERM;
				mes "[盜賊嚮導]";
				mes "既然你已經成為盜賊，就做個盜賊吧。好了，加油去吧！下一位～";
				close;
			}else{
				mes "[盜賊嚮導]";
				if (Sex == SEX_MALE)
					mes "喂，老兄。";
				else
					mes "嘿，寶貝～";
				close;
			}
		}else{
			mes "[盜賊嚮導]";
			if (Sex == SEX_MALE)
				mes "喂，老兄。";
			else
				mes "嘿，寶貝。";
			mes "……嘿！你看起來太乖了，不適合當盜賊！！快滾，我很忙。下一位！";
			close;
		}
	}
	if(q_job_thief == 1) {
		mes "[盜賊公會成員]";
		mes "咦……看起來像是富貴人家心愛的 "+(Sex == SEX_MALE?"son":"daughter")+" 為什麼會來這種骯髒的地方？";
		next;
		mes "[盜賊公會成員]";
		mes "嗯，讓你成為盜賊不是我負責的。我只負責收申請書，懂了吧？";
		mes "想當盜賊的話，去問我旁邊那個眼神銳利的傢伙。";
		close;
	}
	mes "[盜賊公會成員]";
	if(BaseJob != Job_Novice) {
		if(BaseJob == Job_Thief) {
			mes "嘿～有什麼麻煩隨時跟我說，好嗎？";
			close;
		}else{
			mes "搞什麼啊……？";
			if(Sex == SEX_MALE) {
				mes "哼……你這個 "+jobname(Class)+"，笨蛋！";
				next;
				mes "[盜賊公會成員]";
				mes "嘿，兄弟。";
			}else{
				next;
				mes "[盜賊公會成員]";
				mes "嘿，美女。";
			}
			mes "你怎麼會在這裡？回你的地方去～快回去～～";
			close;
		}
	}
	mes "咦……看起來像是富貴人家心愛的 "+(Sex == SEX_MALE?"son":"daughter")+" 為什麼會來這種骯髒的地方？";
	next;
	if(select("我想當盜賊。:沒什麼。") == 2) {
		mes "[盜賊公會成員]";
		mes "你知道沒有申請書是不能當盜賊的吧……";
		mes "你在想什麼……？";
		close;
	}
	mes "[盜賊公會成員]";
	mes "嗯，你有那麼得意嗎？";
	mes "你這麼大聲地跟我說你想當盜賊！怎麼不去每個村子裡宣傳一下，說你要當小偷？";
	mes "「哈哈哈！去掛個布條，上面寫著『我要當個以偷別人東西為榮的盜賊。』」";
	next;
	mes "[盜賊公會成員]";
	mes "你真的那麼想當盜賊嗎？";
	next;
	switch(select("是。:不是。:你呢？")) {
	case 1:
		mes "[盜賊公會成員]";
		mes "喔，是嗎……？哼……嗯……我倒是活到老學到老，見過像你這樣的怪人。";
		next;
		break;
	case 2:
		mes "[盜賊公會成員]";
		mes "那你還來做什麼？你以為當盜賊這麼容易嗎？";
		next;
		break;
	case 3:
		mes "[盜賊公會成員]";
		mes "咦……？我？我？";
		mes "嗯……我天生就適合當盜賊……個性上……這種無聊的事我不在意。";
		next;
		break;
	}
	mes "[盜賊公會成員]";
	mes "總之，在外面可千萬別說你想當盜賊！！";
	next;
	mes "[盜賊公會成員]";
	mes "那麼，你要申請當盜賊嗎？";
	next;
	if(select("是，我要。:不要。") == 2) {
		mes "[盜賊公會成員]";
		mes "嗯……你想做什麼就做吧～隨你的意～";
		close;
	}
	mes "[盜賊公會成員]";
	mes "好，告訴我你的名字。咦？什麼？";
	mes ""+strcharinfo(0)+"。所以 "+strcharinfo(0)+"，對吧……？";
	next;
	if (!callfunc("F_CanChangeJob")) {
		mes "[盜賊公會成員]";
		mes "我看得出你想當盜賊的意志很堅定……";
		mes "不過光有意志，你在真正的戰鬥中可是撐不下去的，對吧？";
		mes "所以去把基本技能練到至少9級吧。";
		close;
	}
	mes "[盜賊公會成員]";
	mes "嗯……我從偵探社的資料裡找到了你有趣的犯罪紀錄。嗯，你看起來很有當好盜賊的資質。";
	mes ""+strcharinfo(0)+"。我現在宣布你通過了面試！";
	next;
	mes "[盜賊公會成員]";
	mes "嗯，讓你成為盜賊不是我負責的。我只負責收申請書，懂了吧？";
	mes "想當盜賊的話，去問我旁邊那位。";
	set q_job_thief,1;
	close;
}

	}
