function	script	F_KO_Survival_dice	{
	mes "位置: " + getarg(0);
	mes "總 LP: " + job_kagero_lp;
	mes "擲骰子需要消耗 " + getarg(1,1) + " ^FD0260LP^000000。";

		mes "因為^FD0260LP^000000不足，你將被送回起點。";
		callfunc "F_KO_Survival_warp",0;
		end;
	}
	set job_kagero_lp, job_kagero_lp - getarg(1,1);
	set .@dice, rand(1,6);
	emotion ET_OTL + .@dice, playerattached();
	mes "骰子點數為 " + .@dice + ".";
	if (getarg(2,0))
		return .@dice;
	else {
		mes "按下關閉前進 " + .@dice + " 格。";
		callfunc "F_KO_Survival_warp", .@dice + getarg(0);
		end;
	}
}

//Return: 1:win, 0:draw, -1:lose
function	script	F_KO_Survival_rps	{
	setarray .@rps$[1], "Scissors", "Rock", "Paper";
	setarray .@emote[1], ET_SCISSOR, ET_ROCK, ET_WRAP;  //might be iRO custom(?)

	mes "剪刀！石頭！布！";
	next;
	set .@pc, select("出剪刀。:出石頭。:出布。");
	set .@npc, rand(1,3);
	emotion .@emote[.@pc], playerattached();
	emotion .@emote[.@npc];
	mes "=遊戲結果=";
	mes "機器: " + .@rps$[.@npc] + ".";
	mes "你: " + .@rps$[.@pc] + ".";
	next;
	if ((.@npc == 1 && .@pc == 2) || (.@npc == 2 && .@pc == 3) || (.@npc == 3 && .@pc == 1)) {
		mes "你贏了";
		return 1;
	} else if (.@pc == .@npc) {
		mes "平手";
		return 0;
	} else if ((.@npc == 1 && .@pc == 3) || (.@npc == 2 && .@pc == 1) || (.@npc == 3 && .@pc == 2)) {
		mes "你輸了";
		return -1;
	} else {
		mes "發生了未知的錯誤。";
		mes "請聯絡客服。";
		close;
	}
}

			mes "武器測驗已設為完成。希望你已完成初始化。";
			break;
		}
		close;
	case 3:
		mes "我會把你的任務狀態改為轉職前，不過在那之前……";
		next;
		mes "請選擇一把武器。";
		next;
		switch(select("大駒短劍:飛鏢")) {
			case 1: set .@quest,5143; break;
			case 2: set .@quest,5144; break;
		}
		setquest 5136;
		completequest 5136;
		setquest 5137;
		completequest 5137;
		setquest 5138;
		completequest 5138;
		setquest .@quest;
		completequest .@quest;
		setquest 5146;
		set job_kagero,8;
		mes "已設定為轉職前狀態。祝你好運！！";
		close;
	case 4:
		set job_kagero,0;
		set job_kagero_lp,0;
		for (set .@i,5131; .@i<=5146; set .@i,.@i+1)
			if (isbegin_quest(.@i)) erasequest .@i;
		mes "初始化作業已完成。你可以透過[目前狀態]選項確認！";
		close;
	}
	end;
